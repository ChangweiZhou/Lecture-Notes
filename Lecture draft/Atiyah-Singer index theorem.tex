\chapter{Geometric preliminaries}

Throughout, \(M\) is a smooth compact manifold without boundary unless
stated otherwise. All vector bundles are finite-dimensional and smooth.

\section{Connections and curvature}

\begin{definition}
A connection on a real or complex vector bundle \(E\to M\) is an
\(\mathbb R\)-linear map
\[
 \nabla:\Gamma(E)\longrightarrow\Omega^1(M;E)
\]
satisfying
\[
 \nabla(fs)=df\otimes s+f\nabla s.
\]
For a vector field \(X\), write \(\nabla_Xs=(\nabla s)(X)\).
\end{definition}

In a local frame \(e=(e_1,\ldots,e_r)\), a section is a column vector and
\[
 \nabla=d+\omega,\qquad
 \omega\in\Omega^1(U;\operatorname{End}E).
\]
If the frame changes by \(e'=eg\), then
\[
 \omega'=g^{-1}\omega g+g^{-1}dg.
\]
This transformation law is exactly what makes the definition independent
of a frame.

The connection extends to \(E\)-valued forms by
\[
 d_\nabla(\alpha\otimes s)
 =
 d\alpha\otimes s+(-1)^{\deg\alpha}\alpha\wedge\nabla s.
\]

\begin{definition}
The curvature is
\[
 F^\nabla=d_\nabla^2\in\Omega^2(M;\operatorname{End}E).
\]
In a local frame,
\[
 F^\nabla=d\omega+\omega\wedge\omega.
\]
\end{definition}

Because \(d_\nabla^2\) is \(C^\infty(M)\)-linear, it is an
endomorphism-valued two-form. The graded Jacobi identity gives the Bianchi
identity
\[
 d_\nabla F^\nabla=0.
\]

\begin{question}
Why is a connection not tensorial while its curvature is?
\end{question}

\begin{answer}
The Leibniz rule contains the derivative \(df\), so
\(\nabla(fs)\neq f\nabla s\). Applying the connection twice makes the
first-derivative terms cancel:
\[
 d_\nabla^2(fs)
 =d_\nabla(df\otimes s+f\nabla s)
 =-df\wedge\nabla s+df\wedge\nabla s+f\,d_\nabla^2s.
\]
Thus \(d_\nabla^2(fs)=f\,d_\nabla^2s\). This calculation is the intrinsic
reason the inhomogeneous term \(g^{-1}dg\) in the transformation law for
\(\omega\) disappears from the transformation law
\(F'=g^{-1}Fg\).
\end{answer}

If \(E\) is Hermitian, a connection is unitary when
\[
 X\langle s_1,s_2\rangle
 =
 \langle\nabla_Xs_1,s_2\rangle+\langle s_1,\nabla_Xs_2\rangle.
\]
In a unitary frame, its connection form is skew-Hermitian and its curvature
is skew-Hermitian.

\section{The Levi--Civita connection}

Let \(g\) be a Riemannian metric. A connection on \(TM\) is metric if
\(\nabla g=0\), and its torsion is
\[
 T(X,Y)=\nabla_XY-\nabla_YX-[X,Y].
\]

\begin{theorem}[Levi--Civita]
There is a unique metric connection on \(TM\) with zero torsion.
It is determined by the Koszul formula
\[
\begin{aligned}
 2g(\nabla_XY,Z)
 ={}&
 Xg(Y,Z)+Yg(Z,X)-Zg(X,Y)\\
 &-g(X,[Y,Z])+g(Y,[Z,X])+g(Z,[X,Y]).
\end{aligned}
\]
\end{theorem}

\begin{proof}
Metric compatibility expresses \(Xg(Y,Z)\), \(Yg(Z,X)\), and
\(Zg(X,Y)\) through covariant derivatives. Add the first two expressions,
subtract the third, and use zero torsion to obtain the displayed formula.
It proves uniqueness. Conversely, the formula defines \(\nabla_XY\);
direct substitution verifies the connection rules, metric compatibility,
and zero torsion.
\end{proof}

The Riemann curvature tensor is
\[
 R(X,Y)Z
 =
 \nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z.
\]

With \(R(X,Y,Z,W)=g(R(X,Y)Z,W)\), it satisfies
\[
\begin{aligned}
 R(X,Y,Z,W)&=-R(Y,X,Z,W),\\
 R(X,Y,Z,W)&=-R(X,Y,W,Z),\\
 R(X,Y,Z,W)&=R(Z,W,X,Y),
\end{aligned}
\]
together with the first Bianchi identity
\[
 R(X,Y)Z+R(Y,Z)X+R(Z,X)Y=0.
\]
The Ricci tensor and scalar curvature are the contractions
\[
 \operatorname{Ric}(X,Y)=\sum_j g(R(e_j,X)Y,e_j),
 \qquad
 \operatorname{scal}=\sum_j\operatorname{Ric}(e_j,e_j).
\]

\section{Chern--Weil forms}

Let \(f\) be an invariant polynomial on a matrix Lie algebra. The form
\(f(F^\nabla)\) is closed, and its de Rham class does not depend on the
connection.

\begin{proposition}[Transgression]
For a smooth family \(\nabla^t\) with curvature \(F_t\),
\[
 \frac{d}{dt}f(F_t)
 =
 d\!\left(
 k\,f(\dot\nabla^t,F_t,\ldots,F_t)
 \right)
\]
when \(f\) is homogeneous of degree \(k\).
\end{proposition}

\begin{proof}
Write \(\nabla^t=d+\omega_t\) in a local frame and
\(\alpha_t=\dot\omega_t\). Since
\[
 F_t=d\omega_t+\omega_t\wedge\omega_t,
\]
differentiation gives
\[
 \dot F_t=d\alpha_t+\omega_t\wedge\alpha_t+\alpha_t\wedge\omega_t
 =d_{\nabla^t}\alpha_t.
\]
Polarize \(f\) to a symmetric \(k\)-linear invariant form. Then
\[
 \frac d{dt}f(F_t,\ldots,F_t)
 =k\,f(d_{\nabla^t}\alpha_t,F_t,\ldots,F_t).
\]
For endomorphism-valued forms, invariance of \(f\) implies the covariant
Leibniz identity
\[
 d\,f(B_1,\ldots,B_k)
 =
 \sum_j(-1)^{\deg B_1+\cdots+\deg B_{j-1}}
 f(B_1,\ldots,d_{\nabla^t}B_j,\ldots,B_k);
\]
the commutator terms cancel inside \(f\). Apply this with
\(B_1=\alpha_t\) and \(B_2=\cdots=B_k=F_t\). The Bianchi identity
\(d_{\nabla^t}F_t=0\) kills every term except the first, proving
\[
 \frac d{dt}f(F_t)
 =d\bigl(kf(\alpha_t,F_t,\ldots,F_t)\bigr).
\]
Integrating from \(t=0\) to \(1\) shows explicitly that two connections
produce cohomologous characteristic forms.
\end{proof}

\begin{discussion}
Chern--Weil theory is the bridge between analysis and topology used later.
The heat kernel produces a particular curvature polynomial. Transgression
shows that its integral on a closed manifold is independent of the
connection used to compute it.
\end{discussion}

For a complex bundle \(E\), the Chern character form is
\[
 \operatorname{ch}(E,\nabla)
 =
 \Tr\exp\!\left(\frac{iF^\nabla}{2\pi}\right).
\]
For a real oriented bundle with curvature matrix \(R\), define
\[
 \Ahat(E,\nabla)
 =
 \det\nolimits^{1/2}
 \left(
 \frac{R/(4\pi i)}{\sinh(R/(4\pi i))}
 \right),
\]
and
\[
 L(E,\nabla)
 =
 \det\nolimits^{1/2}
 \left(
 \frac{R/(2\pi i)}{\tanh(R/(2\pi i))}
 \right).
\]
These power-series expressions are interpreted through invariant
polynomials. Their cohomology classes are denoted
\(\operatorname{ch}(E)\), \(\Ahat(E)\), and \(L(E)\).

For a complex bundle with Chern roots \(x_j\), the Todd form is
\[
 \operatorname{Td}(E)=\prod_j\frac{x_j}{1-e^{-x_j}}.
\]
The splitting principle makes this a well-defined polynomial in the Chern
classes.

\section{Differential complexes}

A differential complex is a sequence
\[
 0\longrightarrow\Gamma(E^0)
 \xrightarrow{D_0}\Gamma(E^1)
 \xrightarrow{D_1}\cdots
 \xrightarrow{D_{r-1}}\Gamma(E^r)
 \longrightarrow0,
 \qquad D_{j+1}D_j=0.
\]
Its cohomology is
\[
 H^j(D)=\ker D_j/\operatorname{im}D_{j-1}.
\]
When these spaces are finite-dimensional, its Euler characteristic is
\[
 \chi(D)=\sum_j(-1)^j\dim H^j(D).
\]

The basic example is the de Rham complex
\[
 0\longrightarrow\Omega^0(M)
 \xrightarrow{d}\Omega^1(M)
 \xrightarrow{d}\cdots
 \xrightarrow{d}\Omega^n(M)
 \longrightarrow0.
\]
Its cohomology is the de Rham cohomology of \(M\), and its Euler
characteristic is the topological Euler characteristic \(\chi(M)\).


\chapter{Elliptic operators and index}

\section{Differential operators and principal symbols}

A differential operator \(P:\Gamma(E)\to\Gamma(F)\) of order at most \(m\)
has the local form
\[
 P=\sum_{|\alpha|\leq m}a_\alpha(x)\partial_x^\alpha.
\]
Its principal symbol is
\[
 \sigma_m(P)(x,\xi)
 =
 \sum_{|\alpha|=m}a_\alpha(x)(i\xi)^\alpha:
 E_x\longrightarrow F_x.
\]
Equivalently,
\[
 \sigma_m(P)(x,df)s
 =
 \lim_{\tau\to\infty}
 \tau^{-m}e^{-i\tau f}P(e^{i\tau f}s)(x).
\]
The latter formula proves coordinate invariance.

\begin{discussion}
The oscillatory test \(e^{i\tau f}s\) isolates the part of \(P\) in which
all \(m\) derivatives fall on the phase. Derivatives of the section, the
coefficients, or lower-order terms carry fewer powers of \(\tau\) and
disappear after multiplication by \(\tau^{-m}\). This is why the principal
symbol is intrinsic although the coefficients \(a_\alpha\) are not.
\end{discussion}

The symbol is multiplicative:
\[
 \sigma_{m+\ell}(PQ)
 =
 \sigma_m(P)\sigma_\ell(Q).
\]
If metrics and a density are fixed, the formal adjoint satisfies
\[
 \sigma_m(P^*)(x,\xi)=\sigma_m(P)(x,\xi)^*.
\]

\begin{definition}
The operator \(P\) is elliptic if \(\sigma_m(P)(x,\xi)\) is invertible for
every \(\xi\neq0\).
\end{definition}

For a first-order complex \((E^\bullet,D_\bullet)\), the symbol sequence at
\((x,\xi)\in T^*M\setminus0\) is
\[
 0\longrightarrow E_x^0
 \xrightarrow{\sigma(D_0)(x,\xi)}E_x^1
 \longrightarrow\cdots
 \longrightarrow E_x^r
 \longrightarrow0.
\]
The complex is elliptic when this sequence is exact for every nonzero
\(\xi\).

Choose bundle metrics. The associated Laplacians are
\[
 \Delta_j=D_j^*D_j+D_{j-1}D_{j-1}^*.
\]

\begin{proposition}
A complex is elliptic if and only if every \(\Delta_j\) is elliptic.
\end{proposition}

\begin{proof}
At a nonzero covector, put \(a_j=\sigma(D_j)\). Since
\(a_ja_{j-1}=0\),
\[
 \sigma(\Delta_j)=a_j^*a_j+a_{j-1}a_{j-1}^*.
\]
Its quadratic form is
\[
 \langle\sigma(\Delta_j)v,v\rangle
 =
 \lVert a_jv\rVert^2+\lVert a_{j-1}^*v\rVert^2.
\]
The kernel is therefore
\(\ker a_j\cap(\operatorname{im}a_{j-1})^\perp\), which vanishes exactly
when \(\ker a_j=\operatorname{im}a_{j-1}\).
\end{proof}

\section{Pseudodifferential parametrices}

A classical pseudodifferential operator of order \(m\) has, in local
coordinates, an oscillatory representation
\[
 Pu(x)
 =
 (2\pi)^{-n}\int e^{i(x-y)\cdot\xi}p(x,\xi)u(y)
 \dd y\,\dd\xi,
\]
modulo a smooth kernel, where
\[
 p(x,\xi)\sim p_m(x,\xi)+p_{m-1}(x,\xi)+\cdots
\]
and \(p_{m-j}\) is homogeneous of degree \(m-j\) for large \(\xi\).
The composition formula is
\[
 p\# q
 \sim
 \sum_\alpha
 \frac{1}{\alpha!}
 \partial_\xi^\alpha p\,D_x^\alpha q,
 \qquad D_x=\frac1i\partial_x.
\]

\begin{theorem}[Elliptic parametrix]
If \(P\in\Psi^m(M;E,F)\) is elliptic, there is
\(Q\in\Psi^{-m}(M;F,E)\) such that
\[
 QP=I-R_L,\qquad PQ=I-R_R,
 \qquad R_L,R_R\in\Psi^{-\infty}.
\]
\end{theorem}

\begin{proof}
Set \(q_{-m}=p_m^{-1}\) outside a neighborhood of the zero section.
Suppose \(q_{-m},\ldots,q_{-m-j+1}\) have been chosen. The component of
degree \(-j\) in \(q\# p-I\) is
\[
 q_{-m-j}p_m+
 \sum_{\substack{k+\ell+|\alpha|=j\\k<j}}
 \frac1{\alpha!}
 \partial_\xi^\alpha q_{-m-k}\,D_x^\alpha p_{m-\ell}.
\]
Therefore set
\[
 q_{-m-j}
 =
 -\left(
 \sum_{\substack{k+\ell+|\alpha|=j\\k<j}}
 \frac1{\alpha!}
 \partial_\xi^\alpha q_{-m-k}\,D_x^\alpha p_{m-\ell}
 \right)p_m^{-1}.
\]
This removes the next homogeneous error. A partition of unity patches the
construction, since discrepancies on overlaps have one lower order.
Borel's lemma produces a classical symbol \(q\sim\sum_jq_{-m-j}\); the
quantized operator \(Q\) satisfies \(QP=I-R_L\) with \(R_L\) smoothing.
The same construction on the other side gives a right parametrix.
Equivalently, ellipticity forces the symbol of \(PQ-I\) to vanish to every
order once \(QP-I\) does. A smoothing correction then gives one operator
with both displayed identities.
\end{proof}

\begin{tikzcd}[column sep=large,row sep=large]
 P
 \arrow[r,"\sigma_m"]
 &p_m
 \arrow[r,"\text{invert}"]
 &p_m^{-1}
 \arrow[d,"\text{recursive corrections}"]\\

\end{tikzcd}
\]
\caption{Ellipticity is converted into Fredholmness through symbol

Let \(H^s(M;E)\) denote the Sobolev space of order \(s\). An operator in
\(\Psi^m\) extends boundedly
\[
 P:H^s(M;E)\longrightarrow H^{s-m}(M;F).
\]
A smoothing operator maps every Sobolev space continuously into every
higher Sobolev space and is compact as an operator \(H^s\to H^s\).

\begin{theorem}[Elliptic regularity]
If \(P\) is elliptic and \(Pu\in H^{s-m}\), then \(u\in H^s\). More
precisely,
\[
 \lVert u\rVert_{H^s}
 \leq C\bigl(\lVert Pu\rVert_{H^{s-m}}+\lVert u\rVert_{H^{s-1}}\bigr).
\]
\end{theorem}

\begin{proof}
From \(u=QPu+R_Lu\), the first term lies in \(H^s\), while the second is
smooth. The estimate follows from boundedness of \(Q\), compactness of the
lower-order inclusion, and the standard contradiction argument.
\end{proof}

\begin{corollary}
On a compact manifold, an elliptic operator
\[
 P:H^s(M;E)\longrightarrow H^{s-m}(M;F)
\]
is Fredholm. Its kernel consists of smooth sections, and
\[
 \operatorname{coker}P\cong\ker P^*.
\]
\end{corollary}

\begin{proof}
The parametrix identities make \(P\) invertible modulo compact operators,
so Atkinson's theorem gives Fredholmness. Elliptic regularity makes the
kernel smooth. The cokernel is the orthogonal complement of the range,
which is \(\ker P^*\).
\end{proof}

\begin{question}
Where are compactness of \(M\) and ellipticity used in Fredholmness?
\end{question}

\begin{answer}
Ellipticity produces a parametrix whose remainders gain arbitrarily many
derivatives. Compactness of \(M\) makes the Sobolev inclusion
\(H^{s+N}\hookrightarrow H^s\) compact, so those remainders are compact.
On a noncompact manifold, translates of one bump function can remain
bounded with no convergent subsequence; smoothing alone then need not imply
compactness. The boundary calculus in the companion notes is designed to
control exactly that failure.
\end{answer}

The analytic index is
\[
 \ind P=\dim\ker P-\dim\ker P^*.
\]
It is independent of \(s\), invariant under continuous elliptic
deformations, and unchanged by lower-order perturbations.

\section{Hodge theory}

Let \((E^\bullet,D_\bullet)\) be an elliptic complex. Define
\[
 \mathcal H^j=\ker\Delta_j.
\]

\begin{theorem}[Hodge decomposition]
For every \(j\),
\[
 L^2(E^j)
 =
 \mathcal H^j
 \mathbin{\widehat\oplus}
 \overline{\operatorname{im}D_{j-1}}
 \mathbin{\widehat\oplus}
 \overline{\operatorname{im}D_j^*}.
\]
For an elliptic complex on a compact manifold, both ranges are closed,
\(\mathcal H^j\) is finite-dimensional and smooth, and
\[
 H^j(D)\cong\mathcal H^j.
\]
\end{theorem}

\begin{proof}
The three spaces are mutually orthogonal. The elliptic operator
\(\Delta_j\) is self-adjoint and Fredholm, hence
\[
 L^2(E^j)=\ker\Delta_j\oplus\operatorname{im}\Delta_j.
\]
If \(G_j\) is the inverse of \(\Delta_j\) on
\(\mathcal H^j{}^\perp\), extended by zero on \(\mathcal H^j\), then
\[
 \Delta_jG_j=G_j\Delta_j=I-H_j,
\]
where \(H_j\) is orthogonal projection onto \(\mathcal H^j\). The complex
relations imply
\[
 D_j\Delta_j=\Delta_{j+1}D_j,
\qquad
 D_{j-1}^*\Delta_j=\Delta_{j-1}D_{j-1}^*.
\]
Functional calculus therefore gives the corresponding commutation
relations with \(G\). Expanding \(\Delta_jG_j=I-H_j\) yields
\[
 u=H_ju+D_{j-1}D_{j-1}^*G_ju+D_j^*D_jG_ju.
\]
Thus every \(u\) is a sum of a harmonic, an exact, and a coexact term.
These three terms are orthogonal: for example,
\[
 \langle D_{j-1}v,D_j^*w\rangle
 =\langle D_jD_{j-1}v,w\rangle=0.
\]
The displayed bounded projections onto the three summands show that both
ranges are closed. Elliptic regularity makes \(\mathcal H^j\)
finite-dimensional and smooth.

If \(D_ju=0\), pair \(u\) with its coexact summand:
\[
 \langle u,D_j^*w\rangle=\langle D_ju,w\rangle=0.
\]
Orthogonality therefore forces that summand to vanish, so \(u\) is
cohomologous to \(H_ju\). A harmonic exact form is zero, proving
uniqueness.
\end{proof}

For the de Rham complex,
\[
 d^*=(-1)^{n(k+1)+1}*d*
 \quad\text{on }\Omega^k(M),
 \qquad
 \Delta=dd^*+d^*d.
\]
Thus every de Rham cohomology class has a unique harmonic representative.

Grade forms by parity and set
\[
 D_{\mathrm{dR}}=d+d^*:
 \Omega^{\mathrm{ev}}(M)\longrightarrow\Omega^{\mathrm{odd}}(M).
\]
Hodge theory gives
\[
 \ind D_{\mathrm{dR}}=\chi(M).
\]

\section{The symbol class and the index theorem}

If \(P:E\to F\) is elliptic, its principal symbol is an isomorphism over
\(T^*M\setminus0\). Hence the triple
\[
 (\pi^*E,\pi^*F,\sigma(P))
\]
defines a compactly supported class
\[
 [\sigma(P)]\in K_c^0(T^*M).
\]
Homotopic elliptic symbols determine the same class.

Let \(\mathcal E(M)\) denote stable homotopy classes of elliptic operators.
The two notions of index fit into the diagram
\begin{figure}[htbp]
\centering
\[
\begin{tikzcd}[column sep=huge,row sep=large]
 \mathcal E(M)
   \arrow[r,"{[\sigma]}"]
   \arrow[dr,"{\operatorname{ind}}"']
 & K_c^0(T^*M) \arrow[d,"{\mathrm{t-ind}}"]\\
 & \mathbb Z .
\end{tikzcd}
\]
\caption{The Atiyah--Singer theorem states that the analytic index factors
through the compactly supported symbol class.}
\end{figure}

\begin{theorem}[Atiyah--Singer index theorem]
There is a topological index homomorphism
\[
 \operatorname{t\text{-}ind}:K_c^0(T^*M)\longrightarrow\Z
\]
constructed from the Thom isomorphism and Bott periodicity, and every
elliptic operator satisfies
\[
 \ind P=\operatorname{t\text{-}ind}[\sigma(P)].
\]
After applying the compactly supported Chern character, the topological
index becomes the characteristic-number pairing determined by the Thom
class and \(\operatorname{Td}(T_\C M)\).
\end{theorem}

The theorem says that the analytic index depends only on the homotopy class
of the principal symbol. For Dirac operators, the Thom-class calculation
reduces the general characteristic-number formula to the explicit
\(\Ahat\)-formula proved by heat kernels below.

To make the construction of \(\operatorname{t\text{-}ind}\) concrete, embed
\(M\) in an even-dimensional Euclidean space \(\mathbb R^N\). Choose
metrics to identify tangent and cotangent bundles. The differential of the
embedding then gives \(T^*M\hookrightarrow T^*\mathbb R^N\). If \(\nu\) is
the normal bundle of \(M\), the normal bundle of this induced embedding is
\(\pi^*(\nu\oplus\nu)\); the rule \(J(v,w)=(-w,v)\) gives it a canonical
complex structure. Its Thom class defines a push-forward
\[
 K_c^0(T^*M)\longrightarrow K_c^0(T^*\mathbb R^N).
\]
Bott periodicity identifies the target with \(\mathbb Z\); the resulting
integer is the topological index. Homotopy invariance of the Thom class
and stabilization show that it is independent of the embedding.

\begin{discussion}
The general Atiyah--Singer theorem above uses two major topological inputs:
the Thom isomorphism and Bott periodicity. The heat-kernel argument in the
following chapters proves the local formula for Dirac operators. The
standard \(K\)-theoretic reduction of a general elliptic symbol to Dirac
symbols then recovers the general theorem. Keeping these two layers
separate prevents the Dirac calculation from being mistaken for a proof
that silently assumes every elliptic operator is already of Dirac type.
\end{discussion}

\section{The K-theoretic proof}

The preceding construction defines the topological index. We now explain
why it equals the analytic index. The proof is short only after three
substantial facts have been isolated: an elliptic symbol can be quantized,
the analytic index commutes with Thom push-forward, and the Bott generator
has index one. This is the standard \(K\)-theoretic proof architecture of
Atiyah and Singer \cite{ASI,ABS}.

First define the analytic index directly on compactly supported
\(K\)-theory. A class in \(K_c^0(T^*M)\) is represented, after
stabilization, by

\[
 \alpha=[\pi^*E,\pi^*F,a],
\]

where \(a:\pi^*E\to\pi^*F\) is invertible outside a compact set. Radial
deformation makes \(a\) homogeneous for large covectors, and symbol
quantization produces an elliptic pseudodifferential operator
\(P_a:E\to F\) with principal symbol \(a\). Set

\[
 \operatorname{a\text{-}ind}(\alpha)=\ind P_a.
\]

This integer is well defined. A homotopy of symbols quantizes to a
continuous family of Fredholm operators between fixed Sobolev spaces, so
its index is constant. Direct sum gives additivity, and adding the symbol
of an invertible bundle map adds an invertible operator and hence index
zero. These are precisely the relations used to define
\(K_c^0(T^*M)\).

\begin{lemma}[Compatibility with push-forward]
Let \(i:M\hookrightarrow N\) be an embedding of closed manifolds. For
every \(\alpha\in K_c^0(T^*M)\),
\[
 \operatorname{a\text{-}ind}(i_!\alpha)
 =\operatorname{a\text{-}ind}(\alpha).
\]
\end{lemma}

\begin{proof}
Choose a tubular neighborhood with normal bundle \(\nu\). The normal
bundle of \(T^*M\) in \(T^*N\) is \(\pi^*(\nu\oplus\nu)\), with complex
structure
\[
 J(v,w)=(-w,v).
\]
Consequently \(i_!\) is, in the tubular neighborhood, exterior product
with the Bott--Thom class, followed by extension to \(T^*N\).

Here is the analytic representative of that class. On a Euclidean fibre
\(V\) act on \(\Lambda^\bullet V_\mathbb C^*\) by
\[
 c(v)=\varepsilon(v^\flat)-\iota_v,
 \qquad
 \widehat c(v)=\varepsilon(v^\flat)+\iota_v.
\]
For an orthonormal basis \(e_1,\ldots,e_r\), the odd Bott operator is
\[
 B_V=\sum_{j=1}^r c(e_j)\frac{\partial}{\partial v_j}
      +\widehat c(v).
\]
Its full symbol in the position and covariable pair \((v,\xi)\) is
\(i\,c(\xi)+\widehat c(v)\), the Bott class on \(V\oplus V^*\). With the
conventions above,
\[
 B_V^2=-\Delta+|v|^2+
 \sum_{j=1}^r c(e_j)\widehat c(e_j).
\]
The harmonic-oscillator decomposition shows that its kernel is the
one-dimensional Gaussian vacuum in even Bott degree and that its odd
kernel is zero. In particular, \(B_V\) is Fredholm and
\[
 \operatorname{ind}B_V=1.
\]

Write \(P_a\) as the odd self-adjoint operator
\[
 \mathcal P_a=
 \begin{pmatrix}0&P_a^*\\ P_a&0\end{pmatrix}
 \quad\text{on }E\oplus F.
\]
Let \(\Gamma_{E\oplus F}\) act by \(+1\) on \(E\) and by \(-1\) on
\(F\). Over the tubular neighborhood, the graded product
\[
 \mathcal P_a\widehat\otimes 1
 +\Gamma_{E\oplus F}\widehat\otimes B_\nu
\]
has symbol \(\alpha\) multiplied by the Thom class. Its kernel is the
kernel of \(\mathcal P_a\) tensored with the Gaussian vacuum; equivalently,
the Fredholm product formula gives
\[
 \operatorname{ind}(P_a\mathbin{\#}B_\nu)
 =\operatorname{ind}(P_a)\operatorname{ind}(B_\nu)
 =\operatorname{ind}(P_a).
\]
A connection patches the fibrewise construction over \(M\). Changing the
connection or the tubular neighborhood changes only lower-order terms
through a Fredholm homotopy. Finally, extension from the tubular
neighborhood is excision: the symbol is invertible off that neighborhood,
so the complementary part contributes index zero. These steps give the
claimed equality.
\end{proof}

We can now finish the proof of the index theorem. Choose an embedding
\(i:M\hookrightarrow\mathbb R^N\). Bott periodicity gives
\[
 K_c^0(T^*\mathbb R^N)
 \cong K_c^0(\mathbb R^{2N})
 \cong\mathbb Z.
\]
Let \(\beta_N\) be the Bott generator, normalized by
\(\operatorname{a\text{-}ind}(\beta_N)=1\). If
\(i_!\alpha=k\beta_N\), then the push-forward lemma gives
\[
 \operatorname{a\text{-}ind}(\alpha)
 =\operatorname{a\text{-}ind}(i_!\alpha)
 =k.
\]
By definition the topological index of \(\alpha\) is the same integer
\(k\). Therefore
\[
 \operatorname{a\text{-}ind}=\operatorname{t\text{-}ind}
 \quad\text{on }K_c^0(T^*M),
\]
which proves the Atiyah--Singer theorem for every elliptic operator.
Applying the compactly supported Chern character and the Thom identity
turns this equality into the cohomological formula. For a Dirac symbol,
the characteristic classes simplify to the
\(\widehat A\operatorname{ch}\)-density computed by the heat kernel.

\begin{question}
Where is the difficult geometry hidden in this short proof?
\end{question}

\begin{answer}
It is concentrated in the push-forward lemma. Quantization converts
symbol homotopy into Fredholm homotopy; the Thom class becomes the
fibrewise Bott operator; and excision identifies a tubular neighborhood
with its normal model. Once those compatibilities and
\(\operatorname{ind}B_V=1\) are proved, the final reduction to
\(K_c^0(\mathbb R^{2N})\cong\mathbb Z\) is formal.
\end{answer}


\chapter{Heat kernels}

\section{Existence and basic properties}

Let \(L\) be a nonnegative, formally self-adjoint elliptic differential
operator of order two on a Hermitian bundle \(E\to M\). Its self-adjoint
extension has discrete spectrum
\[
 0\leq\lambda_0\leq\lambda_1\leq\cdots,\qquad
 \lambda_j\longrightarrow\infty,
\]
with a complete orthonormal basis of smooth eigensections
\(\{\phi_j\}\).

For \(t>0\), functional calculus defines
\[
 e^{-tL}u=\sum_je^{-t\lambda_j}\langle u,\phi_j\rangle\phi_j.
\]
Its Schwartz kernel is
\[
 K(t;x,y)=\sum_je^{-t\lambda_j}\phi_j(x)\otimes\phi_j(y)^*.
\]
The series converges in every smooth topology for \(t\geq t_0>0\).
Consequently \(e^{-tL}\) is smoothing and trace class, and
\[
 \Tr(e^{-tL})=\sum_je^{-t\lambda_j}.
\]

The heat operator satisfies
\[
 (\partial_t+L)K=0,\qquad
 \lim_{t\downarrow0}e^{-tL}u=u,
\]
and the semigroup identity
\[
 e^{-(t+s)L}=e^{-tL}e^{-sL}.
\]

\begin{proposition}[Duhamel formula]
For a differentiable family \(L_r\) of such operators,
\[
 \frac{d}{dr}e^{-tL_r}
 =
 -\int_0^t
 e^{-(t-s)L_r}\dot L_r e^{-sL_r}\,\dd s.
\]
\end{proposition}

\begin{proof}
Let \(U_r(t)\) denote the right side. Differentiation of
\((\partial_t+L_r)e^{-tL_r}=0\) with respect to \(r\) shows that
\(\partial_r e^{-tL_r}\) solves
\[
 (\partial_t+L_r)U=-\dot L_r e^{-tL_r},\qquad U(0)=0.
\]
Variation of constants gives the displayed integral. Uniqueness of the
inhomogeneous heat equation completes the proof.
\end{proof}

\section{The local parametrix}

An operator is of Laplace type if its principal symbol is
\[
 \sigma_2(L)(x,\xi)=|\xi|^2\operatorname{Id}_{E_x}.
\]
Every Laplace-type operator has a unique expression
\[
 L=\nabla^*\nabla+\mathcal R
\]
for a connection \(\nabla\) and an endomorphism \(\mathcal R\).

Near the diagonal, let \(r(x,y)\) be the Riemannian distance. Seek a
parametrix of the form
\[
 H_N(t;x,y)
 =
 (4\pi t)^{-n/2}e^{-r(x,y)^2/(4t)}
 \sum_{j=0}^Nt^j u_j(x,y).
\]

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[>=Latex,thick]
 \draw[fill=gray!6] (0,0) rectangle (5,5);
 \draw[blue!15,line width=18pt] (.35,.35)--(4.65,4.65);
 \draw[blue!70!black,very thick] (.25,.25)--(4.75,4.75)
   node[above left] {\(\Delta\subset M\times M\)};
 \draw[<->,red!70!black] (2.1,2.65)--(2.65,2.1)
   node[midway,below left] {\(O(\sqrt t)\)};
 \node at (2.5,-.55)
   {\(e^{-d(x,y)^2/(4t)}\) localizes near the diagonal};
\end{tikzpicture}
\caption{As \(t\downarrow0\), the heat kernel concentrates in a
\(\sqrt t\)-neighborhood of the diagonal; the transport equations resolve
its coefficients along radial geodesics.}
\end{figure}

Substitution into \((\partial_t+L_x)H_N=0\) first determines \(u_0\) by
parallel transport and the Riemannian density, then determines each
\(u_j\) from a transport equation along the radial geodesic.

More explicitly, let \(\gamma(\rho)=\exp_y(\rho v)\), \(0\leq\rho\leq1\),
with \(x=\exp_yv\). If \(J(x,y)\) is the Jacobian of the exponential map
and \(\tau(x,y):E_y\to E_x\) is parallel transport, then
\[
 u_0(x,y)=J(x,y)^{-1/2}\tau(x,y),
\qquad
 u_0(y,y)=I.
\]
After conjugation by \(u_0\), the \(j\)-th transport equation is an
ordinary radial equation. Its solution is
\[
 u_j(x,y)
 =
 -u_0(x,y)\int_0^1\rho^{j-1}
 u_0(\gamma(\rho),y)^{-1}
 (L_xu_{j-1})(\gamma(\rho),y)\,d\rho,
 \qquad j\geq1.
\]
The integral is smooth at \(x=y\), and induction gives smooth \(u_j\) on a
normal neighborhood of the diagonal.

Multiply by a cutoff supported in a normal neighborhood of the diagonal.
The error
\[
 R_N=(\partial_t+L)H_N
\]
is \(O(t^{N-n/2})\), with corresponding estimates after differentiation.
For \(N\) large, the Volterra series
\[
 K
 =
 H_N-H_N*R_N+H_N*R_N*R_N-\cdots
\]
converges, where
\[
 (A*B)(t)=\int_0^tA(t-s)B(s)\,\dd s.
\]
It solves the heat equation with the correct initial value and therefore
equals the heat kernel.

For completeness, the Volterra series converges because its \(k\)-fold
term is integrated over an ordered simplex. If
\(\|R_N(t)\|\leq Ct^\alpha\), \(\alpha>-1\), then repeated convolution
gives
\[
 \|R_N^{*k}(t)\|
 \leq
 \frac{C^k\Gamma(\alpha+1)^k}
      {\Gamma(k(\alpha+1))}
 t^{k(\alpha+1)-1}.
\]
The gamma function supplies factorial decay. The same estimate for
derivatives gives convergence in every smooth kernel seminorm.

\begin{theorem}[Heat-kernel expansion]
On the diagonal,
\[
 K(t;x,x)
 \sim
 (4\pi t)^{-n/2}
 \sum_{j=0}^\infty t^j a_j(x)
 \qquad(t\downarrow0),
\]
in the smooth topology. Each \(a_j(x)\) is a universal polynomial in the
curvatures of \(M\) and \(E\), the endomorphism \(\mathcal R\), and their
covariant derivatives. In particular,
\[
 a_0=\operatorname{Id}_E,\qquad
 a_1=\frac16\operatorname{scal}\,\operatorname{Id}_E-\mathcal R
\]
for the convention \(L=\nabla^*\nabla+\mathcal R\).
\end{theorem}

Integrating the fibrewise trace gives
\[
 \Tr(e^{-tL})
 \sim
 (4\pi t)^{-n/2}
 \sum_{j=0}^\infty t^j
 \int_M\operatorname{tr}_Ea_j(x)\,\dd\operatorname{vol}(x).
\]

\section{McKean--Singer}

Let
\[
 D=
 \begin{pmatrix}
 0&D^-\\
 D^+&0
 \end{pmatrix}
\]
be an odd, formally self-adjoint elliptic operator on a graded bundle
\(E^+\oplus E^-\). Then
\[
 D^2=
 \begin{pmatrix}
 D^-D^+&0\\
 0&D^+D^-
 \end{pmatrix}.
\]
For a trace-class endomorphism \(T\), define
\[
 \Str(T)=\Tr(T|_{E^+})-\Tr(T|_{E^-}).
\]

\begin{theorem}[McKean--Singer]
For every \(t>0\),
\[
 \Str(e^{-tD^2})=\ind D^+.
\]
\end{theorem}

\begin{proof}
Every positive eigenvalue \(\lambda\) of \(D^-D^+\) is paired with the
same eigenvalue of \(D^+D^-\): the map \(D^+\) is an isomorphism between
the corresponding eigenspaces, with inverse \(\lambda^{-1}D^-\). Their
heat traces cancel in the supertrace. The zero eigenspaces contribute
\(\dim\ker D^+-\dim\ker D^-\), which is the index.
\end{proof}

\begin{question}
Why do the zero modes survive while every positive mode cancels?
\end{question}

\begin{answer}
On a positive eigenspace, \(D^+\) has inverse
\(\lambda^{-1}D^-\), so the even and odd multiplicities are equal and
carry the same heat weight \(e^{-t\lambda}\). At eigenvalue zero this
inverse does not exist. The unpaired spaces are precisely
\(\ker D^+\) and \(\ker D^-\), whose signed dimension is the index.
\end{answer}

Equivalently, cyclicity of the supertrace gives
\[
 \frac{d}{dt}\Str(e^{-tD^2})
 =
 -\frac12\Str([D,De^{-tD^2}]_{\mathrm s})=0.
\]
The small-time limit is therefore the index.

When the pointwise limit exists and
\[
 \operatorname{str}K(t;x,x)
 \sim
 (4\pi t)^{-n/2}\sum_{j\geq0}t^j\operatorname{str}a_j(x),
\]
then
\[
 \ind D^+
 =
 \int_M
 \lim_{t\downarrow0}\operatorname{str}K(t;x,x)
 \,\dd\operatorname{vol}(x).
\]
For a general elliptic operator, this formula still computes the index
after replacing it by the odd self-adjoint operator
\[
 \mathcal D=
 \begin{pmatrix}0&P^*\\P&0\end{pmatrix}.
\]
The remaining task is to identify the constant term of the local
supertrace. For Dirac operators, Clifford symmetry forces the singular
terms to cancel and Getzler rescaling computes the survivor.


\chapter{Clifford modules and Dirac operators}

\section{Clifford algebras}

For a Euclidean vector space \(V\), the complex Clifford algebra is
\[
 \Cl(V)=T(V_\C)\big/
 \langle vw+wv+2\langle v,w\rangle\rangle.
\]
Thus
\[
 c(v)c(w)+c(w)c(v)=-2\langle v,w\rangle.
\]
The algebra is \(\Z_2\)-graded by tensor parity.

If \(\dim V=2m\), the complex spin representation \(S\) has dimension
\(2^m\) and a grading
\[
 S=S^+\oplus S^-.
\]
Clifford multiplication by a vector is odd. The grading operator is the
complex volume element, normalized to square to one and to agree with the
chosen orientation.

\begin{definition}
A graded Clifford module over an oriented even-dimensional Riemannian
manifold is a graded Hermitian bundle
\[
 E=E^+\oplus E^-
\]
with a fibrewise action
\[
 c:\Cl(T^*M)\longrightarrow\operatorname{End}(E)
\]
such that \(c(\xi)\) is odd and skew-adjoint for real \(\xi\).
\end{definition}

Locally, every complex Clifford module in even dimension is of the form
\[
 E\cong S\otimes W,
\]
where the Clifford algebra acts on \(S\) and trivially on the twisting
factor \(W\). The bundle \(W\) need not exist globally unless a spinor
bundle does, but its Chern character is globally represented by the
relative curvature introduced below.

\section{Clifford connections}

\begin{definition}
A connection \(\nabla^E\) on a Clifford module is a Clifford connection if
\[
 [\nabla_X^E,c(\alpha)]
 =
 c(\nabla_X^{T^*M}\alpha)
\]
for every vector field \(X\) and one-form \(\alpha\).
\end{definition}

The associated Dirac operator is
\[
 D=c\circ\nabla^E
 =\sum_{j=1}^n c(e^j)\nabla_{e_j}^E
\]
in a local orthonormal frame. It is odd and formally self-adjoint. Its
principal symbol is
\[
 \sigma_1(D)(x,\xi)=i\,c(\xi).
\]
Since
\[
 (i\,c(\xi))^2=|\xi|^2\operatorname{Id},
\]
the operator is elliptic.

The curvature of a Clifford connection splits into its Clifford and
twisting parts. Define
\[
 F^{E/S}(X,Y)
 =
 F^E(X,Y)
 -\frac14\sum_{j,k}
 \langle R^{TM}(X,Y)e_j,e_k\rangle c(e^j)c(e^k).
\]
This endomorphism commutes with Clifford multiplication. Locally, for
\(E=S\otimes W\), it is exactly \(F^W\).

For an endomorphism-valued two-form \(F\), write
\[
 c(F)=\frac12\sum_{j,k}c(e^j)c(e^k)F(e_j,e_k).
\]

\begin{theorem}[Weitzenböck--Lichnerowicz formula]
A Dirac operator satisfies
\[
 D^2
 =
 (\nabla^E)^*\nabla^E
 +\frac14\operatorname{scal}
 +c(F^{E/S}).
\]
\end{theorem}

\begin{proof}
At a chosen point, take a synchronous orthonormal frame, so
\(\nabla e_j=0\) there. The symmetric part of
\(\sum_{j,k}c(e^j)c(e^k)\nabla_{e_j}\nabla_{e_k}\) is the connection
Laplacian. The antisymmetric part is
\[
 \frac12\sum_{j,k}c(e^j)c(e^k)F^E(e_j,e_k).
\]
Insert the decomposition of \(F^E\). The first Bianchi identity and the
Clifford relations contract the Riemannian curvature term to
\(\operatorname{scal}/4\); the remainder is \(c(F^{E/S})\).

For the contraction, write the spin-curvature contribution as
\[
 \frac18\sum_{i,j,k,\ell}
 R_{ij\,k\ell}\,c(e^i)c(e^j)c(e^k)c(e^\ell).
\]
Terms with four distinct indices cancel by the first Bianchi identity.
Terms with exactly three distinct indices cancel by the skew symmetries of
\(R\). The remaining pairings reduce, using \(c(e^i)^2=-1\), to
\[
 \frac14\sum_{i,j}R_{ij\,ji}
 =\frac14\operatorname{scal}.
\]
\end{proof}

\section{Basic examples}

On differential forms, define
\[
 c(\xi)=\varepsilon(\xi)-\iota_{\xi^\sharp}.
\]
The exterior and contraction operators satisfy
\[
 \varepsilon(\xi)\iota_{\eta^\sharp}
 +\iota_{\eta^\sharp}\varepsilon(\xi)
 =
 \langle\xi,\eta\rangle,
\]
so \(c\) obeys the Clifford relation. The Levi--Civita connection on forms
is a Clifford connection, and its Dirac operator is
\[
 D_{\mathrm{dR}}=d+d^*.
\]
Its grading by even and odd degree gives
\[
 \ind D_{\mathrm{dR}}^+=\chi(M).
\]

If \(M\) is spin, its spinor bundle \(S=S^+\oplus S^-\) carries the spin
connection and spin Dirac operator
\[
 D_S:\Gamma(S^+)\longrightarrow\Gamma(S^-).
\]
For a Hermitian bundle \(W\) with unitary connection, the twisted operator
on \(S\otimes W\) satisfies
\[
 D_W^2
 =
 \nabla^*\nabla+\frac14\operatorname{scal}+c(F^W).
\]

\begin{corollary}[Lichnerowicz vanishing]
If \(M\) is closed, spin, and has strictly positive scalar curvature, then
the untwisted spin Dirac operator has zero kernel and zero index.
\end{corollary}

\begin{proof}
For \(D_S\psi=0\), integration of the Lichnerowicz formula gives
\[
 0
 =
 \lVert\nabla\psi\rVert^2
 +\frac14\int_M\operatorname{scal}|\psi|^2.
\]
Strict positivity forces \(\psi=0\).
\end{proof}

\section{Clifford supertrace}

The supertrace on a graded endomorphism algebra is
\[
 \operatorname{str}(T)=\operatorname{tr}(T|_{E^+})
 -\operatorname{tr}(T|_{E^-}).
\]
It vanishes on supercommutators.

\begin{lemma}[Clifford cancellation]
Let \(\dim V=n\) be even and let \(S\) be the spin representation. The
supertrace of a Clifford monomial of degree less than \(n\) is zero. On the
top Clifford degree, the supertrace is the oriented volume functional,
with the normalization fixed by the chirality operator.
\end{lemma}

\begin{proof}
Odd Clifford monomials are odd endomorphisms and therefore have zero
supertrace. Let \(T=c(e^{i_1})\cdots c(e^{i_{2r}})\) have even degree
\(2r<n\), and choose \(e^j\) not among its factors. Then \(c(e^j)\)
commutes with \(T\) but anticommutes with the chirality operator
\(\Gamma\). Since \(\operatorname{str}(T)=\operatorname{tr}(\Gamma T)\),
ordinary trace invariance under conjugation by \(c(e^j)\) gives
\[
 \operatorname{str}(T)
 =\operatorname{tr}\!\left(c(e^j)^{-1}\Gamma T c(e^j)\right)
 =-\operatorname{str}(T),
\]
so it vanishes. In degree \(n\), the monomial is a scalar multiple of the
complex volume element and hence of \(\Gamma\); its supertrace is nonzero.
The chosen orientation and the relation \(c(e^j)^2=-1\) fix that scalar.
\end{proof}

\begin{discussion}
The cancellation is pointwise. It removes the negative powers of \(t\) in
the local heat supertrace before integration. McKean--Singer says the
integral is independent of \(t\); Clifford cancellation explains why its
small-time limit is already a smooth top-degree differential form.
\end{discussion}

This cancellation is the algebraic reason the divergent coefficients in
the Dirac heat supertrace vanish.


\chapter{The local index theorem}

\section{Relative curvature}

For a graded Clifford module \(E\), define the relative Chern character
form by
\[
 \operatorname{ch}(E/S,\nabla^E)
 =
 \operatorname{tr}_{E/S}
 \exp\!\left(\frac{iF^{E/S}}{2\pi}\right).
\]
The relative trace is characterized locally by
\[
 E\cong S\otimes W
 \quad\Longrightarrow\quad
 \operatorname{ch}(E/S)=\operatorname{ch}(W).
\]
This prescription is independent of the local spinor factorization and
therefore defines a global differential form.

The local index density is
\[
 \left[
 \Ahat(TM,\nabla^{TM})
 \operatorname{ch}(E/S,\nabla^E)
 \right]_{[n]}.
\]

\section{Getzler rescaling}

Fix \(p\in M\). Choose geodesic normal coordinates centered at \(p\) and
trivialize \(E\) by radial parallel transport. In these coordinates,
the connection and metric have expansions whose first nonconstant terms
are determined by curvature.

In radial gauge,
\[
 g_{ij}(x)=\delta_{ij}-\frac13R_{ikj\ell}(p)x^kx^\ell+O(|x|^3),
\qquad
 \nabla_i^E
 =\partial_i-\frac12F^E_{ij}(p)x^j+O(|x|^2).
\]
The second formula follows by differentiating the radial-gauge condition
\(x^i\nabla_i^E=x^i\partial_i\): the linear part of the connection one-form
is forced by its curvature. Splitting \(F^E\) into spin and relative
curvature separates the part that acts through Clifford multiplication
from the part commuting with it.

Assign Getzler degrees
\[
 \deg\partial_j=1,\qquad
 \deg c(dx^j)=1,\qquad
 \deg x^j=-1,\qquad
 \deg\partial_t=2.
\]
Under the parabolic scaling
\[
 x\mapsto\sqrt t\,x,
\]
only the highest Getzler-order part of \(\partial_t+D^2\) contributes to
the constant term of the Clifford supertrace. The Clifford filtration is
identified at its associated graded level with the exterior algebra:
\[
 c(dx^{i_1})\cdots c(dx^{i_k})
 \longmapsto
 dx^{i_1}\wedge\cdots\wedge dx^{i_k}.
\]

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[>=Latex,thick]
 \fill[blue!5] (0,0) circle (2.25);
 \draw[blue!35] (0,0) circle (2.25);
 \draw[blue!55] (0,0) circle (1.45);
 \draw[blue!75!black] (0,0) circle (.65);
 \fill (0,0) circle (2pt) node[below right] {\(p\)};
 \draw[->,red!70!black] (.2,.2)--(3.7,1.8)
   node[right,align=left] {\(X=x/\sqrt t\) fixed\\as \(t\downarrow0\)};
 \node at (0,-2.8)
   {geometry freezes to its curvature model at \(p\)};
\end{tikzpicture}
\caption{Getzler rescaling magnifies a shrinking neighborhood of \(p\).
The metric becomes Euclidean while the first nontrivial connection terms
retain the curvature.}
\end{figure}

The rescaled model of \(D^2\) is a harmonic oscillator with coefficients
given by \(R^{TM}_p\), together with the constant twisting curvature
\(F^{E/S}_p\). All terms of lower Getzler order disappear in the limit.

To state the model, put
\[
 \mathcal R_{ij}
 =
 \frac12\sum_{k,\ell}R_{ij\,k\ell}(p)\,
 dx^k\wedge dx^\ell.
\]
After passing from the Clifford filtration to its exterior associated
graded, the model operator is
\[
 \mathcal H_p
 =
 -\sum_i\left(
 \partial_i-\frac14\sum_j\mathcal R_{ij}x^j
 \right)^2
F^{E/S}_p.
\]
The scalar-curvature term in the Lichnerowicz formula has lower Getzler
degree and disappears; its effect is already encoded in the oscillator
through \(R^{TM}_p\). If \(\delta_\varepsilon u(x)=u(\varepsilon x)\) and
Clifford degree \(k\) is multiplied by \(\varepsilon^{-k}\), then
\[
 \varepsilon^2\delta_\varepsilon^{-1}D^2\delta_\varepsilon
 =\mathcal H_p+O(\varepsilon)
\]
coefficientwise on compact sets. Duhamel's formula and Gaussian estimates
then show convergence of the rescaled heat kernels, with all derivatives.

\begin{lemma}[Model heat kernel]
The heat kernel of the rescaled harmonic oscillator is given by Mehler's
formula. After Clifford supertrace and the Chern--Weil normalization, its
diagonal value at the origin is
\[
 \left[
 \Ahat(TM,\nabla^{TM})
 \operatorname{ch}(E/S,\nabla^E)
 \right]_{[n]}(p).
\]
\end{lemma}

\begin{proof}
Put the skew-symmetric curvature matrix into \(2\times2\) blocks. The model
operator splits into commuting two-dimensional oscillators and a constant
endomorphism. For the oscillator part, Mehler's formula at the origin is
\[
 K_{\mathrm{osc}}(t;0,0)
 =
 (4\pi t)^{-n/2}
 \det\nolimits^{1/2}
 \left(
 \frac{t\mathcal R/2}{\sinh(t\mathcal R/2)}
 \right).
\]
This identity is an equality of formal exterior-power series; it can be
verified block by block from the one-dimensional Gaussian oscillator.
Because \(F^{E/S}_p\) commutes with the Clifford action, it commutes with
the oscillator and contributes the factor \(e^{-tF^{E/S}_p}\).

Finally apply the top Clifford-degree supertrace. In each oriented
two-plane block, the Clifford normalization converts the oscillator factor
to
\[
 \frac{R/(4\pi i)}{\sinh(R/(4\pi i))},
\]
while the relative trace converts the constant-curvature exponential to
\(\operatorname{tr}_{E/S}\exp(iF^{E/S}/2\pi)\). Multiplying the blocks and
taking degree \(n\) gives
\[
 \left[
 \Ahat(TM,\nabla^{TM})
 \operatorname{ch}(E/S,\nabla^E)
 \right]_{[n]}(p).
\]
\end{proof}

The determinant factor is \(\Ahat(TM)\), and the relative trace of the
constant curvature exponential is \(\operatorname{ch}(E/S)\).

\section{The pointwise theorem}

\begin{theorem}[Local index theorem for Dirac operators]
Let \(D\) be the Dirac operator of a graded Clifford module \(E\) over a
closed even-dimensional Riemannian manifold. If \(K(t;x,y)\) is the heat
kernel of \(D^2\), then
\[
 \lim_{t\downarrow0}
 \operatorname{str}_E K(t;x,x)\,\dd\operatorname{vol}(x)
 =
 \left[
 \Ahat(TM,\nabla^{TM})
 \operatorname{ch}(E/S,\nabla^E)
 \right]_{[n]}(x).
\]
\end{theorem}

\begin{proof}
The heat parametrix gives a complete local expansion. Clifford
cancellation shows that coefficients of Getzler degree below \(n\) have
zero supertrace. After parabolic rescaling, the only surviving coefficient
is the heat kernel of the curvature model. The model lemma and the
Clifford top-degree supertrace identify it with the displayed Chern--Weil
form.

For the remainder, choose enough terms in the heat parametrix that its
error is \(O(t^N)\) in every required \(C^k\)-seminorm. Under
\(x=\sqrt t\,X\), a term of Getzler order \(n-j\) gains \(t^{j/2}\).
Terms killed by supertrace vanish identically, and every surviving term
other than the model therefore tends to zero. The Volterra correction has
the same estimate by the ordered-simplex bound from the heat-kernel
construction. This proves smooth convergence of the local density, not
merely convergence after integration.
\end{proof}

\begin{question}
Why does the supertrace retain only the top exterior degree?
\end{question}

\begin{answer}
Getzler rescaling turns the Clifford filtration into the exterior algebra.
The fibrewise supertrace vanishes on every Clifford degree below \(n\), as
proved in the Clifford-cancellation lemma. Degree above \(n\) is zero in
\(\Lambda^\bullet T_p^*M\). Thus only degree \(n\) survives, and that degree
is naturally an \(n\)-form that can be integrated over \(M\).
\end{answer}

Integrating and applying McKean--Singer gives the global formula.

\begin{theorem}[Dirac index theorem]
Under the same hypotheses,
\[
 \boxed{
 \ind D^+
 =
 \int_M
 \left[
 \Ahat(TM)
 \operatorname{ch}(E/S)
 \right]_{[n]}.
 }
\]
\end{theorem}

For a spin manifold and \(E=S\otimes W\),
\[
 \ind D_W^+
 =
 \int_M
 \left[
 \Ahat(TM)\operatorname{ch}(W)
 \right]_{[n]}.
\]
For \(W\) trivial, this is the \(\Ahat\)-genus.

\section{Metric and connection independence}

The left side of the index formula is constant under continuous
deformations through elliptic operators. On the right, the transgression
formula shows that a change of connection alters each characteristic form
by an exact form. Since \(M\) is closed, its integral is unchanged.

This agreement separates the two roles in the proof: heat-kernel analysis
identifies a local curvature density, while Chern--Weil theory proves that
its integral is topological.


\chapter{Gauss--Bonnet and the signature theorem}

\section{The Euler operator}

On an oriented Riemannian manifold, the de Rham operator
\[
 D_{\mathrm{dR}}=d+d^*
\]
is the Dirac operator on the Clifford module
\(\Lambda^\bullet T^*M_\C\), graded by parity.

\begin{theorem}[Chern--Gauss--Bonnet]
If \(M\) is closed, oriented, and even-dimensional, then
\[
 \chi(M)
 =
 \int_M e(TM)
 =
 \int_M\operatorname{Pf}\!\left(\frac{R^{TM}}{2\pi}\right).
\]
\end{theorem}

\begin{proof}
Hodge theory identifies
\[
 \ind D_{\mathrm{dR}}^+
 =
 \sum_j(-1)^j\dim H^j(M;\C)
 =
 \chi(M).
\]
Apply the local index theorem to
\(\Lambda^\bullet T^*M_\C\). The splitting principle reduces its
characteristic form to a product over oriented two-plane summands. In each
summand, the Clifford-module factor cancels the denominator in the
\(\Ahat\)-form and leaves the curvature root. Their product is the Pfaffian
\(\operatorname{Pf}(R^{TM}/2\pi)\).

In detail, if \(x_1,\ldots,x_m\) are the formal oriented curvature roots,
then
\[
 \Ahat(TM)=\prod_{j=1}^m
 \frac{x_j/2}{\sinh(x_j/2)}.
\]
The graded relative Chern character of the de Rham Clifford module is
\[
 \operatorname{ch}(\Lambda^\bullet/S)
 =\prod_{j=1}^m
 \left(e^{x_j/2}-e^{-x_j/2}\right)
 =\prod_{j=1}^m2\sinh(x_j/2).
\]
Their product is \(\prod_jx_j\), which is the Euler form.
\end{proof}

If \(\dim M\) is odd, Poincaré duality gives \(\chi(M)=0\), and there is
no top-degree Euler form.

For a closed oriented surface,
\[
 e(TM)=\frac{K}{2\pi}\,\dd A,
\]
where \(K\) is Gaussian curvature. Hence
\[
 \int_MK\,\dd A=2\pi\chi(M),
\]
the classical Gauss--Bonnet formula.

\section{The signature operator}

Let \(\dim M=4k\). On complex-valued \(p\)-forms define the chirality
operator
\[
 \tau=i^{p(p-1)+2k}*.
\]
Then
\[
 \tau^2=1,\qquad
 \tau D_{\mathrm{dR}}=-D_{\mathrm{dR}}\tau.
\]
Let \(\Omega^\pm\) be the \(\pm1\)-eigenspaces of \(\tau\). The signature
operator is
\[
 D_{\mathrm{sign}}^+:
 \Omega^+(M)\longrightarrow\Omega^-(M),
 \qquad
 D_{\mathrm{sign}}^+=D_{\mathrm{dR}}|_{\Omega^+}.
\]

\begin{proposition}
\[
 \ind D_{\mathrm{sign}}^+=\operatorname{sign}(M),
\]
the signature of the cup-product form on \(H^{2k}(M;\R)\).
\end{proposition}

\begin{proof}
The kernel of \(D_{\mathrm{dR}}\) is the space of harmonic forms. Harmonic
forms outside middle degree occur in Hodge-dual pairs and contribute
equally to the two chirality eigenspaces. In middle degree, \(\tau=*\).
The positive and negative eigenspaces of \(*\) have dimensions equal to
the positive and negative indices of
\[
 ([\alpha],[\beta])\longmapsto\int_M\alpha\wedge\beta.
\]
Their difference is the signature.
\end{proof}

\begin{theorem}[Hirzebruch signature theorem]
For a closed oriented \(4k\)-manifold,
\[
 \boxed{
 \operatorname{sign}(M)
 =
 \int_M L(TM).
 }
\]
\end{theorem}

\begin{proof}
Apply the local index theorem to the signature Clifford module. Under the
splitting principle, its relative Chern character multiplies the
\(\Ahat\)-factor into
\[
 \prod_j\frac{x_j}{\tanh x_j},
\]
which is the Hirzebruch \(L\)-form. Concretely, on each oriented
two-plane block the graded trace of the chirality action gives the ratio
\[
 \frac{x/\tanh x}{(x/2)/\sinh(x/2)}.
\]
Multiplication over the blocks gives the displayed product. The analytic
index is the signature by the preceding proposition.
\end{proof}

In dimension four,
\[
 L(TM)_{[4]}=\frac13p_1(TM),
\qquad
 \operatorname{sign}(M)=\frac13\int_Mp_1(TM).
\]


\chapter{Complex geometry and Riemann--Roch}

\section{Almost complex and complex manifolds}

An almost complex structure on a \(2m\)-manifold is an endomorphism
\(J:TM\to TM\) with \(J^2=-1\). Complexifying the tangent bundle gives
\[
 T_\C M=T^{1,0}M\oplus T^{0,1}M,
\]
the \(+i\) and \(-i\) eigenspaces of \(J\). The projections are
\[
 \Pi^{1,0}=\frac12(I-iJ),\qquad
 \Pi^{0,1}=\frac12(I+iJ).
\]
For a real vector \(v\),
\[
 v^{1,0}=\frac12(v-iJv),\qquad
 v^{0,1}=\frac12(v+iJv).
\]
Dualizing and taking exterior powers gives
\[
 \Lambda^kT_\C^*M
 =
 \bigoplus_{p+q=k}\Lambda^{p,q}T^*M.
\]

\begin{figure}[htbp]
\centering
\[
\begin{tikzcd}[column sep=large,row sep=large]
 &T^{1,0}M\\
 TM
 \arrow[r,"\mathbb C\otimes_{\mathbb R}(-)"]
 &T_\mathbb C M
 \arrow[u,"\Pi^{1,0}"]
 \arrow[d,"\Pi^{0,1}"']\\
 &T^{0,1}M
\end{tikzcd}
\qquad
\Lambda^kT_\mathbb C^*M
=\bigoplus_{p+q=k}\Lambda^{p,q}T^*M.
\]
\caption{An almost complex structure splits the complexified tangent
bundle and induces the \((p,q)\)-decomposition of forms.}
\end{figure}

\begin{proposition}
Every almost complex manifold is canonically oriented.
\end{proposition}

\begin{proof}
Choose real vectors \(v_1,\ldots,v_m\) such that
\[
 v_1,Jv_1,\ldots,v_m,Jv_m
\]
is a basis and declare it positive. A second complex frame is related to
the first by a matrix \(A\in\operatorname{GL}(m,\mathbb C)\). Viewed as a
real matrix,
\[
 \det_{\mathbb R}A=|\det_{\mathbb C}A|^2>0.
\]
Thus every complex change of frame preserves the declared orientation,
which patches smoothly over \(M\).
\end{proof}

The Nijenhuis tensor is
\[
 N_J(X,Y)
 =
 [JX,JY]-J[JX,Y]-J[X,JY]-[X,Y].
\]

\begin{theorem}[Newlander--Nirenberg]
An almost complex structure is induced by holomorphic coordinate charts if
and only if \(N_J=0\).
\end{theorem}

\begin{discussion}
Newlander--Nirenberg is a nonlinear local existence theorem and is used
here as a standard input. The necessity is elementary: coordinate vector
fields of type \((1,0)\) are closed under brackets, which forces
\(N_J=0\). The converse solves a first-order nonlinear system for
holomorphic coordinates and is not part of the index-theoretic proof.
\end{discussion}

Such a structure is called integrable, and \(M\) is then a complex
manifold. On a complex manifold,
\[
 d=\partial+\bar\partial,
\]
with
\[
 \partial:\Omega^{p,q}\to\Omega^{p+1,q},
 \qquad
 \bar\partial:\Omega^{p,q}\to\Omega^{p,q+1},
\]
and \(d^2=0\) implies
\[
 \partial^2=0,\qquad
 \bar\partial^2=0,\qquad
 \partial\bar\partial+\bar\partial\partial=0.
\]

On a nonintegrable almost complex manifold, \(d\) also has components of
bidegrees \((2,-1)\) and \((-1,2)\). In particular, the projected
\(\bar\partial\)-operator does not generally square to zero. This
distinction is essential: the canonical spin\(^{c}\) Dirac operator still
exists, but there need not be a Dolbeault complex.

\begin{question}
What exactly does integrability change?
\end{question}

\begin{answer}
An almost complex structure always splits complexified forms by type and
always determines a canonical spin\(^{c}\) symbol. Integrability is the
extra condition that the \((0,1)\)-part of \(d\) squares to zero. Only then
does \(\bar\partial\) define cohomology and only then can the spin\(^{c}\)
index be interpreted as a holomorphic Euler characteristic.
\end{answer}

\begin{example}[Complex projective space]
The charts
\[
 U_j=\{[z_0:\cdots:z_m]:z_j\ne0\},\qquad
 [z]\longmapsto
 \left(\frac{z_0}{z_j},\ldots,
 \widehat{\frac{z_j}{z_j}},\ldots,\frac{z_m}{z_j}\right)
\]
have holomorphic transition functions and define the complex manifold
\(\mathbb{CP}^m\). Its induced almost complex structure is integrable by
construction, and the Fubini--Study form makes it Kähler.
\end{example}

\section{Almost Hermitian and almost Kähler geometry}

An almost Hermitian structure is a pair \((J,g)\) satisfying
\[
 g(JX,JY)=g(X,Y).
\]
Its fundamental form
\[
 \omega(X,Y)=g(JX,Y)
\]
is a nondegenerate real two-form. Compatible metrics always exist.

\begin{proposition}
Every almost complex structure admits a compatible Riemannian metric.
\end{proposition}

\begin{proof}
Start with any Riemannian metric \(g_0\) and set
\[
 g(X,Y)=\frac12\bigl(g_0(X,Y)+g_0(JX,JY)\bigr).
\]
This is positive definite, and
\[
 g(JX,JY)
 =\frac12\bigl(g_0(JX,JY)+g_0(J^2X,J^2Y)\bigr)
 =g(X,Y).
\]
\end{proof}

An almost Hermitian manifold is \emph{almost Kähler} when
\[
 d\omega=0.
\]
It is \emph{Kähler} when \(J\) is integrable and \(d\omega=0\).
For an almost Hermitian structure,
\[
 \nabla^{TM}J=0
 \quad\Longleftrightarrow\quad
 N_J=0\ \text{and}\ d\omega=0.
\]
Thus parallel \(J\) characterizes the Kähler condition, not the almost
Kähler condition alone.

\begin{figure}[htbp]
\centering
\[
\begin{tikzcd}[column sep=huge,row sep=large]
 &\nabla^{TM}J=0
   \arrow[d,leftrightarrow]&\\
 \text{almost Kähler}
 &\text{Kähler}
   \arrow[l,swap,"d\omega=0"]
   \arrow[r,"N_J=0"]
 &\text{complex Hermitian}
\end{tikzcd}
\]
\caption{Kähler geometry is the simultaneous symplectic and integrable
specialization of almost Hermitian geometry.}
\end{figure}

\begin{discussion}
The two lower conditions are independent in general. A symplectic
manifold with a compatible almost complex structure is almost Kähler, but
the chosen \(J\) need not be integrable. Conversely, a Hermitian metric on
a complex manifold need not have closed fundamental form. Kähler geometry
is the intersection of the symplectic and integrable conditions.
\end{discussion}

The Hermitian metric induces an inner product on \((p,q)\)-forms. The
complex-linear extension of the Hodge star has type
\[
 *:\Omega^{p,q}(M)\longrightarrow\Omega^{m-q,m-p}(M).
\]
On a compact complex Hermitian manifold, define
\[
 \Box_{\bar\partial}
 =
 \bar\partial\bar\partial^*
 +\bar\partial^*\bar\partial.
\]
This is elliptic, and Hodge theory gives
\[
 H^{p,q}_{\bar\partial}(M)
 \cong
 \ker\bigl(\Box_{\bar\partial}:\Omega^{p,q}\to\Omega^{p,q}\bigr).
\]

\begin{theorem}[Kähler identities]
Let \(L=\omega\wedge(-)\) and let \(\Lambda=L^*\). Then
\[
 [\Lambda,\partial]=i\bar\partial^*,\qquad
 [\Lambda,\bar\partial]=-i\partial^*.
\]
Consequently,
\[
 \Box_\partial=\Box_{\bar\partial},
 \qquad
 \Delta_d=2\Box_{\bar\partial}.
\]
\end{theorem}

\begin{proof}
At a fixed point choose a unitary frame whose connection coefficients
vanish. There \(L\) is a sum of products of holomorphic and
antiholomorphic exterior multiplications, while \(\Lambda\) is the
corresponding sum of contractions. The relation
\[
 \iota_v\varepsilon(\alpha)
 +\varepsilon(\alpha)\iota_v=\alpha(v)
\]
gives the two commutator identities term by term. Both sides are
tensorial, so the pointwise calculation is global.

The graded Jacobi identity and
\(\partial\bar\partial+\bar\partial\partial=0\) give
\[
 \Box_{\bar\partial}
 =-i[\bar\partial,[\Lambda,\partial]]
 =i[\partial,[\Lambda,\bar\partial]]
 =\Box_\partial.
\]
The same identities cancel the mixed terms in \(\Delta_d\), leaving
\(\Box_\partial+\Box_{\bar\partial}=2\Box_{\bar\partial}\).
\end{proof}

It follows that
\[
 H^k(M;\C)
 =
 \bigoplus_{p+q=k}H^{p,q}_{\bar\partial}(M)
\]
on every compact Kähler manifold.

\section{The almost Dolbeault operator}

On an almost complex manifold the type decomposition exists, but the
exterior derivative has four components:
\[
 d=\mu+\partial+\bar\partial+\bar\mu,
\]
of bidegrees
\[
 (2,-1),\qquad(1,0),\qquad(0,1),\qquad(-1,2).
\]
The components \(\mu\) and \(\bar\mu\) are algebraic expressions in the
Nijenhuis tensor and vanish exactly when \(J\) is integrable. Thus
\(\bar\partial\) exists by projection for every almost complex structure,
but \(\bar\partial^2\) need not vanish.

Nevertheless,
\[
 S=\Lambda^{0,\bullet}T^*M,\qquad
 S^\pm=\Lambda^{0,\mathrm{ev/odd}}T^*M
\]
is the spinor bundle of the canonical spin\(^{c}\) structure. On a real
covector \(\xi\), Clifford multiplication is
\[
 c(\xi)=\sqrt2\left(
 \varepsilon(\xi^{0,1})
 -\iota_{(\xi^\sharp)^{0,1}}
 \right),
\qquad c(\xi)^2=-|\xi|^2.
\]
Choose a Hermitian connection preserving \(J\) and \(g\). It induces a
Clifford connection on \(S\), hence a Dirac operator
\[
 D_J:C^\infty(S^+)\longrightarrow C^\infty(S^-).
\]
Its principal symbol is \(i\,c(\xi)\), so \(D_J\) is elliptic. The
operator \(\sqrt2(\bar\partial+\bar\partial^*)\) has the same principal
symbol and differs from \(D_J\) by a zero-order torsion term. Therefore
the two operators have the same index even when \(\bar\partial\) is not a
differential.

\begin{theorem}[Almost-complex Riemann--Roch]
Let \(M^{2m}\) be compact almost complex and let \(E\) be a complex vector
bundle. For the canonical spin\(^{c}\) Dirac operator twisted by \(E\),
\[
 \operatorname{ind}D_{J,E}^+
 =
 \int_M
 \left[
 \operatorname{Td}(T^{1,0}M)\operatorname{ch}(E)
 \right]_{[2m]}.
\]
\end{theorem}

\begin{proof}
The determinant line of the canonical spin\(^{c}\) structure is
\[
 K_M^{-1}=\det T^{1,0}M.
\]
The spin\(^{c}\) index theorem gives
\[
 \operatorname{ind}D_{J,E}^+
 =
 \int_M
 \left[
 e^{c_1(K_M^{-1})/2}\Ahat(TM)\operatorname{ch}(E)
 \right]_{[2m]}.
\]
If \(x_1,\ldots,x_m\) are the formal Chern roots of \(T^{1,0}M\), then
\[
 e^{x_j/2}\frac{x_j/2}{\sinh(x_j/2)}
 =\frac{x_j}{1-e^{-x_j}}.
\]
Multiplying these identities yields
\[
 e^{c_1(K_M^{-1})/2}\Ahat(TM)
 =\operatorname{Td}(T^{1,0}M),
\]
which proves the formula.
\end{proof}

\begin{question}
What is lost when the almost complex structure is not integrable?
\end{question}

\begin{answer}
Not the elliptic symbol and not its index. What is lost is the cochain
complex: \(\bar\partial^2\) may be nonzero, so there is no Dolbeault
cohomology whose alternating dimension equals the index. Integrability
turns the same spin\(^{c}\) index into the holomorphic Euler
characteristic.
\end{answer}

\section{Holomorphic vector bundles}

A holomorphic structure on a complex vector bundle \(E\to M\) is an
operator
\[
 \bar\partial_E:\Omega^{0,q}(E)\longrightarrow\Omega^{0,q+1}(E)
\]
satisfying the Leibniz rule and
\[
 \bar\partial_E^2=0.
\]
It defines the Dolbeault complex
\[
 0\longrightarrow\Omega^{0,0}(E)
 \xrightarrow{\bar\partial_E}\Omega^{0,1}(E)
 \longrightarrow\cdots
 \longrightarrow\Omega^{0,m}(E)
 \longrightarrow0.
\]
Its cohomology is denoted \(H^{0,q}(M;E)\), or \(H^q(M,\mathcal O(E))\).

\begin{proposition}
The Dolbeault complex is elliptic.
\end{proposition}

\begin{proof}
At a nonzero real covector \(\xi\), the principal symbol of
\(\bar\partial_E\) is exterior multiplication by \(i\xi^{0,1}\). The symbol
complex is therefore the Koszul complex
\[
 \Lambda^{0,\bullet}T_x^*M\otimes E_x
 \xrightarrow{\;\xi^{0,1}\wedge\;}
 \Lambda^{0,\bullet+1}T_x^*M\otimes E_x.
\]
Contraction by the dual of \(\xi^{0,1}\), divided by
\(|\xi^{0,1}|^2\), is a contracting homotopy. Hence the symbol complex is
exact.

Indeed, if
\[
 h_\xi=\frac{\iota_{(\xi^{0,1})^\sharp}}{|\xi^{0,1}|^2},
\]
then exterior multiplication and contraction satisfy
\[
 h_\xi(\xi^{0,1}\wedge)
 +(\xi^{0,1}\wedge)h_\xi=I.
\]
For a nonzero real covector \(\xi\), its \((0,1)\)-part cannot vanish:
complex conjugation would also make its \((1,0)\)-part vanish. Thus the
homotopy is defined at every point of \(T^*M\setminus0\).
\end{proof}

Given a Hermitian metric on a holomorphic bundle, there is a unique unitary
connection whose \((0,1)\)-part is \(\bar\partial_E\). It is the Chern
connection. Its curvature has type \((1,1)\).

In a holomorphic frame with Hermitian metric matrix \(h\), the Chern
connection has
\[
 \nabla^{0,1}=\bar\partial,
 \qquad
 \nabla^{1,0}=\partial+h^{-1}\partial h,
\]
with the placement of \(h^{-1}\) determined by the column-vector
convention. Metric compatibility gives this formula, hence uniqueness.
Since the \((0,1)\)-part is integrable and the \((2,0)\)-part cancels by
the Maurer--Cartan identity, the curvature is of type \((1,1)\).

The Dolbeault--Dirac operator is
\[
 D_E=\sqrt2\,(\bar\partial_E+\bar\partial_E^*).
\]
It acts from even to odd antiholomorphic degree and has the principal symbol
of the canonical spin\(^{c}\) Dirac operator twisted by \(E\). On a Kähler
manifold the operators agree; in general they differ, for standard
connection choices, by a zero-order torsion term and have the same index.
Hodge theory gives
\[
 \ind D_E^+
 =
 \sum_{q=0}^m(-1)^q
 \dim H^{0,q}(M;E)
 =:\chi(M,E).
\]

The logical chain is
\begin{figure}[htbp]
\centering
\[
\begin{tikzcd}[column sep=large,row sep=large]
 (\Omega^{0,\bullet}(E),\bar\partial_E)
   \arrow[r,"\text{Hodge theory}"]
 & \ker D_E
   \arrow[r,"\text{graded dimension}"]
 & \operatorname{ind}D_E^+ \\
 H^{0,\bullet}(M;E)
   \arrow[u,"\cong"]
   \arrow[rr,"\sum(-1)^q\dim"']
 && \chi(M,E)\arrow[u,equal].
\end{tikzcd}
\]
\caption{For an integrable complex structure, Dolbeault Hodge theory
identifies the Dirac index with the holomorphic Euler characteristic.}
\end{figure}

\section{Hirzebruch--Riemann--Roch}

The canonical spin\(^{c}\) structure of an almost complex manifold has
determinant line \(K_M^{-1}\), where
\[
 K_M=\Lambda^{m,0}T^*M.
\]
The spin\(^{c}\) Dirac index formula is
\[
 \ind D_E^+
 =
 \int_M
 \left[
 e^{c_1(K_M^{-1})/2}
 \Ahat(TM)
 \operatorname{ch}(E)
 \right]_{[2m]}.
\]
For a complex vector bundle \(V\),
\[
 \operatorname{Td}(V)
 =
 e^{c_1(V)/2}\Ahat(V_\mathbb R).
\]
Taking \(V=T^{1,0}M\) yields the Riemann--Roch density.

This identity is checked one Chern root at a time:
\[
 e^{x/2}\frac{x/2}{\sinh(x/2)}
 =\frac{x}{1-e^{-x}}.
\]
The splitting principle then proves it for every complex vector bundle.

\begin{theorem}[Hirzebruch--Riemann--Roch]
If \(M\) is a compact complex manifold and \(E\) is a holomorphic vector
bundle, then
\[
 \boxed{
 \chi(M,E)
 =
 \int_M
 \left[
 \operatorname{Td}(T^{1,0}M)
 \operatorname{ch}(E)
 \right]_{[2m]}.
 }
\]
\end{theorem}

\begin{proof}
The analytic index of the Dolbeault--Dirac operator is \(\chi(M,E)\) by
Dolbeault Hodge theory. It has the same principal symbol as the canonical
spin\(^{c}\) Dirac operator twisted by \(E\); their difference is
zero-order, so their indices agree. Apply the spin\(^{c}\) Dirac index
theorem and use
\[
 c_1(K_M^{-1})=c_1(T^{1,0}M),
 \qquad
 e^{c_1/2}\Ahat(TM)=\operatorname{Td}(T^{1,0}M).
\]
\end{proof}

The same characteristic number is defined for a compact almost complex
manifold and equals the index of its canonical spin\(^{c}\) Dirac operator.
Integrability is needed only to identify that index with holomorphic
sheaf cohomology.

\section{Riemann surfaces}

Let \(X\) be a compact connected Riemann surface of genus \(g\), and let
\(L\to X\) be a holomorphic line bundle. Its degree is
\[
 \deg L=\int_Xc_1(L).
\]
Since only degree-two terms contribute,
\[
 \operatorname{ch}(L)=1+c_1(L),
 \qquad
 \operatorname{Td}(T^{1,0}X)
 =
 1+\frac12c_1(T^{1,0}X).
\]
Gauss--Bonnet gives
\[
 \int_Xc_1(T^{1,0}X)=\chi(X)=2-2g.
\]
Hirzebruch--Riemann--Roch therefore becomes
\[
 \chi(X,L)=\deg L+1-g.
\]

\begin{theorem}[Serre duality for a curve]
There is a natural perfect pairing
\[
 H^1(X,L)\times H^0(X,K_X\otimes L^{-1})
 \longrightarrow\C,
 \qquad
 ([\alpha],s)\longmapsto\int_X\alpha\wedge s.
\]
Hence
\[
 H^1(X,L)\cong H^0(X,K_X\otimes L^{-1})^*.
\]
\end{theorem}

\begin{proof}
First, the pairing is well defined on cohomology. If
\(\alpha\) is replaced by \(\alpha+\bar\partial_Eu\) and \(s\) is
holomorphic, then
\[
 \int_X(\bar\partial_Eu)\wedge s
 =\int_X\bar\partial(u\wedge s)=0
\]
by Stokes' theorem.

Dolbeault theory identifies \(H^1(X,L)\) with harmonic \(L\)-valued
\((0,1)\)-forms. The Hermitian Hodge star, followed by fibrewise duality,
identifies this harmonic space anti-linearly with holomorphic sections of
\(K_X\otimes L^{-1}\). To see holomorphicity, use
\(\bar\partial_E\alpha=0\) and \(\bar\partial_E^*\alpha=0\); under the star,
the second equation becomes the \(\bar\partial\)-equation for the dual
section. Under this identification, the \(L^2\) inner product is the
displayed integral pairing. A nonzero harmonic \(\alpha\) pairs with its
star-dual to give \(\|\alpha\|_{L^2}^2>0\), and conversely. The pairing is
therefore nondegenerate in both factors, proving the duality.
\end{proof}

Combining Serre duality with the index formula gives the classical
Riemann--Roch theorem.

\begin{theorem}[Riemann--Roch for a line bundle]
\[
 \boxed{
 h^0(X,L)-h^0(X,K_X\otimes L^{-1})
 =
 \deg L+1-g.
 }
\]
\end{theorem}

\begin{proof}
Dolbeault theory gives
\[
 \chi(X,L)=h^0(X,L)-h^1(X,L).
\]
Serre duality identifies
\[
 h^1(X,L)=h^0(X,K_X\otimes L^{-1}),
\]
while Hirzebruch--Riemann--Roch in complex dimension one gives
\[
 \chi(X,L)
 =\int_X\left(c_1(L)+\frac12c_1(T^{1,0}X)\right)
 =\deg L+1-g.
\]
Combining the three equalities proves the formula.
\end{proof}

If a holomorphic line bundle has a nonzero holomorphic section, its degree
is the number of zeros of that section counted with multiplicity and is
therefore nonnegative. Thus, if \(\deg L>2g-2\), then
\[
 \deg(K_X\otimes L^{-1})<0
\]
and the second term vanishes:
\[
 h^0(X,L)=\deg L+1-g.
\]

\begin{discussion}
The last vanishing uses only a simple divisor argument. A nonzero
holomorphic section \(s\) has isolated zeros, and the local orders of
vanishing sum to \(\deg L\). Hence a negative-degree line bundle cannot
have a nonzero holomorphic section. Since
\(\deg K_X=2g-2\), the bundle \(K_X\otimes L^{-1}\) has negative degree
under the displayed hypothesis.
\end{discussion}
