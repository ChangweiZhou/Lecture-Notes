\begin{abstract}
Let \(X\) be a compact \(n\)-manifold with boundary, let \(\partial X\) have
\(N\) components, and let \(x\) be a boundary defining function. We
compute the continuous Hochschild homology of the algebra of
b-pseudodifferential operators of integral order with Laurent coefficients
at the boundary, of its quotient
\(\Ab=x^{-\Z}\Psi_b^{\Z}(X)/x^\infty\Psi_b^{-\infty}(X)\), and of the
ideals and quotients cut out by the two natural filtrations, by operator
order and by powers of \(x\). This is the b-calculus analogue of the
computation of Melrose and Nistor for the cusp calculus. The final groups
have the same form as theirs: they are expressed through the de Rham
cohomology of \(\Sbstar\), of its boundary, and of two auxiliary circles
recording symbol order and boundary order. The proofs differ at the
boundary. Conjugation by \(x\) acts on indicial families by the complex
shift \(\lambda\mapsto\lambda-i\), so the boundary algebras are Laurent
series twisted by an automorphism rather than by a derivation, and their
coefficients are entire functions of \(\lambda\) rather than Schwartz
functions. We compute the homology of the boundary ideal by a weight
decomposition and a crossed-product argument. We identify the connecting
map to the residual ideal with the b-trace defect formula and evaluate the
index functional on residual perturbations of the identity, where it
reduces to a Wiener--Hopf winding number and to \(-\tfrac12\) times the
b-eta invariant. The computation for the Laurent symbol algebra
\(\Asb\), which is the b-groupoid case of the Benameur--Nistor theorem,
is retained as Part~\ref{part:symbols}. Each step is labelled as proved,
quoted, or sketched; see Section~\ref{subsec:status}.
\end{abstract}

\section{Introduction}\label{sec:intro}

\subsection{The algebras}\label{subsec:algebras}
Let \(X\) be a compact manifold with boundary of dimension \(n\). Its
boundary \(\partial X=\bigsqcup_{H}H\) has \(N\) components. Fix a
boundary defining function \(x\), a collar
\([0,\varepsilon)_x\times\partial X\hookrightarrow X\), and a smooth positive
density \(\nu_\partial\) on \(\partial X\). Let \(\Psi_b^m(X)\), \(m\in\Z\),
be the small calculus of classical b-pseudodifferential operators
(Section~\ref{sec:bconv}). Conjugation by powers of \(x\) preserves each
\(\Psi_b^m(X)\) (Lemma~\ref{lem:conj}), so
\[
 x^{-\Z}\Psi_b^{\Z}(X)=\bigcup_{\ell\in\Z,\,m\in\Z}x^{\ell}\Psi_b^m(X)
\]
is an algebra. Following Melrose--Nistor \cite[(8)--(10)]{MN}, we consider
its ideal of operators that are residual for both filtrations and the
quotients by it:
\begin{align*}
 \Ib&=x^\infty\Psi_b^{-\infty}(X)=\bigcap_{\ell}x^\ell\Psi_b^{-\infty}(X),
 &\Ab&=x^{-\Z}\Psi_b^{\Z}(X)/\Ib,\\
 \Idb&=x^{-\Z}\Psi_b^{-\infty}(X)/\Ib,
 &\Isb&=x^{\infty}\Psi_b^{\Z}(X)/\Ib,\\
 \Asb&=\Ab/\Idb,\qquad \Adb=\Ab/\Isb,
 &\Adsb&=\Ab/(\Isb+\Idb).
\end{align*}
The kernels of elements of \(\Ib\) are exactly the smooth functions on
\(X^2\) that vanish to infinite order at \(\partial(X^2)\). The algebra
\(\Asb\) is the algebra of Laurent b-complete symbols treated in
Part~\ref{part:symbols}, \(\Adb\) is the algebra of Laurent series of
indicial operators, and \(\Adsb\) is the algebra of Laurent series of
complete symbols at the boundary. These algebras form the diagram of
extensions in Figure~\ref{fig:extensions}, which is the b-version of the
diagram in \cite[p.~4]{MN}.

\subsection{The main theorem}\label{subsec:mainthm}
Set \(Y=\Sbstar\) and \(Z=Y|_{\partial X}={}^bS^*_{\partial X}X\), and let
\(Z_H\) be the part of \(Z\) over the component \(H\). Let \(S_\sigma\) and
\(S_\partial\) be circles, and write \(d\theta_\sigma\) and
\(d\theta_\partial\) for their generators; they correspond to \(d\log r\)
(\(r\) a fibre norm) and to \(d\log x\). Homology is continuous Hochschild
homology in the chain model of Section~\ref{subsec:chainmodel}.
Cohomology has complex coefficients. For \(\beta\) in \(H^{2n-2}(Z)\), or
in \(H^{2n-2}(Y)\) after restriction to \(Z\), write
\(\int_{Z_H}\beta\) for its integral over \(Z_H\), oriented as part of
\(\partial Y\).

\emph{Standing hypotheses.} \(X\) is connected and \(\partial X\neq
\varnothing\). For \(n\geq2\) every \(Z_H\) is connected and oriented. The
case \(n=1\) is treated in Remark~\ref{rem:n1}.

\begin{theorem}[Hochschild homology of the b-calculus]\label{thm:full}
Assume the standing hypotheses, hypothesis~\textup{(C)} of
Section~\ref{sec:Is}, and the inputs listed in
Section~\ref{subsec:status}. Parts (i)--(iii) and (ix) hold for every
\(n\geq1\). Parts (iv)--(viii) are stated for \(n\geq2\).
\begin{enumerate}
\item[(i)] \(\HH_0(\Ib)\cong\C\), with dual basis the trace, and
\(\HH_k(\Ib)=0\) for \(k>0\).
\item[(ii)] \(\HH_k(\Idb)\cong\C^N\) for \(k=0,1\), and \(\HH_k(\Idb)=0\)
for \(k\geq2\). The boundary traces \(\Tr_{\partial,H}\) of \eqref{eq:Trd}
form a dual basis in degree \(0\), and the cyclic cocycles \(\varphi_H\) of
\eqref{eq:phi} form a dual basis in degree \(1\).
\item[(iii)] \(\HH_0(x^{-\Z}\Psi_b^{-\infty}(X))\cong\C^N\),
\(\HH_1(x^{-\Z}\Psi_b^{-\infty}(X))\cong\C^{N-1}\), and the higher groups
vanish. The connecting map \(\HH_1(\Idb)\to\HH_0(\Ib)\) is the b-trace
defect.
\item[(iv)] \(\HH_k(\Isb)\cong H^{2n-k}(Y\times S_\sigma,Z\times
S_\sigma)\).
\item[(v)] \(\HH_k(\Asb)\cong H_L^{2n-k}(Y\times S_\sigma)\cong
H^{2n-k}(Y\times S_\sigma)\oplus H^{2n-k-1}(Z\times S_\sigma)\).
\item[(vi)] \(\HH_k(\Adsb)\cong H^{2n-k}(Z\times S_\sigma\times
S_\partial)\).
\item[(vii)] \(\HH_k(\Adb)\cong\HH_k(\Adsb)\) for \(k\neq1,2\). For
\(k=1,2\), \(\HH_k(\Adb)\) is the kernel of a surjection
\(\HH_k(\Adsb)\to\C^N\). For \(k=1\) the surjection is
\(\beta\mapsto(\int_{Z_H}\beta_\partial)_H\), where \(\beta_\partial\) is
the coefficient of \(d\theta_\partial\). For \(k=2\) it is
\(\beta\mapsto(\int_{Z_H}\beta_0)_H\), where \(\beta_0\) is the component
free of both circle factors.
\item[(viii)] \(\HH_k(\Ab)\cong\HH_k(\Asb)\) for \(k\neq1,2\). Moreover
\begin{align*}
 \HH_1(\Ab)&\cong\C\oplus H^{2n-1}(Y\times S_\sigma)\oplus
 H^{2n-3}(Z),\\
 \HH_2(\Ab)&\cong\ker\Bigl(H^{2n-2}(Y)\xrightarrow{(\int_{Z_H})_H}\C^N
 \Bigr)\oplus H^{2n-3}(Y)\oplus H^{2n-3}(Z\times S_\sigma).
\end{align*}
The topological map \(H^{2n-2}(Y)\to\C^N\) has rank \(N-1\), by Stokes'
theorem and Lefschetz duality. The rank of \(\partial^{\mathcal A}_2\) is
\(\leq N-1\), which is proved given (ii). That \(\partial^{\mathcal A}_2\)
equals the topological map, and the explicit kernels in (vii), depend on (C)
and on the sketch (S2) (Remark~\ref{rem:description}). If \(N=1\), then
\(\HH_2(\Ab)\cong\HH_2(\Asb)\).
\item[(ix)] The index functional, which is the image of the trace under
\(\partial:\HH_1(\Ab)\to\HH_0(\Ib)\), is nonzero. Hence
\(\HH_k(x^{-\Z}\Psi_b^{\Z}(X))\cong\HH_k(\Ab)\) for \(k\neq1\), and
\(\HH_1(x^{-\Z}\Psi_b^{\Z}(X))\) is the kernel of the index functional on
\(\HH_1(\Ab)\).
\end{enumerate}
\end{theorem}

The direct-sum decompositions in (v), (vii) and (viii) depend on choices
(a collar and the splittings of Proposition~\ref{prop:face}); the groups do
not. Remark~\ref{rem:choices} discusses how the generators depend on \(x\)
and on the collar. If \(X\) has \(c\) components, each with nonempty boundary, the rank
in (viii) becomes \(N-c\), and the summand \(\C\) in \(\HH_1(\Ab)\)
becomes \(\C^{c}\) (Remark~\ref{rem:MNcomponents}). Components without
boundary contribute as in the closed case.

For \(N=1\), parts (ii) and (iv)--(viii) have the same form as the
cusp-calculus result of Melrose and Nistor \cite[(16)]{MN}, which is stated
for Hochschild cohomology, with \({}^cS^*X\) replaced by \({}^bS^*X\). Both
cosphere bundles are diffeomorphic to \(S^*X\)
(Proposition~\ref{prop:anchor}). The bookkeeping for several components is
new.

\subsection{Relation to Melrose--Nistor and novelty}\label{subsec:MN}
\begin{remark}[Comparison with Melrose--Nistor]\label{rem:MN}
Melrose and Nistor \cite{MN} compute the Hochschild homology of the full
\emph{cusp} calculus \(x^{-\Z}\Psi_c^{\Z}(X)\). They say that they use the cusp
calculus rather than the b-calculus ``to avoid certain problems with
completeness''. They also remark that the cusp commutation relation
\([x^2\partial_x,x]=x^2\), in place of the b-relation
\([x\partial_x,x]=x\), ``makes quite a lot of difference'', and that
``methods similar to those used here can be applied directly to the
b-calculus'' \cite[pp.~3--4]{MN}. The present notes carry this out. The
differences are as follows.
\begin{enumerate}
\item \emph{Twist.} In the cusp case the boundary algebras are Laurent
series \(\Psi_{\mathrm{sus}}(\partial X)[[x]\) in which \(x\) and the
suspension variable satisfy \([x,\partial_t]=x^2\) \cite[(14)]{MN}; to
leading order the twist is a derivation. In the b-case,
\(x^{-1}Ax\) has indicial family \(I(A,\lambda-i)\)
(Lemma~\ref{lem:conj}), so the twist is the \emph{automorphism}
\(\alpha\) of complex translation by \(-i\). The associated graded
algebra for the \(x\)-filtration is therefore not commutative, and
Melrose--Nistor's \(x\)-adic spectral sequence \cite[proof of
Prop.~6]{MN} is replaced by a weight decomposition and a crossed-product
computation (Section~\ref{sec:Id}).
\item \emph{Entire indicial families.} Indicial families in the small
b-calculus are entire in \(\lambda\) and suspended on every horizontal
strip; residual ones are Schwartz on every strip. The cusp indicial
families are only Schwartz, or symbols, on the real line. Analyticity is
what makes complex shifts possible. It is also the source of the
completeness problem mentioned by Melrose--Nistor: an operator is
invertible modulo \(\Ib\) only if its indicial family is invertible for
\emph{all} \(\lambda\in\C\) (Lemma~\ref{lem:invertible}). Geometric Dirac
operators never satisfy this.
\item \emph{Traces.} The boundary trace on \(\Idb\) is the residue of
\(\Tr(x^w\cdot)\) at \(w=0\). The b-trace is the finite part of the same
function. The connecting map for \(\Ib\subset x^{-\Z}\Psi_b^{-\infty}\) is
Melrose's b-trace defect formula (Theorem~\ref{thm:defect}). The extension
of the boundary trace to \(\Adb\) is Melrose's regularized suspended trace.
Its Hochschild boundary is given by a \emph{shift defect}
(Proposition~\ref{prop:omega}), which replaces \cite[Lemma~13]{MN}.
\end{enumerate}
Melrose--Nistor's Appendix~C \cite[Prop.~26]{MN} lifts b-operators to cusp
operators by the transcendental blow-up \(t=e^{-1/x}\). This lift is useful
for index formulas, but it does not identify the Laurent algebras: Laurent
series in \(t\) and Laurent series in \(x=1/\log(1/t)\) are different
algebras.
\end{remark}

\emph{Novelty.} We have not located a published computation of the
Hochschild homology of the b-calculus operator algebra.
Theorem~\ref{thm:full} should therefore be read as a careful execution of
the programme suggested in \cite{MN}, not as a known result. The symbol
part (v) is the b-groupoid case of the Benameur--Nistor theorem
\cite{BNsymbols,BNfamilies}.

\subsection{Status of the proofs}\label{subsec:status}
We separate what is proved here from what is quoted, sketched, or
assumed. Melrose--Nistor \cite{MN} are cited only for comparison; no
result of theirs is used as an input.

\emph{Proved in these notes.}
\begin{itemize}
\item The structure theorems for \(\Idb\) and \(\Adb\) as twisted Laurent
series (Propositions~\ref{prop:structure} and~\ref{prop:structureA}).
These rest on (Q0) and (C3), and give only an algebraic surjection; no
continuous section is claimed.
\item The explicit H-unitality contractions for \(\Ib\),
\(\Psi^{-\infty}(H)\) and \(\Idb\) (Lemma~\ref{lem:Hunital}).
\item The Morita lemma for smoothing kernels (Lemma~\ref{lem:morita}), and
part (i), which also uses the kernel theorem (C3).
\item The vanishing of the nonzero weights for \(\Idb\)
(Proposition~\ref{prop:weights}), given the excision theorem (Q1).
\item The coinvariants and invariants of the shift (Lemma~\ref{lem:coinv}).
\item The trace \(\Tr_\partial\), the cocycle \(\varphi\), their
nonvanishing, and their independence of choices.
\item The b-trace defect formula on \(x^{-\Z}\Psi_b^{-\infty}\)
(Theorem~\ref{thm:defect}) and the connecting map in (iii).
\item The de Rham lemmas for flat and formal Laurent forms, and, given
(C-HKR), the \(E^2\)-pages of \(\Isb\) and \(\Adsb\)
(Section~\ref{sec:Is}).
\item The shift-defect formula for the extended boundary trace, and its
role as connecting cocycle given (Q1) (Proposition~\ref{prop:omega}). The
residue formula (Lemma~\ref{lem:rhohat}), and the top-order formulas
(Lemma~\ref{lem:top}); part (b) assumes (C-HKR).
\item The algebraic convergence and degeneration theorem
(Proposition~\ref{prop:algconv}).
\item For every \(n\), the surjectivity of the boundary connecting maps,
given (C), (ii) (hence (Q1), (Q2), (S1)) and (Q1). The proof uses
\(\delta_b\)-closed test forms built from the divergence-free field
\(\zeta|\zeta|^{-n}\) (Proposition~\ref{prop:dict}). Their explicit
description is the sketch (S2).
\item For \(n=1\), the surjectivity of both boundary connecting maps of
\(\Adb\) and of \(\partial^{\mathcal A}_1\), by explicit test cycles. The
identification of connecting maps uses (Q1).
\item The fibre-product description of \(\Ab\), and the bound \(N-1\) on the
rank of \(\partial^{\mathcal A}_2\). The bound uses
\(\dim\HH_1(\Idb)=N\) from (ii), and so inherits (Q1), (Q2) and (S1).
\item The invertibility criterion (Lemmas~\ref{lem:autoinv}
and~\ref{lem:invertible}).
\item The Wiener--Hopf index formula for residual perturbations of the
identity (Theorem~\ref{thm:WH}), and the nonvanishing of the index
functional. Part (ix) uses in addition (i), (Q0), (Q1) and (C3).
\end{itemize}

\emph{Quoted inputs.}
\begin{enumerate}
\item[(Q0)] The small b-composition theorem, including the ideal property
\(x^K\Psi_b^m\circ\Psi_b^{m'}\subset x^K\Psi_b^{m+m'}\), and the
trace-class property of \(x^\varepsilon\Psi_b^{-\infty}\) for
\(\varepsilon>0\) \cite{MelroseAPS,Notes}.
\item[(Q1)] Wodzicki's excision theorem in the chain model of
Section~\ref{subsec:chainmodel} (Theorem~\ref{thm:excision}). Its
topological input, exactness of the chain spaces, is proved
(Lemma~\ref{lem:exactchains}).
\item[(Q2)] The weight-zero part of the Hochschild homology of a crossed
product by \(\Z\) (Theorem~\ref{thm:crossed}). The cited proofs are

\item[(C3)] Standard analysis and topology. This includes: nuclearity of

exact on nuclear Fréchet spaces \cite{Treves}; the Schwartz kernel theorem
\(\dot C^\infty(X^2)=\dot C^\infty(X)\widehat\otimes\dot C^\infty(X)\); Taylor
expansion and asymptotic summation of conormal kernels at a face meeting the
conormal set transversally; the suspended calculus on strips; the identity
\(\Tr(PQ)=\Tr(QP)\) when \(\operatorname{ord}P+\operatorname{ord}Q<-\dim H\);
mapping properties on weighted b-Sobolev spaces and the
Calder\'on--Fedosov formula; and Lefschetz duality.
\end{enumerate}

\emph{Hypothesis.}
\begin{enumerate}
\item[(C)] For \(\Asb\), \(\Isb\) and \(\Adsb\): (C-HKR), the continuous HKR
identification of \(E^1\), and (C-stage), the completeness of the order
filtration in the chain model (Section~\ref{subsec:conv}). The algebraic
convergence and degeneration theorem is now proved
(Proposition~\ref{prop:algconv}), and the \(E^2\)-pages are computed here,

For \(\Asb\), (C) is part of the Benameur--Nistor theorem
(Theorem~\ref{thm:BN-conv}); for \(\Isb\) and \(\Adsb\) it is a hypothesis.
\end{enumerate}

\emph{Sketched.}
\begin{enumerate}
\item[(S1)] The continuous HKR theorem for the algebra \(\cE\) of entire
functions that are Schwartz on strips. We prove the division lemma and the
exactness of the Koszul complex, but only sketch the comparison of the
Koszul and bar complexes for this non-unital algebra
(Proposition~\ref{prop:HKRE}). The proof of Connes' theorem in
\cite{MelroseMicro} does not transfer: its normalization step inserts the
unit (dummy variables), and its local-to-global step uses partitions of
unity. Neither is available for entire functions.
\item[(S2)] The explicit description of the boundary connecting maps as
\((\int_{Z_H}\cdot)_H\) under \(*_b\) (Remark~\ref{rem:description}). It
requires the exact constants and the cancellation of the equatorial
logarithmic divergences. The surjectivity of these maps is proved; their
description, the explicit kernels in (vii)--(viii), and the equality
\(\operatorname{rank}\partial^{\mathcal A}_2=N-1\) depend on (S2).
\item[(S3)] The general index formula for fully elliptic operators of
nonzero order (Proposition~\ref{prop:indexgeneral}). The closed-manifold
arguments \cite{MelroseMicro} need a holomorphic family \(Q(z)\) of order
\(-z\) with \(Q(0)=1\), and the meromorphy of \(\Tr(x^wAQ(z))\) in two
variables, including the double poles at the corner where the diagonal
meets the front face. In the b-setting \(Q(z)\) must have indicial families
that are entire and invertible on all of \(\C\); we have not carried this
out.
\end{enumerate}

Dependencies: (i) uses (C3). (ii) and (iii) use (Q0)--(Q2), (C3) and
(S1). (iv)--(vi) use (C). (vii) and (viii) inherit (Q1), (Q2) and (S1)
from (ii), and (C) from (iv)--(vi). The surjectivity of the connecting
maps is proved (Propositions~\ref{prop:n1}, \ref{prop:dict}
and~\ref{prop:connA}). The explicit kernels in (vii)--(viii), and the
exact rank of \(\partial^{\mathcal A}_2\), use (S2) in addition. (ix) uses
(i), (Q0), (Q1) and (C3), together with Theorem~\ref{thm:WH}.

\subsection{Organization}
Part~\ref{part:symbols} keeps the computation for the symbol algebra
\(\Asb\) (Sections~\ref{sec:symbol}--\ref{sec:checks}). Part~\ref{part:ideals}
treats the b-calculus conventions, the structure of the boundary algebras,

\part{The Laurent symbol algebra}\label{part:symbols}

\section{\texorpdfstring{The symbol algebra \(\Asb\)}{The symbol algebra}}\label{sec:symbol}

The phrase \emph{the algebra generated by b-pseudodifferential operators} is
ambiguous. At least four choices affect Hochschild homology: the allowed
operator orders, the allowed powers of a boundary defining function, the
residual ideal, and the locally convex completion. We use the standard algebra
for which the homology has a finite-dimensional geometric description.
It is the b-groupoid specialization of the Laurent symbol algebras studied in
\cite{BNsymbols,BNfamilies}.

\emph{Status of the result.} Theorem~\ref{thm:main} below is the
b-groupoid case of the Benameur--Nistor theorem on Laurent complete symbols
\cite{BNsymbols,BNfamilies}; it is not a new theorem. What these notes add
is a self-contained computation of the \(E^1\)- and \(E^2\)-pages of the
order spectral sequence in the b-setting
(Sections~\ref{sec:graded}--\ref{sec:laurent}), an elementary proof that
the spectral sequence degenerates at \(E^2\) (Section~\ref{sec:collapse}),
and a check of the hypotheses for the b-groupoid. Two inputs are quoted
from Benameur--Nistor: convergence of the order spectral sequence for the
completed continuous Hochschild complex, and the continuous, globalized
Hochschild--Kostant--Rosenberg theorem identifying \(E^1\).

For one boundary hypersurface, the standard b-groupoid has underlying set
\[
 \Gb=X^\circ\times X^\circ
 \;\sqcup\;
 \bigsqcup_{H\in\pi_0(\partial X)}H\times H\times\mathbb R.
\]
On a boundary component, composition adds the last coordinate. Its smooth
structure is fixed by the normal ratio
\(t=\log(x/x')\), and its Lie algebroid is
\(A(\Gb)={}^bTX\). For corners, one uses the standard groupoid with an
\(\mathbb R^j\)-factor over each open codimension-\(j\) face
\cite{BNfamilies}.

Let \(\cO(X)\) be the ring generated by \(C^\infty(X)\) and
\(x_H^{-1}\) for all boundary hypersurfaces \(H\). Benameur and Nistor
define
\[
 \cA_{b,L}(X;E)
 :=\cO(X)\left(
 \Psi^\infty(\Gb;E)/\Psi^{-\infty}(\Gb;E)
 \right)
\]
with their topologically filtered topology and completed Hochschild
complex \cite{BNsymbols,BNfamilies}. Here \(\Psi^\infty(\Gb;E)\) denotes
the \(\Gb\)-pseudodifferential operators acting on sections of \(E\) (their
kernels are distributional sections over \(\Gb\) of the pull-backs of
\(E\) and \(E^*\) by the range and source maps); we write \(E\) throughout. The vector representation identifies
this with the usual properly supported Laurent b-complete symbols. In
particular, one may write its underlying filtered union heuristically as
\[
 \frac{\bigcup_{m,\ell}x^{-\ell}\Psi_b^m(X;E)}
      {\bigcup_\ell x^{-\ell}\Psi_b^{-\infty}(X;E)},
\]
but this quotient notation does not by itself specify the topology used in
the theorem.

Changing the defining function from \(x\) to \(x'=e^f x\) gives
\[
 (x')^{-\ell}=e^{-\ell f}x^{-\ell}.
\]
Hence the change only multiplies a Laurent generator by a smooth
invertible function, and the
algebra, its filtration, and the answer below are canonically independent of
this choice. For corners one allows an arbitrary finite Laurent monomial
\(\prod_Hx_H^{-\ell_H}\), one factor for each boundary hypersurface.

Set \(Y=\Sbstar=(\Tbstar\setminus0)/\mathbb R_+\) and
\(Z=Y|_{\partial X}\). Thus \(\partial Y=Z\). The symbol-order circle is
denoted \(S^1_\sigma\).

\begin{theorem}[Hochschild homology; b-groupoid case of
Benameur--Nistor]\label{thm:main}
Assume the Benameur--Nistor convergence theorem for the order spectral
sequence of \(\cA_{b,L}(X;E)\) (Section~\ref{sec:collapse}). Then for every
\(k\geq0\) the induced filtration of
\(\HH_k^{\mathrm{cont}}(\cA_{b,L}(X;E))\) has exactly one nonzero graded
piece, and there is a canonical isomorphism
\begin{equation}\label{eq:main}
 \boxed{
 \HH_k^{\mathrm{cont}}(\cA_{b,L}(X;E))
 \cong H_L^{2n-k}(Y\times S^1_\sigma;\C).}
\end{equation}
The answer is independent of \(E\). If \(X\) has one embedded boundary
hypersurface, then
\begin{align}\label{eq:expanded}
 \HH_k^{\mathrm{cont}}(\cA_{b,L}(X;E))
 \cong{}& H^{2n-k}(Y)
 \oplus H^{2n-k-1}(Y) \notag\\
 &\oplus H^{2n-k-1}(Z)
 \oplus H^{2n-k-2}(Z),
\end{align}
where a cohomology group in negative degree is zero. In particular the
homology vanishes for \(k>2n\).
\end{theorem}

The isomorphism \eqref{eq:main} is canonical once convergence is granted;
the K\"unneth decomposition for \(S^1_\sigma\) is canonical as well. The
four-summand formula \eqref{eq:expanded} uses one choice: a splitting of
the Laurent face sequence (a collar, defining function, and cutoff;
Proposition~\ref{prop:face}). The subspace
\(H^{2n-k}(Y\times S^1_\sigma)\) and the boundary quotient
\(H^{2n-k-1}(Z\times S^1_\sigma)\) are canonical.

\begin{remark}[The algebraic Laurent quotient]\label{rem:Asb}
The underlying filtered algebra of \(\cA_{b,L}\) is
\(\Asb=x^{-\Z}\Psi_b^{\Z}(X)/x^{-\Z}\Psi_b^{-\infty}(X)\), the quotient of
the Introduction. Throughout, \(\Asb\) carries the Benameur--Nistor
topology, in which the projective completion in decreasing order is taken
at fixed pole order and the inductive limit is taken afterwards
\eqref{eq:BN-chains}. Accordingly \(\HH_*(\Asb)\) means
\(\HH_*^{\mathrm{cont}}(\cA_{b,L})\). Theorem~\ref{thm:main} has the same form
as the cusp result \cite[Prop.~2]{MN}, \(\HH_*(\cA_\sigma)\cong
{}^bH^{2n-*}({}^cS^*X\times S_\sigma)\); this is a comparison, and the
present proof does not use it. Here \(S^1_\sigma\) is the circle
\(S_\sigma\) of Theorem~\ref{thm:full}.
\end{remark}

\section{Continuous chains and the two filtrations}

The topology is part of the theorem.  For a topologically filtered algebra
\(A\), Benameur--Nistor do \emph{not} take the completed projective tensor
power of the whole locally convex space: multiplication need not be jointly
continuous for that topology.  In their notation the degree-\(k\) chain
space is
\begin{equation}\label{eq:BN-chains}
 C_k^{\mathrm{cont}}(A)
 :=\varinjlim_{N}
   \bigl(F^{N}_{N,N}A\bigr)^{\widehat\otimes_\pi(k+1)}.
\end{equation}
(Benameur--Nistor denote this chain space by \(H_k(A)\); we avoid that
notation, which suggests homology.) Here \(F^{N}_{N,N}A\) is the diagonal
cofinal family of filtration pieces of \cite{BNfamilies}; equivalently, one
first takes the projective completion in the decreasing symbol-order
direction and only then enlarges the support and Laurent bounds by an
inductive limit.  We apply this with \(A=\cA_{b,L}\).
Whenever a tensor product below carries no decoration it denotes an
elementary tensor in the dense filtered subcomplex of
\eqref{eq:BN-chains}, not an ordinary completed tensor power of the whole
algebra.

For a chain \(a_0\otimes\cdots\otimes a_k\), the Hochschild boundary is
\begin{align*}
 b(a_0\otimes\cdots\otimes a_k)
 ={}&\sum_{j=0}^{k-1}(-1)^j
 a_0\otimes\cdots\otimes a_ja_{j+1}\otimes\cdots\otimes a_k\\
 &+(-1)^ka_ka_0\otimes a_1\otimes\cdots\otimes a_{k-1}.
\end{align*}
Multiplication is continuous on fixed filtration pieces, so \(b\) extends to
the completed complex and satisfies \(b^2=0\).

Laurent order and symbol order play different roles. The first is built
into \(\cO(X)\); symbol order produces the spectral sequence.  On every
fixed piece in \eqref{eq:BN-chains}, filter elementary tensors by
\[
 \operatorname{ord}(a_0)+\cdots+\operatorname{ord}(a_k)\leq p,
\]
take the projective closure there, and then pass through the same
projective/inductive limits.  Denote the resulting subspace by
\(\mathcal F_pC_k\).
Then \(b(\mathcal F_pC_k)\subseteq\mathcal F_pC_{k-1}\). Exhaustivity,
completeness, and the convergence hypotheses are those verified in the
Benameur--Nistor topologically filtered theorem. Borel quantization explains
asymptotic completeness of symbols, but it is not by itself a convergence
proof for an arbitrarily completed Hochschild complex.

\begin{discussion}
\textbf{Why divide by smoothing operators?}
The principal-symbol filtration sees an operator only through its full
asymptotic symbol. Smoothing kernels form the kernel of that passage. Keeping
them gives an extension problem, treated separately in
Section~\ref{sec:unreduced}; it is not a harmless change of notation.
\end{discussion}

\section{The b-cotangent geometry}

The Lie algebra of b-vector fields is
\[
 \mathcal V_b(X)
 =\{V\in C^\infty(X;TX):V\text{ is tangent to }\partial X\}.
\]
It is the space of smooth sections of a vector bundle \({}^bTX\to X\).
Near a boundary point, choose product coordinates
\[
 (x,y_1,\ldots,y_{n-1}).
\]
Then
\[
 \mathcal V_b(X)
 =C^\infty(X)\text{-span}
 \{x\partial_x,\partial_{y_1},\ldots,\partial_{y_{n-1}}\}.
\]
The dual local frame of \(\Tbstar\) is
\[
 \frac{dx}{x},\quad dy_1,\ldots,dy_{n-1}.
\]
Thus a b-covector has the form
\[
 \xi\frac{dx}{x}+\eta\cdot dy.
\]

On \(\Tbstar\) the tautological b-one-form and its differential are
\[
 \alpha_b=\xi\frac{dx}{x}+\eta\cdot dy,
 \qquad
 \omega_b=d\alpha_b
 =d\xi\wedge\frac{dx}{x}+d\eta\wedge dy.
\]
The form \(\omega_b\) is nondegenerate as a b-two-form. Its inverse is the
b-Poisson tensor
\[
 \pi_b=x\partial_x\wedge\partial_\xi
       +\sum_j\partial_{y_j}\wedge\partial_{\eta_j}.
\]
This is the geometric bracket appearing in the first nonzero differential of
the Hochschild spectral sequence. We set \(\{a,c\}_b=\pi_b(da,dc)\). With
this convention \(\{\xi,x\}_b=-x\) and \(\{\eta_j,y_j\}_b=-1\). The
commutator rule below is \(\sigma([A,C])=i\{a,c\}_b\). It is consistent with
\([D_{y_j},y_j]=-i\) and with \([xD_x,x]=-ix\) of \eqref{eq:weightcomm},
because \(xD_x\) has b-symbol \(\xi\).

\begin{proposition}[Anchor bridge]\label{prop:anchor}
\begin{enumerate}
\item[(i)] The anchor \(a:{}^bTX\to TX\) becomes an isomorphism of Lie
algebroids after adjoining Laurent coefficients:
\[
 \cO(X)\otimes_{C^\infty(X)}\Gamma({}^bTX)
 \xrightarrow{\;a\;}
 \cO(X)\otimes_{C^\infty(X)}\Gamma(TX),
\]
an isomorphism of \(\cO(X)\)-modules preserving brackets.
\item[(ii)] There is a (noncanonical) isomorphism of smooth vector bundles
\({}^bTX\cong TX\). Consequently \(Y=\Sbstar\cong S^*X\) as sphere
bundles over \(X\), by a diffeomorphism carrying \(Z\) onto
\(S^*X|_{\partial X}\), and
\[
 H_L^*(\Sbstar\times S^1_\sigma)
 \cong H_L^*(S^*X\times S^1_\sigma).
\]
\end{enumerate}
The isomorphism in (ii) is not induced by the anchor: the dual anchor
\(T^*X\to\Tbstar\) does not identify the homogeneous symbol spaces.
\end{proposition}

\begin{proof}
(i) Locally the anchor sends \(x\partial_x\) to the ordinary vector field
\(x\partial_x\) and fixes \(\partial_{y_j}\). Its localized inverse uses
the Laurent coefficient \(x^{-1}\):
\(\partial_x\mapsto x^{-1}(x\partial_x)\). The anchor of a Lie algebroid
preserves brackets, hence so does its Laurent localization.

(ii) In a collar \([0,\varepsilon)_x\times\partial X\) choose a smooth
function \(h(x)>0\) with \(h=1\) near \(x=0\) and \(h(x)=x\) for
\(x\geq\varepsilon/2\). Define \(\Phi:{}^bTX\to TX\) by
\(\Phi(x\partial_x)=h(x)\partial_x\), \(\Phi(\partial_{y_j})=\partial_{y_j}\)
in the collar, and \(\Phi=a\) outside it. Then \(\Phi\) is smooth and
fiberwise invertible, since \(h>0\) and \(a\) is invertible over
\(X^\circ\). Its inverse transpose induces a diffeomorphism of cosphere
bundles \(\Sbstar\to S^*X\) over the identity of \(X\), hence carries the
restriction to \(\partial X\) onto the restriction to \(\partial X\).
A boundary defining function of \(S^*X\) pulls back to one of \(Y\), so
the diffeomorphism identifies the Laurent de Rham complexes.

Why not more: in the coordinates above the dual anchor is
\(\xi\,dx+\eta\cdot dy\mapsto (x\xi)\,dx/x+\eta\cdot dy\), so it kills the
conormal direction at \(\partial X\) and does not map \(T^*X\setminus0\)
into \(\Tbstar\setminus0\). For instance, the homogeneous function
\(\xi_b/|\zeta_b|\) pulls back to \(x\xi/\sqrt{x^2\xi^2+|\eta|^2}\), which
is not Laurent-smooth near \(x=\eta=0\). So no chain-level isomorphism of
homogeneous Laurent symbol complexes is induced by the anchor.
\end{proof}

\section{The associated graded algebra}\label{sec:graded}

Filter \(\cA_{b,L}\) by operator order:
\[
 F_m\cA_{b,L}
 =\operatorname{image}\left(
 \bigcup_{\ell\geq0}x^{-\ell}\Psi_b^m(X;E)
 \right),
 \qquad m\in\Z.
\]
The filtration is exhaustive, separated in the complete-symbol sense, and
multiplicative. Principal symbols give
\[
 \gr_m\cA_{b,L}
 \cong \cO_L(\Tbstar\setminus0;\End E)_m,
\]
the Laurent functions that are homogeneous of degree \(m\) in the fibers.
In a b-Darboux chart, a choice of local Weyl quantization gives the
expansion
\begin{equation}\label{eq:symbol-product}
 a\# c\sim ac+\frac{i}{2}\{a,c\}_b
 +\text{terms two orders lower},
\end{equation}
where changing quantization changes the individual lower-order symmetric
terms. Without a chosen global symplectic connection, the full Weyl
formula is not coordinate invariant. What is invariant, and all that is
used below, is the commutator symbol
\[
 \sigma_{m+m'-1}([A,C])=i\{a,c\}_b.
\]
This one-order drop is the origin of the first nonzero differential.

\begin{lemma}[Morita reduction]
The coefficient bundle \(E\) does not change the Hochschild homology.
\end{lemma}

\begin{proof}
The global module of complete symbols with coefficients in \(E\) is a
finitely generated projective equivalence bimodule between the scalar
algebra and its \(\operatorname{End}E\)-coefficient algebra. The associated
matrix-trace and corner maps are continuous and filtration preserving, and
the standard matrix-unit homotopy proves Morita invariance. This is a
global Morita equivalence; no partition of unity is used to glue local
chain homotopies. The topologically filtered version, including graded
bundles, is treated in
\cite{BNsymbols,BNfamilies}.
\end{proof}

We therefore suppress \(E\). The Hochschild--Kostant--Rosenberg map on the
associated graded algebra is
\[
 \chi(a_0\otimes\cdots\otimes a_j)
 =\frac1{j!}a_0\,da_1\wedge\cdots\wedge da_j.
\]

For \(w\in\Z\) let \(\Omega_L^j(\Tbstar\setminus0)_{(w)}\) be the Laurent
\(j\)-forms of fiber weight \(w\), that is, with
\(\mathcal L_{\mathcal E}\alpha=w\alpha\) for the radial (Euler) vector
field \(\mathcal E\), and put
\(\Omega_L^j(\Tbstar\setminus0)_{\mathrm{hom}}
=\bigoplus_{w}\Omega_L^j(\Tbstar\setminus0)_{(w)}\). If \(a_i\) is
homogeneous of degree \(m_i\), then \(a_0\,da_1\wedge\cdots\wedge da_j\)
has weight \(m_0+\cdots+m_j\), since \(d\) commutes with dilation.

\begin{proposition}[The \(E^1\)-page]\label{prop:E1}
The HKR map identifies the first page of the order spectral sequence with
homogeneous Laurent differential forms on \(\Tbstar\setminus0\), the fiber
weight being the total symbol order:
\[
 E^1_{p,q}\cong\Omega_L^{p+q}(\Tbstar\setminus0)_{(p)} .
\]
\end{proposition}

\begin{proof}
On each b-cotangent chart the associated graded product is ordinary
multiplication of scalar homogeneous symbols. Antisymmetrization gives the
inverse to \(\chi\) on Hochschild homology:
\[
 a_0\,da_1\wedge\cdots\wedge da_j
 \longmapsto
 \sum_{\tau\in S_j}\operatorname{sgn}(\tau)
 a_0\otimes a_{\tau(1)}\otimes\cdots\otimes a_{\tau(j)}.
\]
The displayed antisymmetrization is not a literal chain-level inverse;
the two composites become inverse after the Koszul homotopy. For the
topologically filtered Laurent groupoid algebra, continuity, completion,
and globalization are the HKR proposition used by Benameur--Nistor
\cite{BNsymbols,BNfamilies}. It may be proved through the sheaf of local
Koszul resolutions and its fine resolution, but a bare partition of unity
does not patch chain homotopies. Fiber dilation commutes with the published
quasi-isomorphism, so it retains homogeneity. A chain of total order \(p\)
in Hochschild degree \(k=p+q\) is sent to a \(k\)-form of weight \(p\),
which is the displayed indexing.
\end{proof}

\section{The first differential is Poisson homology}\label{sec:poisson}

For classical b-symbols \(a\) and \(c\), symbolic composition gives
\[
 \sigma_{m+m'-1}([A,C])
 =i\{a,c\}_b.
\]
Consequently the first differential after HKR is, up to the harmless factor
\(i\), the Poisson differential
\[
 \delta_b=[\iota_{\pi_b},d]
 =\iota_{\pi_b}d-d\iota_{\pi_b}.
\]
Explicitly,
\begin{align}\label{eq:poisson-boundary}
 \delta_b(a_0\,da_1\wedge\cdots\wedge da_j)
 ={}&\sum_{r=1}^j(-1)^{r+1}\{a_0,a_r\}_b
       \,da_1\wedge\cdots\widehat{da_r}\cdots\wedge da_j\notag\\
 &+\sum_{1\leq r<s\leq j}(-1)^{r+s}a_0\,d\{a_r,a_s\}_b
       \wedge da_1\wedge\cdots
       \widehat{da_r}\cdots\widehat{da_s}\cdots\wedge da_j.
\end{align}
To see this differential directly, insert the order-one-lower part of
\eqref{eq:symbol-product} into each multiplication in the Hochschild
boundary. The symmetric terms cancel after HKR antisymmetrization, while the
commutator terms combine to \eqref{eq:poisson-boundary}.

Let \(N=2n\) be the dimension of \(\Tbstar\), put
\(\mu_b=\omega_b^n/n!\), and let
\(\pi_b^\sharp:\Omega^j\to\Lambda^jT\) be contraction of every covector
with \(\pi_b\). Symplectic duality is the map
\[
 *_b:\Omega_L^j(\Tbstar\setminus0)
 \longrightarrow
 \Omega_L^{N-j}(\Tbstar\setminus0),
 \qquad *_b\alpha=\iota_{\pi_b^\sharp\alpha}\mu_b,
\]
and \(*_b^2=1\) up to the standard degree sign. Applying the two sides to a
monomial in the Darboux b-frame
\[
 \frac{dx}{x},dy_1,\ldots,dy_{n-1},d\xi,d\eta_1,\ldots,d\eta_{n-1}
\]
gives, with the displayed convention for \(\delta_b\),
\begin{equation}\label{eq:symplectic-duality}
 *_b\delta_b=(-1)^{j+1}d*_b
 \quad\text{on }\Omega_L^j.
\end{equation}
The calculation remains valid at \(x=0\): the displayed frame is smooth as a
b-frame, and both operators preserve finite Laurent pole order. Therefore a
\(\delta_b\)-cycle of degree \(j\) is carried to a closed Laurent form of
degree \(2n-j\), while boundaries are carried to exact forms.

The operator \(*_b\) also shifts fiber weight. Since \(\alpha_b\) and hence
\(\omega_b\) have weight \(1\), \(\mu_b\) has weight \(n\) and \(\pi_b\) has
weight \(-1\). Contracting a \(j\)-form of weight \(w\) with \(\pi_b\) in
each slot gives a \(j\)-vector of weight \(w-j\), and contracting this
into \(\mu_b\) gives
\begin{equation}\label{eq:weight-shift}
 *_b:\Omega_L^j(\Tbstar\setminus0)_{(w)}
 \xrightarrow{\;\cong\;}
 \Omega_L^{2n-j}(\Tbstar\setminus0)_{(w+n-j)} .
\end{equation}
In particular \(\delta_b\), which lowers both \(j\) and \(w\) by one,
preserves \(w-j\), consistent with \(d\) preserving weight.

\begin{corollary}\label{cor:poisson}
On the homogeneous complex of the \(E^1\)-page, Poisson homology is
Laurent de Rham cohomology with complementary degree and shifted weight:
\[
 H_j^{\pi_b,L}(\Tbstar\setminus0)_{(w)}
 \cong H_L^{2n-j}(\Tbstar\setminus0)_{(w+n-j)},
 \qquad
 H_j^{\pi_b,L}(\Tbstar\setminus0)_{\mathrm{hom}}
 \cong H_L^{2n-j}(\Tbstar\setminus0)_{\mathrm{hom}} .
\]
\end{corollary}

\section{Homogeneity and the symbol-order circle}\label{sec:homogeneity}

Choose a positive fiber norm \(r\) on \(\Tbstar\setminus0\). Then
\[
 \Tbstar\setminus0\cong Y\times\mathbb R_+,
 \qquad (q,r)\longmapsto rq.
\]
By Proposition~\ref{prop:E1} and \eqref{eq:weight-shift}, the column
\(E^1_{p,\bullet}\) in Hochschild degree \(k\) corresponds under \(*_b\) to
\((2n-k)\)-forms of weight \(p+n-k\). This bookkeeping matters:
\(\delta_b\) lowers fiber homogeneity by one, whereas \(*_b\) changes both
form degree and homogeneity.

The zero-weight reduction is elementary. The Euler vector field
\(\mathcal E=r\partial_r\) is dilation invariant, so \(d\) and
\(\iota_{\mathcal E}\) preserve weight. If a form \(\alpha\) has nonzero
weight \(w\), Cartan's formula gives
\[
 (d\iota_{\mathcal E}+\iota_{\mathcal E}d)\alpha
 =\mathcal L_{\mathcal E}\alpha=w\alpha.
\]
Thus \(w^{-1}\iota_{\mathcal E}\) is a contracting homotopy of the
weight-\(w\) subcomplex, and only weight \(0\) contributes to
\(H_L^*(\Tbstar\setminus0)_{\mathrm{hom}}\). By \eqref{eq:weight-shift}, in
Hochschild degree \(k\) the only surviving column is \(p=k-n\), that is,
\(q=n\) (Section~\ref{sec:collapse}).

At weight zero, restriction to \(r=1\) gives a unique splitting
\[
 \alpha(r,q)=\alpha_0(q)+\frac{dr}{r}\wedge\alpha_1(q).
\]
Since \(d(dr/r)=0\), its differential is
\[
 d\alpha=d_Y\alpha_0-\frac{dr}{r}\wedge d_Y\alpha_1.
\]
Thus the weight-zero complex is the direct sum of the de Rham complex of
\(Y\) and a copy shifted by one degree. The complete-symbol order is an
integer grading with no preferred logarithm; cohomologically this two-term
complex is canonically written as the de Rham complex of an auxiliary circle:
\[
 \alpha_0+\frac{dr}{r}\wedge\alpha_1
 \longleftrightarrow
 \alpha_0+d\theta_\sigma\wedge\alpha_1.
\]
The formal order variable \(dr/r\) is represented cohomologically by the
generator \(d\theta_\sigma\) of a circle. Hence
\begin{equation}\label{eq:order-circle}
 H_L^j(\Tbstar\setminus0)_{\mathrm{hom}=0}
 \cong H_L^j(Y\times S^1_\sigma).
\end{equation}
This is the source of the extra \(S^1\) in the answer; it is not a geometric
circle in the fibers of \(\Tbstar\), and no choice of fiber norm affects its
cohomology class.

\section{Laurent cohomology and the boundary face}\label{sec:laurent}

Let \(V\) be a compact manifold with one embedded boundary hypersurface and
defining function \(x\). The Laurent de Rham complex is
\[
 \Omega_L^j(V)=\bigcup_{\ell\geq0}x^{-\ell}\Omega^j(V),
 \qquad d:\Omega_L^j(V)\to\Omega_L^{j+1}(V).
\]

\begin{proposition}[Face decomposition]\label{prop:face}
There is a split short exact sequence
\[
 0\to H^j(V)\to H_L^j(V)\xrightarrow{\operatorname{Res}}
 H^{j-1}(\partial V)\to0,
\]
hence an isomorphism
\(H_L^j(V)\cong H^j(V)\oplus H^{j-1}(\partial V)\).
A splitting sends \([\beta]\) to the class of
\((-1)^{j-1}\widetilde\beta\wedge d(\chi\log x)\), where \(\beta\) is a
closed \((j-1)\)-form on \(\partial V\), \(\widetilde\beta\) is a collar
extension, and \(\chi\) is a cutoff supported in the collar and equal to
one near the boundary.
\end{proposition}

\begin{proof}
Work in a collar and write a Laurent form uniquely as
\[
 \alpha(x)+\frac{dx}{x}\wedge\beta(x),
\]
where the tangential forms \(\alpha(x),\beta(x)\) are Laurent families with
a finite principal part and a smooth Taylor tail. Filter the collar complex by
pole order and then by normal jet. On its associated graded, a term has a
definite normal weight \(q\). For \(q\ne0\), Cartan's identity gives the
contracting homotopy
\[
 h_q=q^{-1}\iota_{x\partial_x},
 \qquad dh_q+h_qd=1.
\]
Induction over the finite negative weights removes the principal part;
the remaining positive Taylor tail is smooth and belongs to the ordinary
de Rham complex. This jet-filtration argument, rather than freezing all
coefficients at \(x=0\), is the Laurent Poincar\'e lemma. To globalize,
multiply the collar homotopies by a cutoff equal to one near \(\partial V\);
the error terms contain a derivative of the cutoff, are supported away from
\(\partial V\), and are therefore smooth. The only new normal class is the
weight-zero generator \(dx/x\).

In particular, after writing
\[
 \alpha=\alpha_0+\frac{dx}{x}\wedge\beta_0
          +\text{terms of nonzero normal weight},
\]
the residue is
\[
 \operatorname{Res}_{\partial V}:\Omega_L^j(V)\longrightarrow
 \Omega^{j-1}(\partial V),
 \qquad
 \operatorname{Res}_{\partial V}(\alpha)=\beta_0|_{\partial V}.
\]
A direct calculation in the displayed order gives
\[
 \operatorname{Res}(d\alpha)=-d_{\partial V}\operatorname{Res}(\alpha).
\]
Thus residue is a chain map to the degree-shifted boundary complex. The
Laurent Poincar\'e lemma identifies its homotopy kernel with
\(\Omega^*(V)\), yielding the quasi-isomorphism
\[
 \Omega_L^*(V)\simeq
 \Omega^*(V)\oplus\Omega^{*-1}(\partial V).
\]

If \(\beta\) is closed of degree \(j-1\), let
\(p:[0,\varepsilon)_x\times\partial V\to\partial V\) be the collar
projection. Choose \(\chi\) with support in this collar and choose
\(\widetilde\beta=p^*\beta\) on an open set containing
\(\operatorname{supp}d(\chi\log x)\). Extend it arbitrarily beyond that
set. Then \(d\widetilde\beta=0\) everywhere that
\(d(\chi\log x)\) is nonzero, so the representative below is closed. This
collar construction gives the
cohomological splitting
\[
 \beta\longmapsto
 (-1)^{j-1}\widetilde\beta\wedge d(\chi\log x).
\]
Its residue is \(\beta\), because
\(\widetilde\beta\wedge dx/x=(-1)^{j-1}(dx/x)\wedge\widetilde\beta\) near
\(\partial V\). If \(x'=e^fx\), then
\(d\log x'=d\log x+df\); changing the defining function, collar, or cutoff
changes this splitting by an ordinary/exact correction. Hence the direct
sum is natural as a filtered extension but its explicit vector-space
splitting is not canonical.
\end{proof}

\begin{discussion}
\textbf{Where do higher poles go?}
A pole such as \(x^{-q-1}dx\), \(q>0\), looks more singular than
\(dx/x\), but it is \(-q^{-1}d(x^{-q})\). Higher Laurent coefficients occur
in the complex and are essential for closure under composition; they add no
new normal cohomology. Only the logarithmic coefficient survives.
\end{discussion}

Apply Proposition~\ref{prop:face} to \(V=Y\times S^1_\sigma\), whose boundary
is \(Z\times S^1_\sigma\). Then use
\[
 H^r(W\times S^1)
 \cong H^r(W)\oplus H^{r-1}(W).
\]
For \(r=2n-k\) this gives exactly the four summands in
\eqref{eq:expanded}.

\begin{figure}[htbp]

For a manifold with corners, the same local argument has one generator
\(d\log x_H\) for every boundary hypersurface \(H\) through a point. If
\(\mathcal F_j(V)\) denotes the codimension-\(j\) faces, then
\begin{equation}\label{eq:corners}
 H_L^r(V)
 \cong\bigoplus_{j\geq0}
       \bigoplus_{F\in\mathcal F_j(V)}
       H^{r-j}(F;o(N_F)).
\end{equation}
Here \(o(N_F)\) is the orientation line of the conormal directions to
\(F\). If the hypersurfaces have been globally cooriented and ordered,
these lines are trivialized and the untwisted formula results.
Indeed, apply the one-face residue successively. Residues for distinct
hypersurfaces anticommute with the graded sign, so the summand attached to
\(F=H_1\cap\cdots\cap H_j\) is represented by
\[
 \widetilde\gamma\wedge d\log x_{H_1}\wedge\cdots\wedge d\log x_{H_j},
 \qquad [\gamma]\in H^{r-j}(F).
\]
The product of the nonzero-weight contractions gives an acyclic complement.
Changing the order of the hypersurfaces changes the orientation sign of
the exterior logarithmic factors; the local system \(o(N_F)\) records
exactly this ambiguity. Iterated collar extensions provide a noncanonical
splitting. This is only a sketch of \eqref{eq:corners}: the Laurent
Poincar\'e lemma above, applied one hypersurface at a time, reduces
\(\Omega_L^*(V)\) to the logarithmic (b-) de Rham complex, and
\eqref{eq:corners} is then the standard b-de Rham theorem for manifolds
with corners (for one boundary hypersurface see \cite{MelroseAPS}; the
corner case follows by iterating it over the boundary faces). The corner
formula is not used elsewhere in these notes.

\section{Collapse and assembly of the computation}\label{sec:collapse}

Combining HKR, the Poisson differential, symplectic duality, and homogeneity
gives the second page. The new point of this section is that the
degeneration of the spectral sequence is then a matter of bidegrees; the
only global input that remains is convergence.

\begin{proposition}[The \(E^2\)-page and degeneration]\label{prop:E2}
For the order spectral sequence of
\(C_\bullet^{\mathrm{cont}}(\cA_{b,L})\),
\[
 E^2_{p,q}\cong
 \begin{cases}
 H_L^{2n-p-q}(Y\times S^1_\sigma), & q=n,\\
 0, & q\neq n.
 \end{cases}
\]
Consequently \(d^r=0\) for all \(r\geq2\) and \(E^2=E^r\) for every
\(r\geq2\).
\end{proposition}

\begin{proof}
By Proposition~\ref{prop:E1}, \(E^1_{p,q}\) is the space of Laurent
\(k\)-forms of weight \(p\), \(k=p+q\), and \(d^1=i\delta_b\) by
Section~\ref{sec:poisson}. By \eqref{eq:symplectic-duality} and
\eqref{eq:weight-shift}, \(*_b\) identifies \(E^2_{p,q}\) with the
cohomology of the Laurent de Rham complex in degree \(2n-k\) and weight
\(p+n-k=n-q\). By Section~\ref{sec:homogeneity}, nonzero weights are
acyclic, so \(E^2_{p,q}=0\) unless \(q=n\); for \(q=n\) the weight is zero
and \eqref{eq:order-circle} gives \(H_L^{2n-k}(Y\times S^1_\sigma)\).
The differential \(d^r\) maps \(E^r_{p,q}\) to \(E^r_{p-r,q+r-1}\). For
\(r\geq2\) the rows \(q\) and \(q+r-1\) are distinct, so at least one of
source and target vanishes, and \(d^r=0\).
\end{proof}

The degeneration is therefore a consequence of degree counting once the
weight bookkeeping \eqref{eq:weight-shift} is in place. What is not
automatic, and what we quote, is that the spectral sequence computes the
continuous Hochschild homology at all.

\begin{theorem}[Benameur--Nistor convergence theorem; quoted]
\label{thm:BN-conv}
For the topologically filtered Laurent complete-symbol algebra of a
differential groupoid satisfying the hypotheses of
\cite{BNsymbols,BNfamilies}, the continuous HKR map identifies the
\(E^1\)-page as in Proposition~\ref{prop:E1}, and the order spectral
sequence of the continuous Hochschild complex converges to
\(\HH_*^{\mathrm{cont}}\): the induced filtration
\(\mathcal F_p\HH_k\) (the image of the homology of \(\mathcal F_pC_k\)) is
exhaustive and Hausdorff, and
\(E^\infty_{p,q}\cong\mathcal F_p\HH_{p+q}/\mathcal F_{p-1}\HH_{p+q}\).
\end{theorem}

Benameur and Nistor also prove degeneration at \(E^2\); in the present
setting this is Proposition~\ref{prop:E2} and need not be quoted.

\begin{proposition}[Source dictionary]\label{prop:hypotheses}
The algebra \(\cA_{b,L}(X;E)\) is the one-face b-groupoid specialization of
the algebras treated by Benameur--Nistor.
\end{proposition}

\begin{proof}
The dictionary has six entries.

\emph{Groupoid.} The differential groupoid is \(\Gb\), with unit space
\(X\) and Lie algebroid \({}^bTX\).

\emph{Laurent multipliers.} The coefficient ring is precisely
\(\cO(X)=C^\infty(X)[x_H^{-1}]\).

\emph{Algebra and topology.} The algebra is
\(\cO(X)(\Psi^\infty(\Gb;E)/\Psi^{-\infty}(\Gb;E))\), equipped with the
topologically filtered completion of \cite{BNfamilies}; its continuous
Hochschild complex is \eqref{eq:BN-chains}.

\emph{Anchor condition.} The relevant property of the Lie algebroid is
that its anchor becomes an isomorphism of Lie algebroids once the boundary
defining functions are inverted. For \({}^bTX\) this is
Proposition~\ref{prop:anchor}(i). We do not use, and do not claim, any
chain-level identification of b-symbols with ordinary symbols through the
anchor; see the end of the proof of Proposition~\ref{prop:anchor}.

\emph{Local models and coefficients.} Locally near \(\partial X\) the
groupoid \(\Gb\) is the product of the pair groupoid of the boundary chart
with the b-groupoid of \([0,\varepsilon)\). This product structure is the
kind of local model their hypotheses require; we have not compared it with
their precise formulation beyond this. The global Morita lemma removes the
coefficient bundle \(E\).

\emph{Spectral sequence.} Their filtered HKR theorem identifies \(E^1\)
(Proposition~\ref{prop:E1}) and their convergence theorem identifies
\(E^\infty\) with the graded continuous Hochschild homology
(Theorem~\ref{thm:BN-conv}). The \(E^2\)-page and the degeneration are
computed independently here (Proposition~\ref{prop:E2}).
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
By Proposition~\ref{prop:E2}, \(E^\infty_{p,q}=E^2_{p,q}\) vanishes unless
\(q=n\). Fix \(k\). By Theorem~\ref{thm:BN-conv},
\(\mathcal F_p\HH_k/\mathcal F_{p-1}\HH_k=0\) for \(p\neq k-n\). Hence
\(\mathcal F_p\HH_k=\mathcal F_{k-n}\HH_k\) for all \(p\geq k-n\), and
exhaustiveness gives \(\mathcal F_{k-n}\HH_k=\HH_k\). Likewise
\(\mathcal F_p\HH_k=\mathcal F_{k-n-1}\HH_k\) for all \(p\leq k-n-1\), and
the Hausdorff property gives \(\mathcal F_{k-n-1}\HH_k=0\). Therefore
\[
 \HH_k^{\mathrm{cont}}(\cA_{b,L})
 =\mathcal F_{k-n}\HH_k
 \cong E^\infty_{k-n,n}=E^2_{k-n,n}
 \cong H_L^{2n-k}(Y\times S^1_\sigma),
\]
with no choices involved: every map in this chain is canonical, and the
identification \eqref{eq:order-circle} does not depend on the fiber norm.
A Borel quantization, which splits the symbol filtration on the algebra
only noncanonically, is not needed. Finally,
Proposition~\ref{prop:face} and K\"unneth for \(S^1_\sigma\) give the four
summands in \eqref{eq:expanded}; here the choice of collar splitting
enters.
\end{proof}

\begin{discussion}
\textbf{Why is convergence not automatic?}
For a merely algebraic union of symbol orders, an infinite correction can
escape the complex and an \(E^\infty\)-class need not lift to a chain.
Borel summation and the strict Laurent inductive limit are ingredients in
the topologically filtered axioms: the former realizes asymptotic symbols
and the latter keeps every individual chain at finite pole order.  They do
not by themselves prove convergence.  Convergence to Hochschild homology is
the quoted Theorem~\ref{thm:BN-conv}. The bidegree argument of
Proposition~\ref{prop:E2} shows that no higher differentials occur, but by
itself says nothing about whether \(E^\infty\) is the associated graded of
\(\HH_*^{\mathrm{cont}}\).
\end{discussion}

\section{A \v Cech--de Rham route (conditional)}\label{sec:cech}

This section sketches an alternative organization of the proof, which
replaces the global HKR assertion by a local calculation followed by
descent. It is \emph{not} self-contained: it depends on a quoted
localization theorem for continuous Hochschild chains
(Lemma~\ref{lem:cech-descent}), and it still uses the Benameur--Nistor
convergence theorem; \v Cech descent alone computes the \(E^2\)-page, not
the abutment.

Choose a finite good cover \(\mathcal U=\{U_i\}\) of
\(Y=\Sbstar\).  Near \(Z=\partial Y\), take product sets
\([0,\varepsilon)_x\times V_i\), and require every nonempty intersection to
be contractible and collar adapted.  Write
\(\widetilde U_I=U_I\times\mathbb R_+\) for the corresponding conic open set
in \(\Tbstar\setminus0\).

Complete-symbol composition is local in \(\Tbstar\setminus0\), so for a
conic open set \(\widetilde U\) there is an algebra
\(\cA_{b,L}(\widetilde U)\) of Laurent complete symbols on
\(\widetilde U\), with restriction homomorphisms for
\(\widetilde U'\subset\widetilde U\). Let \(\mathcal C_q(U)\) be its
Benameur--Nistor completed degree-\(q\) chain space: one forms the diagonal
pieces \(F^N_{N,N}\) over \(\widetilde U\) first, takes their completed
projective tensor powers, and only then takes the strict inductive limit in
\(N\), as in \eqref{eq:BN-chains}. Restriction of every tensor factor makes
\(U\mapsto\mathcal C_q(U)\) a presheaf of filtered complexes.

Form the double complex
\begin{equation}\label{eq:cech-double}
 K^{p,q}=\check C^p(\mathcal U,\mathcal C_q),
 \qquad
 D=\delta_{\check C}+(-1)^p b.
\end{equation}
The total degree is \(q-p\), since \(b\) lowers Hochschild degree and the
\v Cech differential raises \(p\).

\begin{lemma}[Localization of continuous Hochschild chains; quoted
input]\label{lem:cech-descent}
The augmentation
\[
 C_\bullet^{\mathrm{cont}}(\cA_{b,L})
 \longrightarrow \operatorname{Tot}K^{\bullet,\bullet}
\]
is a filtration-preserving quasi-isomorphism, and it remains a
quasi-isomorphism after passing to any fixed order quotient.
\end{lemma}

\begin{remark}[Status of Lemma~\ref{lem:cech-descent}]
The augmentation is \emph{not} a quasi-isomorphism row by row: a chain
\(a_0\otimes a_1\) whose factors are supported in disjoint open sets
restricts to zero on every \(\widetilde U_i\), although it is a nonzero
chain. Such chains are, however, far from the diagonal, and the needed
input is that continuous Hochschild chains localize to an arbitrarily small
neighbourhood of the diagonal: the subcomplex of \(\mathcal U\)-small
chains, all of whose factors are supported in a common \(\widetilde U_i\),
is a filtered deformation retract of the full complex, and on
\(\mathcal U\)-small chains the \v Cech resolution is exact by a partition
of unity. For the symbol algebra of a closed manifold, localization of
this kind underlies the computation of Brylinski--Getzler \cite{BG}; for
fine algebras it is
Teleman's localization of the Hochschild complex \cite{Teleman}. We assume
the corresponding statement for the topologically filtered Laurent algebra,
compatibly with the order filtration; we do not prove it here.
\end{remark}

Filter \eqref{eq:cech-double} by total symbol order. The associated graded
of the total complex is the total complex of
\(\check C^p(\mathcal U,\gr\mathcal C_q)\), on which multiplication is
commutative. Compute its homology by first taking Hochschild homology
in each \v Cech degree. The local HKR map and the local Koszul homotopy,
which commute with restriction, give
\begin{equation}\label{eq:cech-E1}
 \check C^p\bigl(\mathcal U,
 \Omega_L^q(\Tbstar\setminus0)_{\mathrm{hom}}\bigr).
\end{equation}
This is an intermediate page, not yet \(E^1\) of the total complex: the
\v Cech differential remains. Each weight-\(w\) piece of the sheaf of
Laurent forms with a fixed pole bound is a module over \(C^\infty(Y)\),
hence fine, and the union over pole bounds is still \v Cech acyclic because
multiplication by a partition of unity on \(Y\) preserves pole bounds and
weight. Since the cover is finite, the \v Cech direction is bounded and
\eqref{eq:cech-E1} has \v Cech cohomology only in degree \(p=0\), equal to
the global forms. Hence
\[
 E^1(\operatorname{Tot}K)\cong\Omega_L^*(\Tbstar\setminus0)_{\mathrm{hom}},
\]
the same page as in Proposition~\ref{prop:E1}. The first order-lowering part
of the symbol product gives \(d^1=i\delta_b\), exactly as in
\eqref{eq:poisson-boundary}.

For a closed manifold these are exactly the local filtered steps in the
Brylinski--Getzler calculation \cite{BG}: complete symbols pass to
homogeneous functions, HKR gives forms, and the first quantum correction is
the Poisson boundary.  The present descent argument adds the Laurent
normal-weight contraction, the residue complex at \(Z\), and the strict
topologically filtered completion required at the boundary.

\begin{proposition}[\v Cech computation of the second page]
Granting Lemma~\ref{lem:cech-descent}, the total filtered complex
\eqref{eq:cech-double} has
\[
 E^2_{p,q}\cong H_L^{2n-p-q}(Y\times S^1_\sigma;\C)\ \text{for }q=n,
 \qquad E^2_{p,q}=0\ \text{otherwise}.
\]
This identification is independent of the good cover and the partition of
unity.
\end{proposition}

\begin{proof}
Symplectic duality \eqref{eq:symplectic-duality} converts the Poisson
differential to the Laurent de Rham differential with complementary degree,
with the weight shift \eqref{eq:weight-shift}. The Euler homotopy removes
nonzero fiber weights.  At weight zero a form is
\(\alpha_0+(dr/r)\wedge\alpha_1\), so
\eqref{eq:order-circle} identifies the remaining complex with Laurent forms
on \(Y\times S^1_\sigma\); the row \(q=n\) is singled out as in
Proposition~\ref{prop:E2}.  Refining the cover gives a common refinement, and
the two refinement maps are homotopic on the augmented \v Cech resolution.
This proves independence of all auxiliary choices.
\end{proof}

The boundary term can also be read directly in the descent complex.  One
must not write the generally false short exact sequence
\(0\to\Omega^*(Y)\to\Omega_L^*(Y)\to\Omega^{*-1}(Z)\to0\): a Laurent form
with zero residue can still have higher poles.  Instead let
\(\mathcal K^*=\ker(\operatorname{Res}_Z)\).  The nonzero normal-weight
contraction in the proof of Proposition~\ref{prop:face} gives a
quasi-isomorphism
\[
 \Omega^*(Y)\longrightarrow\mathcal K^*.
\]
The exact sequence
\[
0\longrightarrow\mathcal K^*
 \longrightarrow\Omega_L^*(Y)
 \xrightarrow{\operatorname{Res}_Z}
 \bigl(\Omega^{*-1}(Z),-d_Z\bigr)\longrightarrow0
\]
therefore yields the derived decomposition
\begin{equation}\label{eq:cech-residue-derived}
 R\Gamma(Y;\Omega_L^*)
 \simeq R\Gamma(Y;\Omega^*)
 \oplus R\Gamma(Z;\Omega^{*-1}),
\end{equation}
after choosing a collar splitting.  This is precisely the interior/boundary
decomposition used in \eqref{eq:expanded}; it also shows why the splitting,
unlike the residue filtration, is not canonical.

Finally, Lemma~\ref{lem:cech-descent} identifies the abutment of the order
filtration on \eqref{eq:cech-double} with
\(\HH_*^{\mathrm{cont}}(\cA_{b,L})\), compatibly with filtrations, so
Theorem~\ref{thm:BN-conv} applies to it; the proposition identifies its
second page, and the bidegree argument of Proposition~\ref{prop:E2} gives
\(E^2=E^\infty\). This recovers Theorem~\ref{thm:main}, conditionally on
the localization input. Descent replaces the globalization part of the
HKR step, but it does not manufacture convergence.

\section{A Rees--Fedosov reorganization}\label{sec:rees}

This section makes the deformation behind symbol composition explicit. It
is a heuristic reorganization of the computation, not an independent
proof: nothing in it is used in the proof of Theorem~\ref{thm:main}, the
\(E^2\)-page already follows from
Sections~\ref{sec:graded}--\ref{sec:laurent}, and the formal Hochschild
homology computed below lacks the order circle. It has two logically
separate parts: a formal Lie-algebroid formality statement and a
finite-order Rees comparison with classical symbols, which is elementary.

This is also the deformation-theoretic refinement of the
Brylinski--Getzler filtration: their principal-symbol and Poisson
differentials are the first two coefficients of the Rees star product below
\cite{BG}.

\subsection{The Rees deformation}

The algebraic Rees algebra of the order filtration is
\[
 \mathcal R_F(\cA_{b,L})
 =\bigoplus_{m\in\mathbb Z}t^mF_m\cA_{b,L}
 \subset \cA_{b,L}[t,t^{-1}].
\]
It records both specialization to the associated graded and the return to
complete symbols at \(t=1\).  Since
\([F_m,F_{m'}]\subset F_{m+m'-1}\), its commutator is divisible by the
Rees parameter in an associated-graded chart.  Equivalently, for a symbol
of fixed upper order
\(a\sim a_m+a_{m-1}+a_{m-2}+\cdots\), factor off the bookkeeping degree
\(t^m\) and encode the descending expansion by
\begin{equation}\label{eq:formal-Rees}
 a_m+\hbar a_{m-1}+\hbar^2a_{m-2}+\cdots .
\end{equation}
The power \(t^m\) records the absolute order.  After it is factored off in a
fixed sector, we write the same Rees parameter as \(\hbar\); its successive
powers record loss of order.  Thus \eqref{eq:formal-Rees} is an
\(\hbar\)-adic chart on one upper-order sector, not a global
\(\mathbb C[[\hbar]]\)-module structure on the union of all orders.

On Laurent functions on \(M=\Tbstar\setminus0\), full symbol composition in
this chart becomes
\begin{equation}\label{eq:Rees-star}
 a\star c=\sum_{j\geq0}\hbar^j C_j(a,c),
 \qquad C_0(a,c)=ac,
\end{equation}
where \(C_j\) is b-bidifferential and lowers the unscaled symbol order by
\(j\).  Its antisymmetric first term is
\[
 C_1(a,c)-C_1(c,a)=i\{a,c\}_b,
\]
so \([a,c]_\star=i\hbar\{a,c\}_b+O(\hbar^2)\).

The space \(M\) has the symplectic Lie algebroid \({}^bTM\) with symplectic
form \(\omega_b\).  Choose a torsion-free b-symplectic connection and use
the homogeneous cotangent Fedosov construction, so that positive fiber
dilation acts equivariantly.  The Fedosov Weyl bundle has a flat differential
of the form
\[
 D_F=-\delta+\nabla+\frac{i}{\hbar}\operatorname{ad}(r_F),
 \qquad D_F^2=0.
\]
Flat sections, transported to Laurent functions by the Fedosov--Taylor map,
should give a star product equivalent to \eqref{eq:Rees-star}.
Fedosov quantization of symplectic Lie algebroids is developed by
Nest--Tsygan \cite{NestTsygan}; \({}^bTM\) with \(\omega_b\) is a symplectic
Lie algebroid, but we have not checked the homogeneous version or the
comparison with the b-symbol product in detail.  Equivalent star products induce isomorphic Hochschild
complexes, so the choice of connection does not affect the answer.

\subsection{Hochschild-chain formality}

Calaque--Dolgushev--Halbout prove formality theorems for Hochschild chains
in the Lie-algebroid setting \cite{CDH}; Dolgushev's smooth-manifold
theorem is the corresponding prototype \cite{Dolgushev}.  Their chain
complexes are built from Lie-algebroid jets and polydifferential data; we
have not shown that they agree with the Laurent chains used here, so the
following is a heuristic.  Twisting their module quasi-isomorphism by the
Maurer--Cartan element defining \eqref{eq:Rees-star} would give
\begin{equation}\label{eq:chain-formality}
 \left(C_\bullet(\mathcal O_L(M)((\hbar)),\star),b_\star\right)
 \simeq
 \left(\Omega_L^\bullet(M)((\hbar)),
 \mathcal L_{\pi_\hbar}\right),
\end{equation}
where \(\pi_\hbar=\hbar\pi_b+O(\hbar^2)\) is the formal Poisson tensor and
\(\mathcal L_{\pi_\hbar}=[\iota_{\pi_\hbar},d]\).

The formal tensor \(\pi_\hbar\) is symplectic over \(\C((\hbar))\); let
\(\omega_\hbar\) be its inverse.  The formal symplectic star operator
\(*_{\omega_\hbar}\) satisfies the same Darboux calculation as
\eqref{eq:symplectic-duality} and converts
\(\mathcal L_{\pi_\hbar}\) to \(d\), up to an invertible scalar and the
degree sign.  Therefore
\begin{equation}\label{eq:formal-HH}
 \HH_k^{\mathbb C((\hbar))}
 \bigl(\mathcal O_L(M)((\hbar)),\star\bigr)
 \cong H_L^{2n-k}(M;\C)((\hbar)).
\end{equation}
The formality maps are b-polydifferential.  Hence they preserve finite
Laurent pole order: locally \(x\partial_x(x^{-\ell})=-\ell x^{-\ell}\), and
tangential b-derivatives do not create a larger pole.

Equation~\eqref{eq:formal-HH} concerns the extended formal star algebra,
before restriction to classical homogeneous expansions and before Borel
realization.  It must not be read as the complete-symbol answer.  In
particular, \(H_L^*(M)\) itself has no order-circle factor: that factor
appears only after the homogeneous order complex is formed.

\begin{remark}[Where the order circle does, and does not, come from]
The complex in \eqref{eq:chain-formality} is relative to the base field
\(\C((\hbar))\).  Thus \(d\hbar=0\), and it is incorrect to insert an
additional class \(d\hbar/\hbar\) into \eqref{eq:formal-HH}.  The single
class giving \(S^1_\sigma\) is the radial zero-weight class \(dr/r\) in
Section~\ref{sec:homogeneity}.

One can instead compute absolute Hochschild homology over \(\C\) for the
Laurent extension of a Rees algebra.  Then a class \(d\log t\) can appear.
The Rees grading ties the exponent of \(t\) to fiber homogeneity, so on the
diagonal order complex \(d\log t\) and \(-d\log r\) describe the same order
direction.  After specialization at \(t=1\), only the radial class remains.
Either convention yields one order circle, never two.
\end{remark}

\subsection{From formal series to classical complete symbols}

Formal formality does not by itself compute the continuous homology of
\(\cA_{b,L}\).  What follows directly from the classical symbol calculus is
a comparison at every finite depth of the order filtration.

\begin{proposition}[Finite-order Rees comparison]
\label{prop:Borel-formality}
After fixing a homogeneous classical symbol map, for every
\(m\in\mathbb Z\) and \(N\geq0\), the first \(N+1\) symbol components give an
isomorphism, continuous at every fixed pole bound,
\begin{equation}\label{eq:finite-Rees}
 F_m\cA_{b,L}/F_{m-N-1}\cA_{b,L}
 \cong
 \bigoplus_{j=0}^{N}\mathcal O_L(M)_{m-j}.
\end{equation}
Under \eqref{eq:finite-Rees}, operator composition is the star product
\eqref{eq:Rees-star} modulo \(\hbar^{N+1}\).  These isomorphisms are
compatible with truncation, completed projective tensor powers on each
diagonal piece, and the Hochschild boundary.  Consequently the Rees and
classical order spectral sequences have the same \(E^1\)-page and the same
Poisson differential \(d^1\).
\end{proposition}

\begin{proof}
There are three points to check.

\emph{Algebraic realization.}
Taylor expansion in symbol order gives \eqref{eq:finite-Rees}; no infinite
summation is needed at finite \(N\).  The b-symbol composition formula shows
coefficient by coefficient that the two products agree modulo
\(\hbar^{N+1}\).  Its first antisymmetric correction is the b-Poisson
bracket, proving the assertion about \(d^1\).

\emph{Topology.}
For a fixed Laurent bound, both sides of \eqref{eq:finite-Rees} are finite
direct sums of nuclear Fr\'echet spaces.  Truncation and the coefficient maps
are continuous.  Taking completed projective tensor powers first and the
strict inductive limit over diagonal pieces afterward is exactly the order
of operations in \eqref{eq:BN-chains}; hence the comparison extends to the
finite quotients of the continuous Hochschild complex.

\emph{Inverse limit.}
Fix a pole bound \(\ell\). If \(x^{-\ell}A\in F_{m-N-1}\) with
\(A\in\Psi_b^m\), then the full symbol of \(A\) has order \(m-N-1\), so
\(x^{-\ell}\Psi_b^m\cap F_{m-N-1}=x^{-\ell}\Psi_b^{m-N-1}\). The b-symbol
Borel lemma realizes every compatible sequence of truncations at this pole
bound by a classical symbol, uniquely modulo smoothing; thus
\[
 \varprojlim_N x^{-\ell}\Psi_b^m/x^{-\ell}\Psi_b^{m-N-1}
 \cong x^{-\ell}\Psi_b^m/x^{-\ell}\Psi_b^{-\infty}.
\]
The statement is false without the pole bound: an element of
\(\varprojlim_NF_m/F_{m-N-1}\) may need pole orders \(\ell_N\to\infty\) as
\(N\to\infty\), and so need not come from \(F_m\). This asymptotic
completeness does not, by itself,
prove that Hochschild homology commutes with the projective/inductive limits;
that separate convergence assertion is exactly the topologically filtered
theorem quoted in Section~\ref{sec:collapse}.
\end{proof}

\begin{remark}[Scope of the Rees--Fedosov picture]\label{rem:rees-scope}
Formality would organize the local differential beyond its first Poisson
term, and Proposition~\ref{prop:Borel-formality} identifies the finite
classical jets to which such a calculation applies. On the homogeneous
Poisson complex the Euler contraction makes every nonzero weight acyclic,
and the zero-weight splitting
\(\Omega_L^j(Y)\oplus(dr/r)\wedge\Omega_L^{j-1}(Y)\) has cohomology
\(H_L^j(Y\times S^1_\sigma)\). This reproduces the \(E^2\)-page of
Proposition~\ref{prop:E2}, but only because those steps are the ones
already carried out in Sections~\ref{sec:poisson}--\ref{sec:homogeneity};
neither \eqref{eq:chain-formality} nor \eqref{eq:formal-HH} is used. The
formal answer \eqref{eq:formal-HH} is a different group (over
\(\C((\hbar))\), for all of \(M\), without the order circle), and nothing
here licenses specialization of formal Hochschild homology to continuous
Hochschild homology. The Rees picture is therefore a heuristic explanation
of where the Poisson differential comes from, not a second proof of
Theorem~\ref{thm:main}.
\end{remark}

\section{Checks and consequences}\label{sec:checks}

\subsection{The closed-manifold case}

If \(\partial X=\varnothing\), Laurent cohomology is ordinary cohomology and
\(Z=\varnothing\). Formula \eqref{eq:expanded} reduces to
\[
 \HH_k^{\mathrm{cont}}(\cA(X;E))
 \cong H^{2n-k}(S^*X)
 \oplus H^{2n-k-1}(S^*X),
\]
equivalently \(H^{2n-k}(S^*X\times S^1_\sigma)\). This is the
Brylinski--Getzler computation \cite{BG}.

\subsection{Degree zero and residue traces}

For the completed chain complex \eqref{eq:BN-chains}, degree-zero
Hochschild homology is
\[
 \HH_0^{\mathrm{cont}}(\cA)
 =\cA/\operatorname{im}(b:C_1^{\mathrm{cont}}\to\cA).
\]
We abbreviate the image by \([\cA,\cA]_{\mathrm{cont}}\); it need not be
closed.  Its Hausdorffization is the different quotient
\[
 \HH_0^{\mathrm{cont}}(\cA)_{\mathrm H}
 =\cA/\overline{[\cA,\cA]_{\mathrm{cont}}}.
\]
Theorem~\ref{thm:main} computes the former group; by itself it does not
prove that the commutator image is closed (but see below for \(k=0\)).
Continuous traces factor through the Hausdorffization.

Now assume \(n\geq2\), \(X\) connected and \(\partial X\neq\varnothing\).
Then \(Y\) is connected, since its fibers \(S^{n-1}\) are connected, and
\(Z\) has one component over each component of \(\partial X\). The
manifold \(Y\) is orientable, being diffeomorphic to \(S^*X\)
(Proposition~\ref{prop:anchor}(ii)), a cooriented hypersurface in the
symplectic manifold \(T^*X\setminus0\); hence its boundary \(Z\) is a
closed orientable manifold of even dimension \(2n-2\). (It is not a contact
manifold, which would require odd dimension.) In \eqref{eq:expanded} with
\(k=0\), \(H^{2n}(Y)=0\) and \(H^{2n-1}(Z)=0\) for dimension reasons, and
\(H^{2n-1}(Y)=0\) because \(Y\) is a connected compact manifold with
nonempty boundary. Hence
\[
 \HH_0^{\mathrm{cont}}(\cA_{b,L})\cong H^{2n-2}(Z)\cong\C^{\pi_0(\partial X)}.
\]
Continuous traces are continuous linear functionals vanishing on
\([\cA,\cA]_{\mathrm{cont}}\), so they form a space of dimension at most
\(|\pi_0(\partial X)|\). Conversely, the boundary-face residue functionals,
one for each component of \(\partial X\) (see \cite{BNfamilies}; compare
the residue traces of the cusp calculus in \cite{MN}), are
continuous traces; granting their linear independence, the space of
continuous traces has dimension exactly \(|\pi_0(\partial X)|\). In that
case \([\cA,\cA]_{\mathrm{cont}}\) is the joint kernel of finitely many
continuous functionals, so it is closed and
\(\HH_0^{\mathrm{cont}}(\cA)=\HH_0^{\mathrm{cont}}(\cA)_{\mathrm H}\). If
\(\partial X\) is connected, the trace is the b-residue trace up to scale.

\subsection{A product cylinder}

Let \(X=[0,1]\times N\), where \(N\) is closed of dimension \(n-1\), and put
\[
 W=S(\mathbb R\oplus T^*N).
\]
Then \(Y\simeq[0,1]\times W\) and \(Z=W\sqcup W\). Therefore
\[
 \HH_k^{\mathrm{cont}}(\cA_{b,L})
 \cong H^{2n-k}(W)
 \oplus H^{2n-k-1}(W)^{\oplus3}
 \oplus H^{2n-k-2}(W)^{\oplus2}.
\]
This example separates the order-circle contribution from the two boundary
faces.

\part{The residual and boundary ideals}\label{part:ideals}

\section{From symbols to operators}\label{sec:unreduced}

Part~\ref{part:symbols} treats the quotient of the operator algebra by
\emph{all} smoothing operators with Laurent coefficients. To recover the
operator algebra one has to choose the residual ideal and its topology
(Section~\ref{sec:symbol}). We make the choice of Melrose--Nistor
\cite[(8)--(10)]{MN}: the small calculus, Laurent powers \(x^{-\Z}\), and the
ideal \(\Ib=x^\infty\Psi_b^{-\infty}\) of operators with kernels in
\(\dot C^\infty(X^2)\). With these choices the remaining pieces are
\(\Ib\), the boundary ideal \(\Idb\), and the extensions of
Figure~\ref{fig:extensions}. The H-unitality of the ideals
(Lemma~\ref{lem:Hunital}) puts the excision theorem at our disposal. At
the level of homology, the extension
\[
 0\to\Idb\to\Ab\to\Asb\to0
\]
replaces the extension by \(\Psi_{b,L}^{-\infty}\) that was discussed in
earlier versions of these notes. Its connecting map is computed in
Section~\ref{sec:A}.

\section{Conventions for the b-calculus}\label{sec:bconv}

We follow the companion notes \cite{Notes}, whose conventions are recalled
here.

\subsection{Kernels and coordinates}
Extend \(\mu_b=\frac{dx}{x}\nu_\partial\) from the collar to a positive
b-density on \(X\). Operators act on functions, and the kernel of \(A\) is
defined by \(Au(p)=\int_XK_A(p,p')u(p')\,\mu_b(p')\). The b-stretched
product \(X_b^2\) is obtained from \(X^2\) by blowing up \(H\times H\) for
every component \(H\). It has front faces \(\mathrm{ff}_H\) and the lifts
\(\mathrm{lb}\), \(\mathrm{rb}\) of \(\partial X\times X\) and
\(X\times\partial X\). Near \(\mathrm{ff}_H\) we use the projective
coordinates
\begin{equation}\label{eq:proj}
 (x',s,y,y'),\qquad s=\frac{x}{x'}\in[0,\infty),\qquad z=\log s,
\end{equation}
in which \(\mathrm{ff}_H=\{x'=0\}\), \(\mathrm{lb}=\{s=0\}\) and the lifted
diagonal is \(\Delta_b=\{s=1,\ y=y'\}\). Near \(\mathrm{rb}\) one uses
\((x,s'=x'/x,y,y')\) instead. We write
\(K_A=\kappa_A(x',s,y,y')\) in these coordinates.

The space \(\Psi_b^{-\infty}(X)\) consists of the operators whose kernels
lift to smooth functions on \(X_b^2\) that vanish to infinite order at
\(\mathrm{lb}\cup\mathrm{rb}\). The space \(\Psi_b^m(X)\) consists of
kernels that are classical conormal of order \(m\) at \(\Delta_b\), smooth up
to the front faces, and vanish to infinite order at \(\mathrm{lb}\cup
\mathrm{rb}\). These spaces are closed under composition (Q0)
\cite[Thm.~``Small b-composition'']{Notes}.

\subsection{Indicial families}
For \(A\in\Psi_b^m(X)\), the restriction \(\kappa_A(0,s,y,y')\) of its
kernel to \(\mathrm{ff}_H\) is the convolution kernel of the normal
operator. The indicial family is its Mellin transform,
\begin{equation}\label{eq:indicial}
 I_H(A,\lambda)=\int_0^\infty s^{-i\lambda}\kappa_A(0,s,\cdot,\cdot)
 \frac{ds}{s}
 =\int_{\mathbb R}e^{-iz\lambda}\kappa_A(0,e^z,\cdot,\cdot)\,dz,
\end{equation}
regarded as an operator on \(H\) with kernel relative to \(\nu_\partial\).
This is the convention of \cite[Sect.~``The normal operator'']{Notes}. For
differential operators it gives \(I(P,\lambda)=x^{-i\lambda}Px^{i\lambda}
|_{\partial X}\); in particular, with \(D_x=-i\partial_x\),
\begin{equation}\label{eq:IxDx}
 I(xD_x,\lambda)=\lambda .
\end{equation}
The inversion formula is
\(\kappa_A(0,e^z)=\frac1{2\pi}\int e^{iz\lambda}I_H(A,\lambda)\,d\lambda\).
At \(z=0\) it reads
\begin{equation}\label{eq:diagvalue}
 \kappa_A(0,1,y,y')=\frac1{2\pi}\int_{\mathbb R}I_H(A,\lambda)(y,y')\,
 d\lambda .
\end{equation}

\begin{lemma}[Conjugation by powers of \(x\)]\label{lem:conj}
Let \(A\in\Psi_b^m(X)\), \(m\in\Z\cup\{-\infty\}\), and \(w\in\C\). Then
\(x^{-w}Ax^{w}\in\Psi_b^m(X)\), its kernel is \(s^{-w}K_A\) near each front
face, and
\[
 I_H(x^{-w}Ax^{w},\lambda)=I_H(A,\lambda-iw).
\]
In particular \(\alpha(A):=x^{-1}Ax\) has \(I_H(\alpha(A),\lambda)=
I_H(A,\lambda-i)\). The operator
\[
 D(A):=\frac{d}{dw}\Big|_{w=0}x^{w}Ax^{-w}=[\log x,A]
\]
has kernel \(z\,K_A\), lies in \(\Psi_b^m(X)\), and satisfies
\(I_H(D(A),\lambda)=i\,\partial_\lambda I_H(A,\lambda)\). The map \(D\) is a
derivation of \(x^{-\Z}\Psi_b^{\Z}(X)\) that commutes with \(\alpha\) and
satisfies \(D(x)=0\).
\end{lemma}

\begin{proof}
Since \((x^{-w}Ax^{w}u)(p)=\int x(p)^{-w}K_A(p,p')x(p')^{w}u(p')\,\mu_b\),
the new kernel is \((x'/x)^wK_A\), which equals \(s^{-w}K_A\) near the front
faces. On \(X_b^2\) the function \(x'/x\) is a smooth positive multiple of
\(\rho_{\mathrm{rb}}/\rho_{\mathrm{lb}}\). The factor \((x'/x)^w\) is
therefore smooth on the interior and near \(\Delta_b\), where \(s=1\), and
it grows at most like a power of \(\rho_{\mathrm{lb}}\rho_{\mathrm{rb}}\);
infinite-order vanishing at \(\mathrm{lb}\) and \(\mathrm{rb}\) absorbs this
growth. Since \(s^{-i\lambda}s^{-w}=s^{-i(\lambda-iw)}\), the Mellin
transform is shifted as stated. Differentiating \(s^{w}\) at \(w=0\) gives
\(\log s=z\), and \(ze^{-iz\lambda}=i\partial_\lambda e^{-iz\lambda}\).
Differentiating \(x^w(AB)x^{-w}=(x^wAx^{-w})(x^wBx^{-w})\) shows that \(D\)
is a derivation. It commutes with \(\alpha\) because conjugations by powers
of \(x\) commute, and \(D(x)=0\) because \(x\) commutes with \(x^w\).
\end{proof}

Fix a b-vector field equal to \(x\partial_x\) in the collar and put
\(h=-i\,x\partial_x\) there; we write \(h=xD_x\). Then
\(I(h,\lambda)=\lambda\), and in the collar
\begin{equation}\label{eq:weightcomm}
 [h,x^k]=-ik\,x^k .
\end{equation}

\subsection{Entire functions that are Schwartz on strips}
For a Fréchet space \(F\) let \(\Sexp(\mathbb R;F)\) be the space of smooth
\(g:\mathbb R\to F\) with \(\sup_ze^{N|z|}p(g^{(j)}(z))<\infty\) for all
\(N\), \(j\) and continuous seminorms \(p\). Let \(\cE(F)\) be the space of
entire \(f:\C\to F\) with
\[
 \sup_{|\operatorname{Im}\lambda|\leq N}\langle\lambda\rangle^jp(f(\lambda))
 <\infty\quad\text{for all }N,j,p .
\]
By Cauchy's estimates on slightly wider strips, every derivative of \(f\) satisfies the same
bounds. Write \(\cE=\cE(\C)\).

\begin{lemma}[Paley--Wiener]\label{lem:PW}
The Fourier transform \(\widehat g(\lambda)=\int e^{-iz\lambda}g(z)\,dz\) is a
topological isomorphism \(\Sexp(\mathbb R;F)\to\cE(F)\). It carries
convolution to pointwise product, and multiplication by \(e^{-z}\) to the
shift \(f\mapsto f(\cdot-i)\).
\end{lemma}

\begin{proof}
If \(g\in\Sexp\), then \(\widehat g\) is entire, and for
\(|\operatorname{Im}\lambda|\leq N\) we have
\(|\lambda^j\widehat g(\lambda)|\leq\int e^{N|z|}|g^{(j)}(z)|\,dz\).
Conversely, let \(f\in\cE(F)\) and
\(g(z)=\frac1{2\pi}\int_{\mathbb R}e^{iz\lambda}f(\lambda)\,d\lambda\). For
\(z>0\), Cauchy's theorem moves the contour to
\(\operatorname{Im}\lambda=N\); the vertical sides vanish by the decay on
the strip. This gives \(|g(z)|\leq e^{-Nz}\frac1{2\pi}\int|f(\xi+iN)|\,d\xi\).
For \(z<0\) one uses \(\operatorname{Im}\lambda=-N\), and derivatives are
handled in the same way. Finally,
\(\int e^{-iz(\lambda-i)}g\,dz=\int e^{-iz\lambda}e^{-z}g\,dz\).
\end{proof}

\begin{definition}[Model algebras]\label{def:model}
For a component \(H\), let \(B_H=\cE(\Psi^{-\infty}(H))\) with pointwise
composition, and let \(B'_H=\Psi^{\Z}_{\mathrm{sus,ent}}(H)\) be the algebra
of entire families \(\lambda\mapsto a(\lambda)\) of pseudodifferential
operators on \(H\) of integral order \(m\) that are suspended of order \(m\)
uniformly on every strip \(|\operatorname{Im}\lambda|\leq C\)
\cite{Notes,MelroseEta}. By Lemma~\ref{lem:PW}, \(B_H\) is exactly the set
of indicial families of dilation-invariant operators on the model
\(M_H=[0,\infty)_x\times H\) whose kernels \(\beta(x/x',y,y')\) satisfy
\(\beta(e^z,\cdot)\in\Sexp(\mathbb R;C^\infty(H^2))\). For
\(b\in B'_H\) we write \(b(xD_x)\) for the dilation-invariant operator on
\(M_H\) with indicial family \(b\), and \(\alpha(b)(\lambda)=b(\lambda-i)\).
\end{definition}

\section{Taylor series at the front face}\label{sec:taylor}

\begin{lemma}[Composition at the front face]\label{lem:compff}
Let \(A,A'\in\Psi_b^{-\infty}(X)\) have kernels supported in
\(\{x,x'<\varepsilon\}\), and near \(\mathrm{ff}_H\) let
\(\kappa_A\sim\sum_{j\geq0}x'^j\kappa_j(s,y,y')\) and
\(\kappa_{A'}\sim\sum_{l\geq0}x'^l\kappa'_l(s,y,y')\) be the Taylor
expansions. Then \(\kappa_{AA'}\sim\sum_kx'^k\kappa''_k\), with
\begin{equation}\label{eq:taylorcomp}
 \kappa''_k(s)=\sum_{j+l=k}\int_0^\infty t^j\,\kappa_j(s/t)\circ
 \kappa'_l(t)\,\frac{dt}{t},
\end{equation}
where \(\circ\) denotes composition of kernels on \(H\).
\end{lemma}

\begin{proof}
In the coordinates \eqref{eq:proj},
\[
 K_{AA'}(x,y;x',y')=\int_0^\infty\!\!\int_H
 \kappa_A\Bigl(x'',\frac{x}{x''},y,y''\Bigr)
 \kappa_{A'}\Bigl(x',\frac{x''}{x'},y'',y'\Bigr)\,
 \nu_\partial(y'')\frac{dx''}{x''}.
\]
Substituting \(x''=x't\) gives
\(\int\kappa_A(x't,s/t,\cdot)\circ\kappa_{A'}(x',t,\cdot)\,dt/t\). The
second factor decreases faster than any power of \(t\) and of \(t^{-1}\),
and the first decreases faster than any power of \(s/t\) and of \(t/s\).
Both bounds hold uniformly with all \(x'\)-derivatives, and \(x't<\varepsilon\)
on the support. Hence one may differentiate under the integral sign in
\(x'\) at \(x'=0\). Expanding \(\kappa_A(x't,\cdot)=\sum_j(x't)^j\kappa_j\)
gives \eqref{eq:taylorcomp}.
\end{proof}

\begin{definition}[Twisted Laurent series and the Taylor map]
For an algebra \(\mathcal B\) of entire families, closed under
\(\alpha\), let \(\mathcal B[[x;\alpha]\) be the space of formal series
\(\sum_{k\geq-\ell}x^kb_k\), \(b_k\in\mathcal B\), with the product
\begin{equation}\label{eq:skew}
 (x^ja)(x^kb)=x^{j+k}\alpha^k(a)\,b,\qquad
 \alpha(a)(\lambda)=a(\lambda-i).
\end{equation}
This product is associative: both bracketings of
\((x^ia)(x^jb)(x^kc)\) equal \(x^{i+j+k}\alpha^{j+k}(a)\alpha^k(b)c\).
Every coefficient of a product is a finite sum. For
\(A\in x^{-\ell}\Psi_b^{-\infty}(X)\) the kernel has, near
\(\mathrm{ff}_H\), an expansion \(\kappa_A\sim\sum_{k\geq-\ell}x'^k
\kappa_k(s,y,y')\), because \(x^{-\ell}=x'^{-\ell}s^{-\ell}\). Put
\(\beta_k=s^{-k}\kappa_k\), let \(b_k\) be the Mellin transform of
\(\beta_k\), and define
\[
 T_H(A)=\sum_{k\geq-\ell}x^kb_k\in B_H[[x;\alpha],\qquad
 T=(T_H)_H .
\]
The term \(x^kb_k\) is the operator \(x^k\circ b_k(xD_x)\) on \(M_H\): its
kernel is \(x^k\beta_k(s)=x'^k\kappa_k(s)\).
\end{definition}

\begin{proposition}[Structure of \(\Idb\)]\label{prop:structure}
The Taylor map
\[
 T:x^{-\Z}\Psi_b^{-\infty}(X)\longrightarrow\bigoplus_HB_H[[x;\alpha]
\]
is a surjective algebra homomorphism with kernel \(\Ib\). It is
continuous on each piece \(x^{-\ell}\Psi_b^{-\infty}(X)\). Hence \(\Idb\cong\bigoplus_HB_H[[x;\alpha]\). In addition,
\(T(D(A))=\sum_kx^k\,i\partial_\lambda b_k\), and the constant term of
\(T_H(A)\) is \(I_H(A,\cdot)\) when \(A\in\Psi_b^{-\infty}(X)\).
\end{proposition}

\begin{proof}
\emph{Kernel.} If \(T(A)=0\), the kernel vanishes to infinite order at every
front face, and also at \(\mathrm{lb}\) and \(\mathrm{rb}\). Hence it lies
in \(\dot C^\infty(X_b^2)=\dot C^\infty(X^2)\), which is \(\Ib\). The
converse is clear.

\emph{Cut-off.} Let \(\chi\in C_c^\infty([0,\varepsilon))\) equal \(1\) near
\(0\). The kernel \((1-\chi(x)\chi(x'))K_A\) vanishes near the front faces
and to infinite order at \(\mathrm{lb}\) and \(\mathrm{rb}\), so it lies in
\(\Ib\). Modulo \(\Ib\) we may therefore assume that kernels are supported
in the collar.

\emph{Multiplicativity.} For \(A,A'\in\Psi_b^{-\infty}\),
Lemma~\ref{lem:compff} gives the Taylor coefficients of \(AA'\). Since
\(\kappa_j=s^j\beta_j\) and \(\kappa'_l=s^l\beta'_l\),
\[
 s^{-k}t^j\kappa_j(s/t)\kappa'_l(t)
 =(s/t)^{-l}\beta_j(s/t)\,\beta'_l(t)\qquad(j+l=k),
\]
and \((s/t)^{-l}\beta_j(s/t)\) is the kernel of \(\alpha^l(b_j(xD_x))\) by
Lemma~\ref{lem:conj}. Hence the \(k\)-th coefficient of \(T(AA')\) is
\(\sum_{j+l=k}\alpha^l(b_j)b'_l\), which is the \(k\)-th coefficient of
\(T(A)T(A')\) by \eqref{eq:skew}. From the definitions,
\(T(x^{-\ell}A)=x^{-\ell}T(A)\) and \(T(Ax^{-\ell})=T(A)x^{-\ell}\): the
kernel \(x'^{k-\ell}\kappa_k\) gives \(\beta=s^{\ell}\beta_k\), which is the
kernel of \(\alpha^{-\ell}(b_k)\), while
\((x^kb_k)x^{-\ell}=x^{k-\ell}\alpha^{-\ell}(b_k)\). Since
\(x^{\ell'}\Psi_b^{-\infty}x^{-\ell'}=\Psi_b^{-\infty}\), multiplicativity
on \(x^{-\Z}\Psi_b^{-\infty}\) follows.

\emph{Surjectivity.} Let \(F_H\) be the Fréchet space of smooth functions of
\((s,y,y')\in(0,\infty)\times H^2\) that decrease rapidly, with all
derivatives in \(s\partial_s\) and \(y,y'\), as \(s\to0\) and as
\(s\to\infty\). Given \((b_k)_{k\geq-\ell}\), the functions
\(\kappa_k=s^k\beta_k\) lie in \(F_H\) by Lemma~\ref{lem:PW}. Let
\(\{p_j\}\) be an increasing sequence of seminorms defining \(F_H\), and let
\(\phi\in C_c^\infty([0,1))\) equal \(1\) near \(0\). By Borel's argument,
\(G=\sum_{k\geq0}x'^k\phi(x'/\varepsilon_k)\kappa_{k-\ell}\) converges in
\(C^\infty([0,1);F_H)\) and has the prescribed Taylor series, provided
\(\varepsilon_k\) is so small that the \(k\)-th term has \(C^{k-1}\)-norm,
measured with \(p_k\), at most \(2^{-k}\). Put
\(K=x'^{-\ell}\chi(x)\chi(x')G\) near \(\mathrm{ff}_H\), extended by zero.
In the chart \((x,s')\) near \(\mathrm{rb}\) we have \(x'=xs'\) and
\(s=1/s'\), and the rapid decrease as \(s\to\infty\), uniform in \(x'\),
gives smoothness and infinite-order vanishing at \(\mathrm{rb}\). No
continuous linear right inverse is claimed: the Borel map
\(\prod_kF_H\to C^\infty([0,1/2];F_H)\) has none. A Fréchet space
with a continuous norm, such as \(C^\infty([0,1/2];F_H)\) when \(F_H\) has
one, cannot contain a copy of \(\prod_k\C\), which has no continuous norm.
A continuous linear right inverse would embed one. Continuity of \(T\) itself is clear.
The last assertions follow from Lemma~\ref{lem:conj}.
\end{proof}

\begin{proposition}[Structure of \(\Adb\)]\label{prop:structureA}
The Taylor map extends to a surjective homomorphism
\(x^{-\Z}\Psi_b^{\Z}(X)\to\bigoplus_HB'_H[[x;\alpha]\) with kernel
\(x^\infty\Psi_b^{\Z}(X)\), so that \(\Adb\cong\bigoplus_HB'_H[[x;\alpha]\).
\end{proposition}

\begin{proof}
We use (Q0) in the following form: \(\Psi_b^{\Z}\) is an algebra; for
\(K\geq0\), \(x^K\Psi_b^m\circ\Psi_b^{m'}\) and
\(\Psi_b^{m'}\circ x^K\Psi_b^m\) lie in \(x^K\Psi_b^{m+m'}\) (the latter by
Lemma~\ref{lem:conj}); and conormal kernels admit Taylor expansion at
\(\mathrm{ff}_H\), which \(\Delta_b\) meets transversally, and asymptotic
summation (C3).

\emph{Model identity.} On the model \(M_H\), the operators
\(x^ja(xD_x)\) with \(a\in B'_H\) satisfy, exactly,
\[
 (x^ja(xD_x))(x^kb(xD_x))=x^{j+k}(\alpha^k(a)b)(xD_x).
\]
Indeed,
\(a(xD_x)x^k=x^k(x^{-k}a(xD_x)x^k)\), and by Lemma~\ref{lem:conj} the last
factor has indicial family \(a(\lambda-ik)\). Dilation-invariant operators
compose by multiplying indicial families.

\emph{Cut-off model operators.} For \(a\in B'_H\), put
\(Q_k(a)=\chi(x)\,x^ka(xD_x)\,\chi(x')\), transplanted to \(X\) via the
collar, where \(\chi\in C_c^\infty([0,\varepsilon))\) equals \(1\) on
\([0,\varepsilon/2]\). We claim
\[
 Q_j(a)Q_k(b)-Q_{j+k}(\alpha^k(a)b)\in x^\infty\Psi_b^{\Z}(X) .
\]
The difference is \(\chi x^ja(xD_x)(\chi^2-1)x^kb(xD_x)\chi\). Its kernel is
an integral over \(x''\) of kernels evaluated at \(x/x''\) and
\(x''/x'\), where \(x,x'<\varepsilon\) and \(x''\geq\varepsilon/2\) on the
support of \(\chi^2-1\). Both kernels are smooth there and decrease faster
than any power of \(x/x''\), resp.\ \(x'/x''\), wherever \(x\) and
\(x'\) stay away from the support of \(\chi^2-1\). Near \(\mathrm{ff}\) the
difference therefore vanishes to infinite order. Away from \(\mathrm{ff}\)
it is an operator of order \(\leq m+m'\) supported where
\(x,x'\geq\varepsilon/2\) or near the singular set \(x\approx x''\). Its
kernel need not be smoothing: for \(a=b=1\) the difference is the
multiplication operator \(\chi(\chi^2-1)x^{j+k}\). But it vanishes to
infinite order at \(\mathrm{ff}\), so it lies in \(x^\infty\Psi_b\), which is
the kernel of the Taylor map \(T\). This is all that is used below.

\emph{Taylor expansion.} By transversality, the kernel of
\(A\in x^{-\ell}\Psi_b^m\) has a Taylor expansion at \(\mathrm{ff}_H\) whose
coefficients are dilation-invariant kernels conormal at \(\{s=1,y=y'\}\),
rapidly decreasing as \(|z|\to\infty\). As in
Proposition~\ref{prop:structure}, this defines
\(T_H(A)=\sum x^ka_k\) with \(a_k\in B'_H\), and
\[
 A-\sum_{k<K}Q_k(a_k)\in x^{K}\Psi_b^m(X)+\Ib\quad\text{for every }K .
\]
\emph{Multiplicativity.} Write \(A=\sum_{j<K}Q_j(a_j)+R\) and
\(A'=\sum_{k<K}Q_k(a'_k)+R'\), with \(R,R'\in x^{K}\Psi_b+\Ib\). By the
claim and the ideal property, \(AA'\equiv\sum_{j+k<K}Q_{j+k}(\alpha^k(a_j)
a'_k)\) modulo \(x^{K}\Psi_b\), since \(\Ib\) and \(x^\infty\Psi_b\) are
contained in \(x^K\Psi_b\). Since \(K\) is arbitrary,
\(T(AA')=T(A)T(A')\).

\emph{Kernel and surjectivity.} \(T(A)=0\) means that \(A\in x^K\Psi_b^m\) for
every \(K\). Conversely, given \(\sum_{k\geq-\ell}x^ka_k\) with
\(a_k\in B'_{H,m}\), asymptotic summation of the kernels of \(Q_k(a_k)\)
(C3) produces \(A\in x^{-\ell}\Psi_b^m\) with this Taylor series.
\end{proof}

\begin{example}[The model \(X=[0,1)\)]\label{ex:model}
Let \(\partial X=\{0\}\). Then \(B=\cE\), \(B'\) is the algebra of entire
functions of \(\lambda\) with classical symbol expansions on strips, and
\[
 \Idb\cong\cE[[x;\alpha]
 =\Bigl\{\sum_{k\geq-\ell}x^k\,b_k(xD_x):b_k\in\cE\Bigr\}.
\]
The rule \eqref{eq:skew} is
\(b(xD_x)\,x^k=x^k\,b(xD_x-ik)\), which follows from \eqref{eq:weightcomm}.
The operator \(h=xD_x\) does not belong to \(\Idb\), but it is a
multiplier of \(\Idb\) and an element of \(\Adb\), and
\([h,x^kb(h)]=-ik\,x^kb(h)\). The \(x\)-adic associated graded algebra is
again \(\cE[x,x^{-1};\alpha]\), which is not commutative. This is where the
b-case departs from \cite[proof of Prop.~6]{MN}.
\end{example}

\section{Hochschild tools}\label{sec:hhtools}

\subsection{The chain model}\label{subsec:chainmodel}
All algebras in these notes are treated in one chain model.

\begin{definition}[Filtered nuclear algebras and their chains]
A \emph{filtered nuclear algebra} is an increasing union
\(\mathfrak A=\bigcup_{N\geq0}\mathfrak A_N\) of nuclear Fréchet spaces,
each closed in the next, such that multiplication maps
\(\mathfrak A_N\times\mathfrak A_M\to\mathfrak A_{N+M}\) continuously. Its
chain spaces are
\[
 C_k(\mathfrak A)=\varinjlim_N\mathfrak A_N^{\widehat\otimes(k+1)},
\]
with completed projective tensor products. Write \(d_i\), \(0\leq i<k\), for
multiplication of the factors in slots \(i\) and \(i+1\), and \(d_k\) for
\(a_0\otimes\cdots\otimes a_k\mapsto a_ka_0\otimes a_1\otimes\cdots\otimes
a_{k-1}\). Then \(b=\sum_{i=0}^k(-1)^id_i\) maps the \(N\)-th piece in
degree \(k\) continuously into the \((k+1)N\)-th piece, and
\(b'=\sum_{i=0}^{k-1}(-1)^id_i\). If \(J\) is a closed ideal, then \(J\) and
\(\mathfrak A/J\) are filtered nuclear algebras with pieces
\(J_N=J\cap\mathfrak A_N\) and \(\mathfrak A_N/J_N\).
\end{definition}

The filtrations used are as follows.
\begin{itemize}
\item \(x^{-\Z}\Psi_b^{\Z}(X)\): \(\mathfrak A_N=x^{-N}\Psi_b^N(X)\), with
the Fréchet topology of classical conormal kernels on \(X_b^2\) (symbol
seminorms at \(\Delta_b\), and \(C^\infty\) seminorms weighted by
\(\rho_{\mathrm{lb}}^{-M}\rho_{\mathrm{rb}}^{-M}\) away from \(\Delta_b\)).
\item \(x^{-\Z}\Psi_b^{-\infty}(X)\): \(\mathfrak A_N=x^{-N}
\Psi_b^{-\infty}(X)\), with the corresponding \(C^\infty\) topology.
\item \(\Ib\): the single Fréchet space \(\dot C^\infty(X^2)\).
\item \(\Ab\), \(\Asb\), \(\Isb\), \(\Idb\), \(\Adb\), \(\Adsb\): the
induced quotient and subspace filtrations.
\end{itemize}
For \(\Asb\) these pieces are the diagonal pieces \(F^N_{N,N}\) of
\eqref{eq:BN-chains}, so Part~\ref{part:symbols} uses the same model. Via
the Taylor map (Propositions~\ref{prop:structure}
and~\ref{prop:structureA}), the pieces of \(\Idb\) and \(\Adb\) become
\(x^{-N}\prod_{k\geq0}x^kB_H\) and \(x^{-N}\prod_{k\geq0}x^kB'_{H,N}\), where
\(B'_{H,N}\) consists of the elements of order \(\leq N\), with suspended
seminorms on strips. All ideals that occur (\(\Ib\), the smoothing ideal,
and the ideals of infinite order of vanishing at the boundary) are kernels
of continuous maps (full symbol, Taylor series at the front faces), hence
closed. On a closed subspace the induced Fréchet topology coincides with
any finer Fréchet topology for which the inclusion is continuous, by the
open mapping theorem. For example, \(\Ib\) has the same topology whether it is regarded
as \(\dot C^\infty(X^2)\) or as a subspace of \(x^{-N}\Psi_b^N\).

For \(R=B[[x;\alpha]\) the completed tensor products of nuclear Fréchet
spaces commute with countable products. A \(k\)-chain is therefore a family
\((c_{\mathbf j})\) indexed by multi-indices \(\mathbf j=(j_0,\dots,j_k)\)
with all \(j_i\geq-N\) for some \(N\), where \(c_{\mathbf j}\in
B^{\widehat\otimes(k+1)}\). For \(B=\cE(\Psi^{-\infty}(H))\), the space
\(B^{\widehat\otimes(k+1)}\) consists of entire functions of
\((\lambda_0,\dots,\lambda_k)\) with values in \(\Psi^{-\infty}(H^{k+1})\),
Schwartz on products of strips. In this picture, by \eqref{eq:skew},
\(d_i\) evaluates slot \(i\) at \(\lambda-ij_{i+1}\) and slot \(i+1\) at
\(\lambda\), composes in the \(H\)-variables, and gives the new slot the
exponent \(j_i+j_{i+1}\). The \emph{weight} \(g=\sum_ij_i\) is preserved by
every face map.

\begin{lemma}[Exactness of chain spaces]\label{lem:exactchains}
Let \(J\subset\mathfrak A\) be a closed ideal of a filtered nuclear algebra.
Then for every \(k\)
\[
 0\to\mathcal K_k\to C_k(\mathfrak A)\to C_k(\mathfrak A/J)\to0
\]
is exact. Here \(\mathcal K_k\) is the direct limit over \(N\) of the closed
sums \(\sum_i\mathfrak A_N^{\widehat\otimes i}\widehat\otimes J_N\widehat
\otimes\mathfrak A_N^{\widehat\otimes(k-i)}\).
\end{lemma}

\begin{proof}
For Fréchet spaces, \(\widehat\otimes_\pi\) carries quotient maps to quotient
maps. When one factor is nuclear it also preserves injections, so it is
exact on short exact sequences of nuclear Fréchet spaces \cite[Chs.~43,~50]
{Treves}. Apply this factor by factor to
\(0\to J_N\to\mathfrak A_N\to\mathfrak A_N/J_N\to0\), and then pass to the
direct limit. The limit is taken algebraically, as a filtered colimit of
vector spaces, which is exact. The transition maps are injective by
nuclearity, and homology is computed algebraically, so no strictness or
regularity of the LF topology is needed.
\end{proof}

\subsection{H-unitality and excision}
A filtered nuclear algebra is \emph{topologically H-unital} if its
\(b'\)-complex admits a contraction \(E\), with \(b'E+Eb'=1\) in all degrees
(\(b'=0\) on \(C_0\)), that maps \(\mathfrak A_N^{\widehat\otimes(k+1)}\)
continuously into \(\mathfrak A_N^{\widehat\otimes(k+2)}\).

\begin{theorem}[Excision; quoted input (Q1)]\label{thm:excision}
Let \(J\subset\mathfrak A\) be a closed ideal of a filtered nuclear algebra,
and suppose \(J\) is topologically H-unital. Then
\(C(J)\to\mathcal K\) is a quasi-isomorphism, where \(\mathcal K\) is the
kernel of Lemma~\ref{lem:exactchains}. Consequently there is a long exact
sequence
\[
 \cdots\to\HH_k(J)\to\HH_k(\mathfrak A)\to\HH_k(\mathfrak A/J)
 \xrightarrow{\partial}\HH_{k-1}(J)\to\cdots .
\]
To compute \(\partial[c]\), lift \(c\) to a chain \(\tilde c\) of
\(\mathfrak A\), which is possible by Lemma~\ref{lem:exactchains}; then
\(\partial[c]\) is the class of \(b\tilde c\in\mathcal K\). This is
Wodzicki's theorem \cite{Wodzicki}. His proof is algebraic, and in the
present model its inputs are the exactness of Lemma~\ref{lem:exactchains}
and the contraction \(E\). We have not rewritten the proof in the completed
setting, so we list the theorem as an input. No continuous linear section
of \(\mathfrak A\to\mathfrak A/J\) is assumed. The Taylor maps have no such
section, because of the Borel-map obstruction.
\end{theorem}

\begin{lemma}[Connecting maps on cocycles]\label{lem:conn}
In the situation of Theorem~\ref{thm:excision}, let \(\tau\) be a
\((k-1)\)-cocycle on \(J\), and let \(\hat\tau\) be a continuous
\((k-1)\)-cochain on \(\mathfrak A\) that extends \(\tau\) and is such that
\(b\hat\tau\) vanishes whenever one of its arguments lies in \(J\). Then
\(b\hat\tau\) descends to a cocycle \(\overline{b\hat\tau}\) on
\(\mathfrak A/J\), and \(\langle\tau,\partial[c]\rangle=
\langle\overline{b\hat\tau},c\rangle\) for every cycle \(c\).
\end{lemma}

\begin{proof}
The hypothesis says that \(\hat\tau|_{\mathcal K}\) is a cocycle on
\(\mathcal K\). It restricts to \(\tau\) on \(C(J)\), so under the
quasi-isomorphism it represents \([\tau]\). Hence
\(\langle\tau,\partial[c]\rangle=\langle\hat\tau,b\tilde c\rangle=
\langle b\hat\tau,\tilde c\rangle=\langle\overline{b\hat\tau},c\rangle\).
\end{proof}

\begin{lemma}[Explicit contractions]\label{lem:Hunital}
\begin{enumerate}
\item[(a)] Let \(V,W\) be Fréchet spaces with a continuous pairing
\(\langle\cdot,\cdot\rangle:W\times V\to\C\), and let
\(\mathcal K=V\widehat\otimes W\) carry the product
\((v\otimes w)(v'\otimes w')=\langle w,v'\rangle v\otimes w'\). Choose
\(\psi\in W\) and \(\tilde\psi\in V\) with \(\langle\psi,\tilde\psi\rangle=1\).
Then
\[
 E\bigl((v\otimes w)\otimes c\bigr)=(v\otimes\psi)\otimes
 (\tilde\psi\otimes w)\otimes c
\]
is a continuous contraction of the \(b'\)-complex. This applies to
\(\Psi^{-\infty}(H)=C^\infty(H)\widehat\otimes C^\infty(H)\) and to
\(\Ib=\dot C^\infty(X)\widehat\otimes\dot C^\infty(X)\), with the pairings
given by \(\nu_\partial\) and \(\mu_b\).
\item[(b)] Let \(B=\cE(\Psi^{-\infty}(H))\) and let \(R\) be \(B[[x;\alpha]\),
\(B\rtimes_\alpha\Z=\bigoplus_kx^kB\), or the same algebras with \(\cE\) in
place of \(B\). Let \(\phi(\zeta)=e^{-\zeta^2}\). On the component
\(\mathbf j=(j_0,\dots,j_k)\) define \(EF\) as the component
\((0,j_0,\dots,j_k)\) given by
\[
 (EF)(\mu,\lambda_0,\dots;v,v',u,u',\dots)
 =\phi(\mu-\lambda_0+ij_0)\,\psi(v')\tilde\psi(u)\,
 F(\lambda_0,\dots;v,u',\dots).
\]
Here \((v,v')\) are the kernel variables of the new slot, \((u,u')\) those
of the old slot \(0\), and \(\int_H\psi\tilde\psi\,\nu_\partial=1\). Then
\(E\) is a continuous contraction of the \(b'\)-complex of \(R\). It
preserves weights and pole bounds.
\end{enumerate}
Consequently \(\Ib\), \(\Psi^{-\infty}(H)\), \(\Idb\), \(B\rtimes_\alpha\Z\),
\(\cE[[x;\alpha]\) and \(\cE\rtimes_\alpha\Z\) are topologically H-unital.
\end{lemma}

\begin{proof}
In both cases \(d_0E=1\) and \(d_{i+1}E=Ed_i\) for \(i\geq0\). This gives
\(b'E=1-Eb'\). For (a):
\((v\otimes\psi)(\tilde\psi\otimes w)=v\otimes w\), and
\(d_1E((v\otimes w)\otimes a_1\otimes\cdots)=(v\otimes\psi)\otimes
(\tilde\psi\otimes w)a_1\otimes\cdots=Ed_0(\cdots)\), because
\((v\otimes w)a_1=v\otimes(w\cdot a_1)\).

For (b), merging the new slot (exponent \(0\), variable \(\mu\)) with the old
slot \(0\) (exponent \(j_0\)) sets \(\mu=\lambda-ij_0\) and
\(\lambda_0=\lambda\), by \eqref{eq:skew}. Then
\(\phi(\lambda-ij_0-\lambda+ij_0)=\phi(0)=1\), and the kernel factors give
\(\int\psi\tilde\psi=1\); hence \(d_0E=1\). Merging the old slots \(0\) and
\(1\) sets \(\lambda_0=\lambda-ij_1\) and \(\lambda_1=\lambda\), and turns
the prefactor into \(\phi(\mu-\lambda+i(j_0+j_1))\). This is the prefactor
of \(E\) on the merged component, whose first exponent is \(j_0+j_1\);
hence \(d_1E=Ed_0\). For \(i\geq2\) the identity \(d_iE=Ed_{i-1}\) is
clear. Finally,
\(|\phi(\zeta)|=e^{-(\operatorname{Re}\zeta)^2+(\operatorname{Im}\zeta)^2}\).
Together with the decay of \(F\) in \(\lambda_0\), this shows that \(EF\) is
Schwartz on products of strips, with continuous dependence on \(F\).
\end{proof}

\begin{lemma}[Inner derivations {\cite[1.3.3]{Loday}}]\label{lem:inner}
Let \(\mathfrak A\) be unital and \(h\in\mathfrak A\). Put
\(s_h(a_0\otimes\cdots\otimes a_k)=\sum_{i=0}^k(-1)^{i+1}a_0\otimes\cdots
\otimes a_i\otimes h\otimes a_{i+1}\otimes\cdots\otimes a_k\). Then
\(bs_h+s_hb=L_h\), where \(L_h(a_0\otimes\cdots\otimes a_k)=\sum_ia_0\otimes
\cdots\otimes[h,a_i]\otimes\cdots\otimes a_k\).
\end{lemma}

\begin{proof}
This is a direct computation. For \(k=0\):
\(bs_h(a_0)=-b(a_0\otimes h)=[h,a_0]\).
\end{proof}

\begin{lemma}[Morita invariance for smoothing kernels]\label{lem:morita}
Let \(\mathcal K=V\widehat\otimes W\), and let \(\psi\) and
\(\tilde\psi\) be as in Lemma~\ref{lem:Hunital}(a), and put \(e=\tilde\psi\otimes\psi\). Let
\(\mathfrak A\) be unital or topologically H-unital. Then
\[
 \iota(a_0\otimes\cdots\otimes a_k)=(e\otimes a_0)\otimes\cdots\otimes
 (e\otimes a_k)
\]
and the generalized trace
\(\Tr_*\bigl((\kappa_0\otimes a_0)\otimes\cdots\otimes(\kappa_k\otimes a_k)
\bigr)=\Tr(\kappa_0\cdots\kappa_k)\,a_0\otimes\cdots\otimes a_k\) are chain
maps with \(\Tr_*\iota=1\). Moreover, \(C(\mathcal K\widehat\otimes
\mathfrak A)=\iota C(\mathfrak A)\oplus C'\) with \(C'=\ker\Tr_*\) acyclic.
If \(\mathfrak A\) is graded by a weight and the contraction preserves
weights, the statement holds weight by weight.
\end{lemma}

\begin{proof}
Write a chain as
\((v_0\otimes w_0\otimes a_0)\otimes\cdots\otimes(v_k\otimes w_k\otimes a_k)\)
and rearrange it as the \emph{letters} \(\ell_i=w_i\otimes v_{i+1}\in
U:=W\widehat\otimes V\) (indices mod \(k+1\)), together with
\(a_0\otimes\cdots\otimes a_k\). Let \(\varepsilon(w\otimes v)=\langle w,v
\rangle\). The face \(d_i\) applies \(\varepsilon\) to the letter
\(\ell_i\), multiplies \(a_i\) with \(a_{i+1}\) (for \(i=k\) it puts
\(a_ka_0\) in slot \(0\)), and leaves the other letters unchanged. Split
\(U=\C u_0\oplus U'\), where \(u_0=\psi\otimes\tilde\psi\) and
\(U'=\ker\varepsilon\). Because \(\varepsilon(U')=0\), the number \(m\) of
letters lying in \(U'\) is preserved by \(b\). For \(m=0\) all letters are
\(u_0\), so every slot is \(\tilde\psi\otimes\psi=e\); this summand is
\(\iota C(\mathfrak A)\), and \(b\) acts on it as the Hochschild boundary
of \(\mathfrak A\). Projection onto it is \(\Tr_*\), since
\(\prod_i\varepsilon(\ell_i)=\Tr(\kappa_0\cdots\kappa_k)\).

For \(m\geq1\), filter by the position \(w\) of the last \(U'\)-letter. A
face \(d_i\) with \(i<w\) lowers \(w\), the face \(d_w\) vanishes, and the
faces \(d_i\) with \(w<i\leq k\) preserve \(w\). In the associated graded
complex only the faces \(d_{w+1},\dots,d_k\) survive, with alternating
signs \((-1)^i\). They multiply consecutive elements of the block
\((a_{w+1},\dots,a_k,a_0)\), which is a \(b'\)-complex of \(\mathfrak A\) of
length \(k-w+1\geq1\), tensored with the spectators \(a_1,\dots,a_w\) and
the earlier letters. That \(b'\)-complex is contractible: by \(1\otimes\)
in the unital case, or by \(E\), inserted at the front of the block. In
each degree the filtration is finite, so \(C'\) is acyclic.
\end{proof}

\begin{theorem}[Crossed products by \(\Z\); quoted (Q2)]\label{thm:crossed}
Let \(\mathfrak B\) be unital or topologically H-unital (with a
contraction compatible with \(\alpha\)), let \(\alpha\) be an automorphism,
and let \(\mathfrak B\rtimes_\alpha\Z=\bigoplus_kx^k\mathfrak B\) with the
product \eqref{eq:skew}. Let \(C^{(0)}\) be the weight-zero part of its
Hochschild complex. Then there are natural short exact sequences
\[
 0\to\HH_q(\mathfrak B)_\alpha\to H_q(C^{(0)})\to
 \HH_{q-1}(\mathfrak B)^\alpha\to0,
\]
where \(V_\alpha=V/(1-\alpha_*)V\) and \(V^\alpha=\ker(1-\alpha_*)\). In the
unital case, and algebraically, \(C^{(0)}\) is quasi-isomorphic to the
mapping cone of \(1-\alpha_*\) on \(C(\mathfrak B)\)
\cite{FeiginTsygan,NistorCrossed,GetzlerJones}. We assume the same
statement in the chain model of Section~\ref{subsec:chainmodel} for
\(\mathfrak B=\cE\). We also assume the reduction of the H-unital case to
the unital one (apply the unital case to the unitization, then
Theorem~\ref{thm:excision} weight by weight). Neither step is proved here.
\end{theorem}

\section{The residual ideal}\label{sec:I}

\begin{proposition}\label{prop:I}
The ideal \(\Ib\) is topologically H-unital, and \(\HH_0(\Ib)\cong\C\) by
the trace \(\Tr(A)=\int_XK_A(p,p)\,\mu_b(p)\), while \(\HH_k(\Ib)=0\) for
\(k\geq1\). The same holds for \(\Psi^{-\infty}(H)\).
\end{proposition}

\begin{proof}
By the kernel theorem, \(\dot C^\infty(X^2)=\dot C^\infty(X)\widehat\otimes
\dot C^\infty(X)\) (the Schwartz kernel theorem for flat functions), and composition is
the product of Lemma~\ref{lem:Hunital}(a) for the pairing
\(\int_Xwv\,\mu_b\). Apply Lemma~\ref{lem:Hunital}(a) and
Lemma~\ref{lem:morita} with \(\mathfrak A=\C\), for which \(C(\C)\) is
\(\C\) in degree \(0\) up to an acyclic complex. The generalized trace is
\(\Tr\).
\end{proof}

\section{The boundary ideal}\label{sec:Id}

We first treat the scalar model \(\cE[[x;\alpha]\) of
Example~\ref{ex:model}. The general case then follows by
Lemma~\ref{lem:morita}.

\subsection{\texorpdfstring{The algebra \(\cE\)}{The algebra E}}
Chains of \(\cE\) are entire functions \(F(\lambda_0,\dots,\lambda_k)\)
that are Schwartz on products of strips, and
\[
 (bF)(\lambda_0,\dots,\lambda_{k-1})
 =\sum_{i<k}(-1)^iF(\dots,\lambda_i,\lambda_i,\dots)
 +(-1)^kF(\lambda_0,\dots,\lambda_{k-1},\lambda_0).
\]
Let \(\chi_0=1\), \(\chi_1(F)=(\partial_{\lambda_1}F)(\lambda,\lambda)\,
d\lambda\), and \(\chi_k=0\) for \(k\geq2\). Let \(\iota_0=1\) and
\(\iota_1(f\,d\lambda)=f(\lambda_0)\psi_1(\lambda_1-\lambda_0)\), where
\(\psi_1(\zeta)=\zeta e^{-\zeta^2}\).

\begin{proposition}[HKR for \(\cE\)]\label{prop:HKRE}
\begin{enumerate}
\item[(a)] The maps \(\chi\) and \(\iota\) are chain maps between
\(C(\cE)\) and the complex \(\cE\leftarrow\cE\,d\lambda\) with zero
differential. They satisfy \(\chi\iota=1\), and they commute with the
diagonal action of \(\alpha\).
\item[(b)] \emph{(Division lemma.)} If \(G\in\cE(\C^2)\) and
\(G(\lambda,\lambda)=0\), then \(G=(\lambda_0-\lambda_1)G_1\) with
\(G_1\in\cE(\C^2)\) depending continuously on \(G\).
\item[(c)] The Koszul complex \(0\to\cE(\C^2)\xrightarrow{\lambda_0-\lambda_1}
\cE(\C^2)\to\cE\to0\), whose last map is restriction to the diagonal, is
exact. Restriction has the continuous right inverse
\(f\mapsto f(\lambda_0)e^{-(\lambda_0-\lambda_1)^2}\).
\item[(d)] \emph{(Sketch, (S1).)} \(\chi\) is a quasi-isomorphism. Thus
\(\HH_0(\cE)=\cE\), \(\HH_1(\cE)=\cE\,d\lambda\), and \(\HH_q(\cE)=0\) for
\(q\geq2\).
\end{enumerate}
\end{proposition}

\begin{proof}
(a) On \(C_1\), \(bF(\lambda)=F(\lambda,\lambda)-F(\lambda,\lambda)=0\), so
\(\chi_0b=0\). For \(F\in C_2\),
\[
 bF(\mu_0,\mu_1)=F(\mu_0,\mu_0,\mu_1)-F(\mu_0,\mu_1,\mu_1)+
 F(\mu_0,\mu_1,\mu_0).
\]
Applying \(\partial_{\mu_1}\) at \(\mu_0=\mu_1\)
gives \(\partial_3F-(\partial_2+\partial_3)F+\partial_2F=0\), so
\(\chi_1b=0\). Also \(\chi_1\iota_1(f)=f\,\psi_1'(0)=f\). Both maps are
built from diagonal evaluations and are therefore translation-equivariant.

(b) Put \(G_1(\lambda_0,\lambda_1)=-\int_0^1(\partial_2G)(\lambda_0,
\lambda_0+t(\lambda_1-\lambda_0))\,dt\), which is entire. If
\(|\operatorname{Re}(\lambda_0-\lambda_1)|\geq1\), then \(|G_1|\leq|G|\).
Otherwise every point of the segment has real part within \(1\) of
\(\operatorname{Re}\lambda_1\). Cauchy estimates for \(\partial_2G\) on
slightly wider strips then give decay in both variables. Derivatives are
estimated in the same way.

(c) Multiplication by \(\lambda_0-\lambda_1\) is injective on entire
functions; exactness in the middle is (b); and the stated map is a right
inverse.

(d) For a unital algebra, (c) is a resolution of the diagonal bimodule by
free bimodules. The comparison theorem then makes the Hochschild complex
homotopy equivalent to \(\cE\otimes_{\cE^e}(\text{Koszul})=(\cE\xleftarrow0
\cE)\), and \(\chi\) realizes this equivalence. The algebra \(\cE\) is not
unital. Its bar complex is still exact, by H-unitality
(Lemma~\ref{lem:Hunital}), but the modules \(\cE(\C^{k+2})\) are not free
bimodules, and the comparison homotopy has to be built by hand, using the
coupled unit \(\phi(\lambda_0-\lambda_1)\) of Lemma~\ref{lem:Hunital}(b) in
place of \(1\otimes1\). We have not written this construction out. What is
proved is that \(\HH_0(\cE)=\cE\), which is immediate since \(b=0\) on
\(C_1\), and that \(\chi_1\) maps \(\HH_1(\cE)\) onto \(\cE\,d\lambda\).
\end{proof}

\subsection{Coinvariants of the shift}

\begin{lemma}\label{lem:coinv}
Let \(\alpha f(\lambda)=f(\lambda-i)\) on \(\cE\). Then
\(\ker(1-\alpha)=0\), and the image of \(1-\alpha\) is the closed
hyperplane \(\{f:\int_{\mathbb R}f\,d\lambda=0\}\), with a continuous
inverse given by
\[
 (1-\alpha)^{-1}f=\widehat{g_f},\qquad
 g_f(z)=\frac{\check f(z)}{1-e^{-z}},\qquad
 \check f(z)=\frac1{2\pi}\int e^{iz\lambda}f(\lambda)\,d\lambda .
\]
Consequently \(\cE_\alpha\cong\C\) by \(f\mapsto\frac1{2\pi}\int f\,
d\lambda\) and \(\cE^\alpha=0\). The same holds for \(\cE\,d\lambda\).
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:PW}, \(1-\alpha\) corresponds to multiplication of
\(\check f\in\Sexp\) by \(1-e^{-z}\). This function vanishes on
\(\mathbb R\) only at \(z=0\), where its zero is simple, so the kernel is
zero. If \(\int f=2\pi\check f(0)=0\), then \(\check f/(1-e^{-z})\) is
smooth by the one-variable division lemma. For \(z\geq1\) the factor
\((1-e^{-z})^{-1}\) is bounded with bounded derivatives. For \(z\leq-1\) it
is bounded by \(2e^{z}\), again with derivatives. Hence the quotient lies
in \(\Sexp\) and depends continuously on \(f\). Conversely,
\(\int(1-\alpha)g\,d\lambda=0\) by a contour shift, as in the proof of
Proposition~\ref{prop:generators}(a). The shift preserves \(d\lambda\).
\end{proof}

\subsection{The weight decomposition}

\begin{proposition}[Nonzero weights are acyclic]\label{prop:weights}
Let \(R=B[[x;\alpha]\) with \(B=\cE\) or \(B=\cE(\Psi^{-\infty}(H))\). Then:
\begin{enumerate}
\item[(a)] \(C(R)=C^{(0)}(R)\oplus C^{\neq0}(R)\) as complexes, where
\(C^{(0)}(R)\) consists of the components of weight zero;
\item[(b)] \(C^{(0)}(R)=C^{(0)}(B\rtimes_\alpha\Z)\);
\item[(c)] \(C^{\neq0}(R)\) is acyclic.
\end{enumerate}
Hence \(\HH_*(R)\cong H_*(C^{(0)}(B\rtimes_\alpha\Z))\).
\end{proposition}

\begin{proof}
(a) Projection onto the components of weight zero is defined
componentwise and commutes with \(b\), because every face preserves the
weight.

(b) If all \(j_i\geq-\ell\) and \(\sum j_i=0\), then \(j_i\leq k\ell\), so
only finitely many multi-indices occur. A chain of weight zero is
therefore a finite sum, that is, a chain of the polynomial crossed product.
The converse is clear.

(c) Let \(\tilde R=R\oplus\C[h]\), where \(h\) acts as the multiplier
\(h\cdot x^ka=x^k(\lambda-ik)a\) and \(x^ka\cdot h=x^ka\lambda\). This is a
unital algebra containing \(R\) as an ideal, with \(\tilde R/R=\C[h]\) and
the obvious linear section. By Lemma~\ref{lem:Hunital}(b) and
Theorem~\ref{thm:excision}, \(C(R)\to\mathcal K=\ker(C(\tilde R)\to C(\C[h]))\)
is a quasi-isomorphism. Both complexes split by weight, and the map
respects the splitting, since \(h\) has weight \(0\). By
\eqref{eq:weightcomm}, \([h,x^ka]=-ik\,x^ka\), so \(L_h=-ig\) on chains of
weight \(g\). The homotopy \(s_h\) of Lemma~\ref{lem:inner} preserves
\(\mathcal K\), weights, and pole bounds. Therefore
\(\Sigma=\sum_{g\neq0}(-ig)^{-1}s_hP_g\), where \(P_g\) is the projection to
weight \(g\), is a well-defined contracting homotopy of
\(\mathcal K^{\neq0}\). Hence \(H(\mathcal K^{\neq0})=0\), and so
\(H(C^{\neq0}(R))=0\).
\end{proof}

\subsection{Generators and the theorem}
For \(A\in\Idb\) with \(T_H(A)=\sum_kx^kb_k\), define
\begin{equation}\label{eq:Trd}
 \Tr_{\partial,H}(A)=\frac1{2\pi}\int_{\mathbb R}\Tr_H\,b_0(\lambda)\,
 d\lambda ,
\end{equation}
\begin{equation}\label{eq:phi}
 \varphi_H(A,B)=\Tr_{\partial,H}\bigl(A\,D(B)\bigr),\qquad
 T(D(B))=\sum_kx^k\,i\partial_\lambda b_k .
\end{equation}

\begin{proposition}\label{prop:generators}
\begin{enumerate}
\item[(a)] \(\Tr_{\partial,H}\) is a continuous trace on \(\Idb\).
\item[(b)] \(\Tr_{\partial,H}\circ D=0\), and \(\varphi_H\) is a continuous
cyclic Hochschild \(1\)-cocycle on \(\Idb\).
\item[(c)] There are elements \(e'_H\) and \(1\)-cycles \(z_H\) of weight
zero with \(\Tr_{\partial,H'}(e'_H)=\delta_{HH'}\) and
\(\varphi_{H'}(z_H)=\delta_{HH'}\).
\end{enumerate}
\end{proposition}

\begin{proof}
(a) By \eqref{eq:skew}, for \(j+k=0\),
\[
 \Tr_\partial((x^ja)(x^kb))=\frac1{2\pi}\int\Tr(a(\lambda-ki)b(\lambda))\,
 d\lambda,\quad
 \Tr_\partial((x^kb)(x^ja))=\frac1{2\pi}\int\Tr(b(\lambda+ki)a(\lambda))\,
 d\lambda,
\]
and both vanish if \(j+k\neq0\). Shift the contour by \(\lambda\mapsto\lambda-ki\). This is
justified by Cauchy's theorem on a rectangle, because the integrand is
Schwartz on the strip and the vertical sides therefore tend to zero. The
trace property of \(\Tr_H\) on smoothing operators then gives equality.

(b) \(\Tr_\partial(D(b))=\frac{i}{2\pi}\int\Tr\,\partial_\lambda b_0\,
d\lambda=0\). Using (a) and the Leibniz rule,
\((b\varphi)(a,b,c)=\Tr(abDc)-\Tr(aD(b)c)-\Tr(abDc)+\Tr(caDb)=0\), and
\(\varphi(b,a)=-\Tr(D(b)a)=-\varphi(a,b)\).

(c) Let \(e_H\) be a rank-one projection in \(\Psi^{-\infty}(H)\), and let
\(f,g\in\cE\). Put \(e'_H=fe_H\) with \(\int f=2\pi\), and
\(z_H=(fe_H)\otimes(ge_H)\) in weight zero. Then
\(bz_H=fge_H-gfe_H=0\) and \(\varphi_H(z_H)=\frac{i}{2\pi}\int fg'\,
d\lambda\). For \(g=e^{-\lambda^2}\) and \(f=-c\lambda e^{-\lambda^2}\),
this equals \(\frac{ic}{\pi}\int\lambda^2e^{-2\lambda^2}d\lambda\neq0\);
normalize \(c\). The components \(H'\neq H\) do not see these elements.
\end{proof}

\begin{theorem}\label{thm:Id}
\(\HH_k(\Idb)\cong\C^N\) for \(k=0,1\), and \(\HH_k(\Idb)=0\) for
\(k\geq2\). The functionals \(\Tr_{\partial,H}\) and \(\varphi_H\) are dual
bases in degrees \(0\) and \(1\). This uses (Q1), (Q2) and (S1).
\end{theorem}

\begin{proof}
By Proposition~\ref{prop:structure}, \(\Idb=\bigoplus_HR_H\). Excision for
the split ideals \(R_H\) (Theorem~\ref{thm:excision}) reduces the
computation to a single \(R_H=B_H[[x;\alpha]\).
Proposition~\ref{prop:weights} reduces it further to the weight-zero part
of \(B_H\rtimes_\alpha\Z=\Psi^{-\infty}(H)\widehat\otimes(\cE\rtimes_\alpha
\Z)\). Lemma~\ref{lem:morita}, applied weight by weight, reduces it to
\(C^{(0)}(\cE\rtimes_\alpha\Z)\). Theorem~\ref{thm:crossed} and
Proposition~\ref{prop:HKRE}(d) then give exact sequences with
\(\HH_q(\cE)_\alpha\) and \(\HH_{q-1}(\cE)^\alpha\). By
Lemma~\ref{lem:coinv}, \(\HH_0(\cE)_\alpha\cong\C\),
\((\cE\,d\lambda)_\alpha\cong\C\), and all invariants vanish. Hence
\(H_0\cong\C\), \(H_1\cong\C\), and \(H_q=0\) for \(q\geq2\). By
Proposition~\ref{prop:generators}, \(\Tr_{\partial,H}\) and \(\varphi_H\)
are nonzero on these one-dimensional groups.
\end{proof}

\begin{remark}
In the language of \cite[(16), (68)]{MN},
\(\HH_k(\Idb)\cong\bigoplus_HH^{1-k}(S_\partial)\). The trace
\(\Tr_{\partial,H}\) is a residue in \(x\) (Lemma~\ref{lem:residue}) and
corresponds to \(d\theta_\partial\). The class of \(\varphi_H\) corresponds
to \(1\). In the cusp case, Melrose--Nistor reach the same answer by an
\(x\)-adic spectral sequence whose \(E^1\)-term is commutative; here the
weight argument replaces it.
\end{remark}

\section{The b-trace and the Laurent residual algebra}\label{sec:btrace}

\begin{lemma}[Diagonal residues]\label{lem:residue}
Let \(A\in x^{-\ell}\Psi_b^{-\infty}(X)\). The function
\(\Tr(x^wA)=\int_Xx^wK_A(p,p)\,\mu_b(p)\) is holomorphic for
\(\operatorname{Re}w>\ell\), extends meromorphically to \(\C\) with at
most simple poles at \(w\in\{-k:k\geq-\ell\}\), and
\[
 \operatorname{Res}_{w=0}\Tr(x^wA)=\sum_H\Tr_{\partial,H}(T(A)).
\]
The same holds for holomorphic families \(A(w)\), with residue
\(\sum_H\Tr_{\partial,H}(T(A(0)))\).
\end{lemma}

\begin{proof}
On \(\Delta_b\) the coordinates are \(s=1\), \(x'=x\), and \(y=y'\), so
\(K_A(p,p)\sim\sum_{k\geq-\ell}x^k\kappa_k(1,y,y)\). Moreover
\(\kappa_k(1)=\beta_k(1)=\frac1{2\pi}\int b_k\,d\lambda\) by
\eqref{eq:diagvalue}. Away from the collar the integral is entire. In the
collar, the term \(x^k\) contributes
\(\int_0^\varepsilon x^{w+k-1}dx=\varepsilon^{w+k}/(w+k)\), and the Taylor
remainder of order \(M\) is holomorphic for \(\operatorname{Re}w>-M\). The
residue at \(w=0\) is \(\int_H\kappa_0(1,y,y)\nu_\partial\), which is
\(\frac1{2\pi}\int\Tr_Hb_0\,d\lambda\).
\end{proof}

We define the b-trace by
\({}^b\Tr(A)=\operatorname{FP}_{w=0}\Tr(x^wA)\) for
\(A\in x^{-\Z}\Psi_b^{-\infty}(X)\). On \(\Ib\) it is the trace, and on
\(\Psi_b^{-\infty}\) it is the b-trace of \cite[Sect.~``The b-trace
defect'']{Notes}.

\begin{theorem}[b-trace defect]\label{thm:defect}
For \(A,B\in x^{-\Z}\Psi_b^{-\infty}(X)\) with \(T_H(A)=\sum x^ka_k\) and
\(T_H(B)=\sum x^kb_k\),
\[
 {}^b\Tr([A,B])=\sum_H\Tr_{\partial,H}\bigl(D(A)B\bigr)
 =\sum_H\frac{i}{2\pi}\sum_k\int_{\mathbb R}
 \Tr_H\bigl((\partial_\lambda a_k)(\lambda+ki)\,b_{-k}(\lambda)\bigr)\,
 d\lambda .
\]
For \(A,B\in\Psi_b^{-\infty}(X)\) this is
\(\frac{i}{2\pi}\sum_H\int\Tr_H(\partial_\lambda I_H(A,\lambda)\,
I_H(B,\lambda))\,d\lambda\), which is the formula of
\cite[Thm.~``b-trace defect formula'']{Notes}.
\end{theorem}

\begin{proof}
For \(\operatorname{Re}w\) large, \(x^wAB\), \(x^wBA\) and \(Ax^wB\) are
trace class (Q0), and their traces are integrals of their kernels over the
diagonal. Both \(\Tr(x^wBA)\) and \(\Tr(Ax^wB)\) equal the absolutely
convergent double integral \(\iint x(p)^wK_B(p,q)K_A(q,p)\,\mu_b\mu_b\). Hence
\[
 \Tr(x^w[A,B])=\Tr\bigl((x^wA-Ax^w)B\bigr)=w\,\Tr\bigl(x^wC(w)B\bigr),
 \qquad C(w)=\frac{A-x^{-w}Ax^{w}}{w}.
\]
By Lemma~\ref{lem:conj}, \(C(w)\) has kernel \(w^{-1}(1-s^{-w})K_A\). This
is an entire family in \(x^{-\ell}\Psi_b^{-\infty}\) with \(C(0)=D(A)\).
By Lemma~\ref{lem:residue}, \(\Tr(x^wC(w)B)\) has at most a simple pole at
\(0\), with residue \(\sum_H\Tr_{\partial,H}(D(A)B)\). Hence
\(\Tr(x^w[A,B])\) is regular at \(0\), and its value there is that residue.
The explicit form follows from \eqref{eq:skew}: the constant term of
\((x^ki\partial_\lambda a_k)(x^{-k}b_{-k})\) is
\(\alpha^{-k}(i\partial_\lambda a_k)b_{-k}\).
\end{proof}

\begin{theorem}\label{thm:Laurentres}
Let \(\mathcal M=x^{-\Z}\Psi_b^{-\infty}(X)\). In the long exact sequence of
\(0\to\Ib\to\mathcal M\to\Idb\to0\), the composite
\(\Tr\circ\partial:\HH_1(\Idb)\to\C\) equals \(-\sum_H\varphi_H\).
Consequently \(\HH_0(\mathcal M)\cong\C^N\), spanned by the residues
\(\operatorname{Res}_{w=0}\Tr(x^w\,\cdot\,)\) at the components, and
\(\HH_1(\mathcal M)\cong\C^{N-1}\), while \(\HH_k(\mathcal M)=0\) for
\(k\geq2\).
\end{theorem}

\begin{proof}
The ideal \(\Ib\) is H-unital (Lemma~\ref{lem:Hunital}), and
\(\Ib\) is a closed ideal of the filtered nuclear algebra
\(x^{-\Z}\Psi_b^{-\infty}\), whose quotient is identified with \(\Idb\) by
Proposition~\ref{prop:structure}. So Theorem~\ref{thm:excision} applies. Let \(c=\sum a_i\otimes b_i\) be a
\(1\)-cycle, with lifts \(\tilde a_i,\tilde b_i\). Then
\(\partial[c]=[\sum[\tilde a_i,\tilde b_i]]\). Since \({}^b\Tr=\Tr\) on
\(\Ib\), Theorem~\ref{thm:defect} gives \(\Tr\partial[c]=\sum_{H,i}
\Tr_{\partial,H}(D(a_i)b_i)=-\sum_{H,i}\varphi_H(a_i,b_i)\), using the
cyclicity in Proposition~\ref{prop:generators}. This map
\(\C^N\to\C\) is surjective. Now insert this into the long exact sequence,
together with Proposition~\ref{prop:I} and Theorem~\ref{thm:Id}.
\end{proof}

\begin{remark}[Dependence on choices]\label{rem:choices}
The algebras and extensions depend only on \(X\), because
\(x^{-\Z}\Psi_b^{\Z}\) and the ideals are unchanged under \(x\mapsto ax\)
with \(0<a\in C^\infty(X)\). The Taylor maps \(T_H\) depend on \(x\) and on
the collar. The residue \(\Tr_{\partial,H}=\operatorname{Res}_{w=0}
\Tr(x^w\,\cdot\,)\) does not: since \(a^w=1+w\log a+O(w^2)\), the extra
term \(w\Tr(x^w\log a\,A)\) is regular at \(0\). The b-trace changes by
\(\Tr_\partial(\log a\,\cdot\,)\). The derivation changes to
\(D'=D+[\log a,\cdot]\), so
\[
 \varphi'_H(A,B)=\varphi_H(A,B)+\Tr_{\partial,H}(A[\log a,B])
 =\varphi_H(A,B)-(b\psi)(A,B),\qquad\psi(A)=\Tr_{\partial,H}(A\log a),
\]
using the trace property. Hence the class of \(\varphi_H\) is
independent of the choices.
\end{remark}

\begin{remark}
Theorem~\ref{thm:Laurentres} is a homological form of the statement that
the b-trace is not a trace. The trace on \(\Ib\) admits no extension to a
trace on \(\mathcal M\), and the obstruction is exactly
\(\sum_H\varphi_H\). The traces of \(\mathcal M\) are the \(N\) boundary
residues.
\end{remark}

\section{The interior ideal and the boundary symbol algebra}\label{sec:Is}

\subsection{Convergence and degeneration}\label{subsec:conv}
The algebras \(\Isb\) and \(\Adsb\) are computed by the same order spectral
sequence as \(\Asb\) in Part~\ref{part:symbols}; only the coefficient
functions change. In this subsection we separate the purely algebraic part
of the convergence question from the analytic one.

Let \((C_\bullet,b)\) be a complex such that each \(C_k\) is a
\emph{subspace} of the space of series \(c=\sum_{p\leq P}c_p\) with
\(c_p\in C_{p,k}\), containing every component \(C_{p,k}\), where \(b=\sum_{l\geq0}b_l\)
and \(b_l:C_{p,k}\to C_{p-l,k-1}\). Put \(F_pC_k=\{c:c_q=0\text{ for }q>p\}\),
let \(E^1_{p,k}=H_k(C_{p,\bullet},b_0)\), and let \(d^1\) be induced by
\(b_1\). Since \(\sum_{i+j=l}b_ib_j=0\), \(d^1\) is well defined and
\((d^1)^2=0\); let \(E^2\) be its homology. Because \(C_k\) is only a subspace, completeness is a real condition.
We call the filtration
\emph{complete} if every series \(\sum_je_j\) with \(e_j\in F_{p_j}C_k\),
\(\sup p_j<\infty\) and \(p_j\to-\infty\), defines an element of \(C_k\), and
\(b\) commutes with such sums.

\begin{proposition}[Algebraic convergence]\label{prop:algconv}
Assume the filtration is complete and that, for some integer \(n\),
\(E^2_{p,k}=0\) whenever \(p\neq k-n\). Then every class in \(H_k(C,b)\)
has a representative in \(F_{k-n}C_k\), and
\[
 \Theta:H_k(C,b)\to E^2_{k-n,k},\qquad[c]\mapsto[c_{k-n}],
\]
is an isomorphism. All differentials \(d^r\), \(r\geq2\), vanish.
\end{proposition}

\begin{proof}
This is the argument of \cite[Ch.~``Hochschild homology'', convergence proposition]{MelroseMicro},
written out in our notation.

\emph{Step (P).} Let \(c\in F_pC_k\) with \(b_0c_p=0\) and
\(b_1c_p+b_0c_{p-1}=0\), and suppose \([c_p]=0\) in \(E^2_{p,k}\). Then
\([c_p]=d^1[x]\) for a \(b_0\)-cycle \(x\in C_{p+1,k+1}\), so
\(c_p-b_1x=b_0y\) with \(y\in C_{p,k+1}\). The components of \(b(x+y)\) of
degrees \(p+1\) and \(p\) are \(b_0x=0\) and \(b_1x+b_0y=c_p\), so
\(c-b(x+y)\in F_{p-1}\).

\emph{Representatives.} If \(c\in F_PC_k\) is a cycle and \(P>k-n\), then
\(E^2_{P,k}=0\), so (P) lowers \(P\). Finitely many steps reach
\(F_{k-n}\).

\emph{Well defined.} Let \(c=be\in F_{k-n}\) with \(e\in F_q\). If
\(q>k+1-n\), then (P) applies to \(e\), since
\(b_0e_q=(be)_q=0\), \(b_1e_q+b_0e_{q-1}=(be)_{q-1}=0\) and
\(E^2_{q,k+1}=0\). It replaces \(e\) by an element of \(F_{q-1}\) with the
same boundary. Once \(q\leq k+1-n\), we have
\(c_{k-n}=b_1e_{k+1-n}+b_0e_{k-n}\), so \([c_{k-n}]=d^1[e_{k+1-n}]=0\) in
\(E^2\).

\emph{Injective.} If \(bc=0\), \(c\in F_{k-n}\), and \([c_{k-n}]=0\), apply
(P) repeatedly; this is possible because \(E^2_{p,k}=0\) for \(p<k-n\).
It yields \(e_p\in F_{p+1}C_{k+1}\), \(p=k-n,k-n-1,\dots\), with
\(c-b(\sum_{p\geq P}e_p)\in F_{P-1}\). By completeness, \(c=b\sum_pe_p\).

\emph{Surjective.} Let \(x\in C_{k-n,k}\) be a \(b_0\)-cycle with
\(d^1[x]=0\). Then \(b_1x=-b_0y\) for some \(y\), and \(c^{(1)}=x+y\) has
\(bc^{(1)}\in F_{k-n-2}\). Inductively, if \(z=bc^{(r)}\in F_p\) with
\(p=k-n-r-1\), then \(z\) is a cycle and \(p\neq k-1-n\), so (P) gives
\(x',y'\) of degree \(\leq p+1<k-n\) with \(z-b(x'+y')\in F_{p-1}\). Set
\(c^{(r+1)}=c^{(r)}-x'-y'\). By completeness the limit \(c\) exists,
\(bc=0\), and \(\Theta[c]=[x]\).

\emph{Degeneration.} The differential \(d^r\) maps \(E^r_{p,k}\) to
\(E^r_{p-r,k-1}\). Source and target can both be nonzero only if
\(p=k-n\) and \(p-r=(k-1)-n\), that is, only if \(r=1\). Hence \(d^r=0\) for
\(r\geq2\).
\end{proof}

By Propositions~\ref{prop:E2}, \ref{prop:Is} and~\ref{prop:Ads} below,
the \(E^2\)-hypothesis holds for \(\Asb\), \(\Isb\) and \(\Adsb\) as soon as
the \(E^1\)-page is identified. The analytic input is therefore:

\begin{quote}
\textbf{Hypothesis (C).} For \(\mathfrak S\in\{\Asb,\Isb,\Adsb\}\):
\begin{enumerate}
\item[(C-HKR)] the continuous HKR map identifies \(E^1\) with the
homogeneous forms whose coefficients have the corresponding boundary
behaviour (Laurent, flat, or formal Laurent); and
\item[(C-stage)] for every \(k\) and every stage \(N\) there is a stage
\(N'\geq N\), depending only on \((k,N)\), such that the chains of stage
\(N'\) in degrees \(k\) and \(k\pm1\) form a subcomplex that is stable under
step (P). That is, if \(c\) has stage \(N'\), then the corrections \(x,y\)
can again be chosen of stage \(N'\). Then every iterate in the proof of
Proposition~\ref{prop:algconv} lies in stage \(N'\), and the filtration is
complete there.
\end{enumerate}
\end{quote}

Within a fixed stage \(N\), each component \(C_{p,k}\) is a finite sum of
multihomogeneous pieces, because each factor has order \(\leq N\). The
stage itself is the countable product over \(p\leq(k+1)N\) (since
\(\widehat\otimes_\pi\) commutes with countable products, (C3)). So series
with bounded stage converge, and (C-stage) is exactly the requirement that
iterating (P) does not leave the direct limit. The de Rham homotopies used
at \(E^2\) all preserve pole bounds: the fibre Euler contraction, the
\(x\)-weight contraction of Lemma~\ref{lem:formal}, the Laurent
Poincar\'e lemma, and \(*_b\). The faces of \(b\), however, add the orders
of adjacent factors. In the model of formal sums of multihomogeneous
functions without a per-factor bound (as in \cite{BG,MelroseMicro}),
completeness holds tautologically. In that model, Proposition~\ref{prop:algconv}
proves convergence given (C-HKR). We have not compared that model with
ours, which is needed for the excision sequences. For \(\Asb\), (C) is
part of the Benameur--Nistor theorem (Theorem~\ref{thm:BN-conv}). For
\(\Isb\) and \(\Adsb\), (C) is a hypothesis.

\subsection{Two de Rham lemmas}

\begin{lemma}[Flat forms]\label{lem:flat}
Let \(V\) be a compact manifold with boundary, and let
\(\dot\Omega^*(V)\) be the complex of forms vanishing to infinite order at
\(\partial V\). Then \(H^*(\dot\Omega^*(V))\cong H^*(V,\partial V)\).
\end{lemma}

\begin{proof}
Let \(J^*\) be the complex of formal Taylor series at \(\partial V\) of
forms on \(V\). In a collar these are
\(\sum_{k\geq0}x^k(a_k+dx\wedge c_k)\) with \(a_k,c_k\) forms on
\(\partial V\). By Borel's lemma,
\[
 0\to\dot\Omega^*(V)\to\Omega^*(V)\to J^*\to0
\]
is exact. On \(J^*\) put \(K(\sum x^k(a_k+dx\wedge c_k))=\sum
x^{k+1}c_k/(k+1)\), which is the formal integral from \(0\). The
fundamental theorem of calculus gives \(dK+Kd=1-\pi^*\rho\), where \(\rho\)
restricts to \(x=0\) and \(\pi\) is the collar projection. Hence
\(\rho:J^*\to\Omega^*(\partial V)\) is a quasi-isomorphism. The relative
complex \(\Omega^*(V,\partial V)=\Omega^*(V)\oplus\Omega^{*-1}(\partial V)\),
with \(D(\alpha,\beta)=(d\alpha,\alpha|_{\partial V}-d\beta)\), receives the
chain map \(\alpha\mapsto(\alpha,0)\) from \(\dot\Omega^*(V)\). This map is
compatible with the long exact sequences of the displayed short exact
sequence and of the pair \((V,\partial V)\). The five lemma gives the
claim.
\end{proof}

\begin{lemma}[Formal Laurent forms]\label{lem:formal}
Let \(W\) be a manifold, and let \(\Omega^*(W)((x))\) be formal Laurent
series in \(x\) with coefficients in forms on \(W\), together with the
generator \(dx/x\). The complex
\(\Omega^*(W)((x))\oplus\frac{dx}{x}\wedge\Omega^{*-1}(W)((x))\) has
cohomology \(H^*(W)\oplus\frac{dx}{x}H^{*-1}(W)\cong H^*(W\times
S_\partial)\).
\end{lemma}

\begin{proof}
Let \(\mathcal E_x=x\partial_x\). It preserves the \(x\)-degree \(k\), and
on degree \(k\) we have \(d\iota_{\mathcal E_x}+\iota_{\mathcal E_x}d=
\mathcal L_{\mathcal E_x}=k\). For \(k\neq0\) the map
\(k^{-1}\iota_{\mathcal E_x}\) is a contracting homotopy. These homotopies
can be summed over \(k\), because they do not lower the \(x\)-degree. In
degree \(0\) the complex is
\(\Omega^*(W)\oplus\frac{dx}{x}\wedge\Omega^{*-1}(W)\) with
\(d(\alpha_0+\frac{dx}{x}\wedge\alpha_1)=d\alpha_0-\frac{dx}{x}\wedge
d\alpha_1\).
\end{proof}

\subsection{\texorpdfstring{The \(E^2\)-pages}{The E2-pages}}

\begin{proposition}[Interior ideal]\label{prop:Is}
\(E^2_{p,q}(\Isb)=H^{2n-p-q}(Y\times S_\sigma,Z\times S_\sigma)\) for
\(q=n\), and \(E^2_{p,q}(\Isb)=0\) otherwise, under (C-HKR). Under (C),
\(\HH_k(\Isb)\cong H^{2n-k}(Y\times S_\sigma,Z\times S_\sigma)\). The ideal
\(\Isb\) is the same algebra for the b- and cusp calculi.
\end{proposition}

\begin{proof}
The arguments of Sections~\ref{sec:graded}--\ref{sec:homogeneity} apply
verbatim to symbols that vanish to infinite order at \(x=0\), because
\(\pi_b\), \(*_b\) and the Euler field preserve flatness. The weight-zero
complex is \(\dot\Omega^*(Y\times S_\sigma)\), and Lemma~\ref{lem:flat}
computes its cohomology. The rows \(q\neq n\) vanish as in
Proposition~\ref{prop:E2}, so the spectral sequence degenerates. Kernels
conormal to the diagonal that vanish to infinite order at \(\partial(X^2)\)
do not depend on the blow-up, so \(x^\infty\Psi_b^{\Z}=x^\infty\Psi_c^{\Z}\).
\end{proof}

\begin{proposition}[Boundary symbol algebra]\label{prop:Ads}
\(E^2_{p,q}(\Adsb)=H^{2n-p-q}(Z\times S_\sigma\times S_\partial)\) for
\(q=n\), and \(E^2_{p,q}(\Adsb)=0\) otherwise, under (C-HKR). Under (C),
\(\HH_k(\Adsb)\cong H^{2n-k}(Z\times S_\sigma\times S_\partial)\).
\end{proposition}

\begin{proof}
Elements of \(\Adsb\) are formal Laurent series in \(x\) of complete b-symbols
at \(\partial X\). On the \(E^1\)-page the coefficients are formal Laurent
series in \(x\) with values in homogeneous forms on
\({}^bT^*_{\partial X}X\setminus0\). The b-Poisson tensor
\(\pi_b=x\partial_x\wedge\partial_\xi+\sum\partial_{y_j}\wedge\partial_{\eta_j}\)
and the duality \(*_b\) preserve the \(x\)-degree, so \(d^1\) and the
symplectic duality of Section~\ref{sec:poisson} apply degree by degree.
The fibre Euler field kills the nonzero fibre weights
(Section~\ref{sec:homogeneity}); the weight-zero part produces
\(S_\sigma\). Lemma~\ref{lem:formal}, with \(W=Z\times S_\sigma\), kills
the nonzero \(x\)-degrees and produces \(S_\partial\). The row argument of
Proposition~\ref{prop:E2} gives degeneration.
\end{proof}

\begin{remark}[\(n=1\)]\label{rem:n1}
Let \(n=1\). Then \(Y\) is two copies of \(X\) (\(\xi=\pm1\)), and each
\(Z_H\) is two points with the orientation of \(\partial Y\).
Parts (iv)--(vi) of Theorem~\ref{thm:full} hold as stated. In (vii) and
(viii), \(H^{2n-2}(Z)=H^0(Z)\cong\C^{2N}\), and the summand \(H^{2n-2}(Z)\)
must be replaced by the kernel of
\(\beta\mapsto(\int_{Z_H}\beta)_H\in\C^N\), where \(\int_{Z_H}\) is the signed
sum over the two points. Concretely:
\begin{itemize}
\item \(\HH_1(\Ab)\cong\C\oplus H^1(Y\times S_\sigma)\oplus
\ker\bigl(H^0(Z)\to\C^N\bigr)\);
\item \(\HH_2(\Ab)\cong\ker\bigl(H^0(Y)\to\C^N\bigr)\), where the map is
\(\beta\mapsto(\int_{Z_H}\beta)_H\);
\item for \(k=1,2\), \(\HH_k(\Adb)\) is the kernel of the surjection
\(\HH_k(\Adsb)\to\C^N\) given by the same signed sums.
\end{itemize}
For \(n=1\) the surjectivity of the connecting maps of \(\Adb\) and of
\(\partial^{\mathcal A}_1\) is proved by explicit test cycles
(Propositions~\ref{prop:n1} and~\ref{prop:connA}(d)), without (C).
For \(X=[0,1]\), write \(Y=Y_+\sqcup Y_-\) (\(\xi=\pm1\)), with the
symplectic orientation, so that \(Y_-\) is oriented by \(-dx\). Then
\(H^0(Y)=\C^2\), and the map to \(\C^2\) is, up to an overall sign,
\((c_+,c_-)\mapsto(c_--c_+,\,c_+-c_-)\). This agrees with the difference
\(g(1)-g(-1)\) in \(\hat\rho_H\). Its kernel is the constants; for example,
the cycle \(x^{-1}\otimes h\otimes x-x^{-1}\otimes x\otimes h\) represents
\((1,1)\), and \(\Omega_H\) vanishes on it. The rank is \(1=N-c\), so
\(\HH_2(\Ab)\cong\C\neq0\). Identifying \(\partial^{\mathcal A}_2\) with this
map uses (C) and the sketch of Remark~\ref{rem:description}.
\end{remark}

\section{\texorpdfstring{The indicial algebra \(\Adb\)}{The indicial algebra}}\label{sec:Ad}

\subsection{The regularized suspended trace}
Let \(a\in B'_H\) have order \(m\) and put \(d=\dim H=n-1\). For \(p>m+d\)
the operator \(\partial_\lambda^pa(\lambda)\) has order \(m-p<-d\), so it is
trace class. On \(\mathbb R\) the function
\(f_p(\lambda)=\Tr_H\,\partial_\lambda^pa(\lambda)\) has a classical
expansion \(f_p\sim\sum_jc_j^\pm|\lambda|^{m-p+d-j}\) as
\(\lambda\to\pm\infty\). To see this, apply the trace formula
\(\Tr P=(2\pi)^{-d}\iint\sigma_P\,dy\,d\eta\) to the suspended symbol and
substitute \(\eta=|\lambda|\eta'\) in each homogeneous term. Let \(g_p\) be the
\(p\)-fold iterated integral of \(f_p\) from \(0\). Then
\(\int_{-T}^Tg_p\,d\lambda\) has an asymptotic expansion in powers \(T^j\)
and terms \(T^j\log T\) as \(T\to\infty\). Define \(\overline\Tr(a)\) as
\(1/2\pi\) times its constant term.

\begin{lemma}\label{lem:bartr}
\begin{enumerate}
\item[(1)] \(\overline\Tr(a)\) does not depend on \(p\). If \(a\) has order
\(<-n\), then \(\overline\Tr(a)=\frac1{2\pi}\int\Tr_Ha\,d\lambda\).
\item[(2)] \(\overline\Tr([a,b])=0\) for \(a,b\in B'_H\).
\item[(3)] If \(a\in B_H\), then \(\overline\Tr(\partial_\lambda^pa)=0\) for
\(p\geq1\).
\end{enumerate}
\end{lemma}

\begin{proof}
(1) Since \(g_{p+1}^{(p+1)}=f_{p+1}=f_p'=g_p^{(p+1)}\), the difference
\(g_{p+1}-g_p\) is a polynomial. The constant term of
\(\int_{-T}^T\lambda^j\,d\lambda\) is zero for every \(j\). For order
\(<-n\), \(g_p\) differs from \(\Tr_Ha\) by its Taylor polynomial at \(0\),
and \(\Tr_Ha\) is integrable.
(2) Here \(f_p([a,b])=\sum_j\binom pj\Tr_H[\partial^ja,\partial^{p-j}b]\).
Each term vanishes, because \(\Tr(PQ)=\Tr(QP)\) for pseudodifferential
operators on a closed manifold whose orders add up to less than \(-d\).
(3) follows from (1), since \(\int\partial_\lambda(\Tr\,\partial^{p-1}a)
\,d\lambda=0\).
\end{proof}

\begin{lemma}[Shift defect]\label{lem:shift}
Let \(a\in B'_H\) have order \(m\), let \(c\in\C\), and let \(P>m+n\). Then
\(a_c(\lambda)=a(\lambda+c)\) lies in \(B'_H\), and
\[
 \overline\Tr(a_c)-\overline\Tr(a)
 =\sum_{p=1}^{P-1}\frac{c^p}{p!}\,\overline\Tr(\partial_\lambda^pa).
\]
\end{lemma}

\begin{proof}
By Taylor's formula,
\(a_c=\sum_{p<P}\frac{c^p}{p!}\partial^pa+R\), where
\(R(\lambda)=\frac{c^P}{(P-1)!}\int_0^1(1-t)^{P-1}(\partial^Pa)
(\lambda+tc)\,dt\). The family \(\partial^Pa\) has order \(<-n\), uniformly
on strips, so by Lemma~\ref{lem:bartr}(1) we have
\(\overline\Tr(R)=\frac1{2\pi}\int\Tr R\,d\lambda\). A contour shift on the
strip turns \(\int\Tr(\partial^Pa)(\lambda+tc)\,d\lambda\) into
\(\int\partial_\lambda\Tr(\partial^{P-1}a)\,d\lambda\). This vanishes,
because \(\partial^{P-1}a\) has order \(<-d\) and its trace tends to \(0\)
at \(\pm\infty\).
\end{proof}

For \(A\in\Adb\) with \(T_H(A)=\sum x^ka_k\), put
\(\widehat\Tr_{\partial,H}(A)=\overline\Tr(a_0)\). By
Lemma~\ref{lem:bartr}(1), this extends \(\Tr_{\partial,H}\) from \(\Idb\).

\begin{proposition}[The connecting cocycle]\label{prop:omega}
For \(A,B\in\Adb\) with \(T_H(A)=\sum x^ka_k\) and \(T_H(B)=\sum x^kb_k\),
\begin{equation}\label{eq:omega}
 \omega_H(A,B):=\widehat\Tr_{\partial,H}([A,B])
 =\sum_{k\neq0}\sum_{p\geq1}\frac{(ki)^p}{p!}\,
 \overline\Tr\bigl(\partial_\lambda^p(a_k\,\alpha^k(b_{-k}))\bigr),
\end{equation}
a finite sum. It vanishes if \(A\) or \(B\) lies in \(\Idb\), so it is a
Hochschild \(1\)-cocycle on \(\Adsb\). In the long exact sequence of
\(0\to\Idb\to\Adb\to\Adsb\to0\) we have
\(\Tr_{\partial,H}\circ\partial=\langle\omega_H,\cdot\rangle\) on
\(\HH_1(\Adsb)\). The cocycle
\[
 \Omega_H(A,B,C)=-\omega_H(A\,D(B),C)
\]
on \(\Adsb\) satisfies \(\varphi_H\circ\partial=\langle\Omega_H,\cdot\rangle\)
on \(\HH_2(\Adsb)\).
\end{proposition}

\begin{proof}
By \eqref{eq:skew}, the constant term of \([A,B]\) is
\(\sum_k(\alpha^{-k}(a_k)b_{-k}-\alpha^k(b_{-k})a_k)\). Put
\(F_k=a_k\alpha^k(b_{-k})\). Then \(\alpha^{-k}(F_k)=\alpha^{-k}(a_k)b_{-k}\),
and \(\overline\Tr(\alpha^k(b_{-k})a_k)=\overline\Tr(F_k)\) by
Lemma~\ref{lem:bartr}(2). Since \(\alpha^{-k}F(\lambda)=F(\lambda+ki)\),
Lemma~\ref{lem:shift} gives \eqref{eq:omega}. If \(a_k\in B_H\) or
\(b_{-k}\in B_H\), then \(F_k\in B_H\), and each term vanishes by
Lemma~\ref{lem:bartr}(3). The ideal \(\Idb\) is H-unital, so
Lemma~\ref{lem:conn} applies with \(\tau=\Tr_{\partial,H}\) and
\(\hat\tau=\widehat\Tr_{\partial,H}\).

For \(\varphi_H\), take \(\hat\varphi(A,B)=\widehat\Tr_{\partial,H}(AD(B))\).
It extends \(\varphi_H\), because \(D\) preserves \(\Idb\). A direct
expansion gives \(b\hat\varphi(A,B,C)=-\widehat\Tr([AD(B),C])\), which
vanishes when an argument lies in \(\Idb\). Lemma~\ref{lem:conn} then
gives the last assertion.
\end{proof}

\begin{proposition}[Nonvanishing for \(n=1\)]\label{prop:n1}
Let \(n=1\), so that each \(H\) is a point and \(B'_H\) is the algebra of
entire functions with classical symbol expansions on strips. Let
\(\pi_+(\lambda)=\frac12(1+\operatorname{erf}\lambda)\in B'_H\), and work
in the summand of \(H\). Then
\[
 c_H=x^{-1}\pi_+\otimes x,\qquad
 c'_H=x^{-1}\pi_+\otimes h\otimes x-x^{-1}\pi_+\otimes x\otimes h
\]
are cycles in \(\Adsb\), and
\(\langle\omega_{H'},c_H\rangle=-\frac{i}{2\pi}\delta_{HH'}\) and
\(\langle\Omega_{H'},c'_H\rangle=-\frac1{2\pi}\delta_{HH'}\). Hence both
connecting maps \(\HH_1(\Adsb)\to\HH_0(\Idb)\) and
\(\HH_2(\Adsb)\to\HH_1(\Idb)\) are surjective.
\end{proposition}

\begin{proof}
By \eqref{eq:skew}, \((x^{-1}\pi_+)x=\alpha(\pi_+)\) and
\(x(x^{-1}\pi_+)=\pi_+\). Hence \(bc_H=\pi_+(\lambda-i)-\pi_+(\lambda)\).
This lies in \(\cE\), since \(\operatorname{erf}\) tends to \(\pm1\) on
strips with Gaussian error, so \(c_H\) is a cycle in \(\Adsb\). In
\eqref{eq:omega} only \(k=-1\) occurs, with \(F_{-1}=\pi_+\). Here
\(\overline\Tr(\pi_+')=\frac1{2\pi}\), while
\(\overline\Tr(\pi_+^{(p)})=0\) for \(p\geq2\). Therefore
\(\omega_H(c_H)=-i/2\pi\).

For \(c'_H\), use \(hx=x(h-i)\) and \(hx^{-1}=x^{-1}(h+i)\), which follow
from \eqref{eq:skew}. A direct computation gives
\(bc'_H=(\pi_+-\alpha(\pi_+))\otimes h\), which is zero in \(\Adsb\). Since
\(D(h)=i\) and \(D(x)=0\),
\(\Omega_H(c'_H)=-\omega_H(i\,x^{-1}\pi_+,x)=-\frac1{2\pi}\).
\end{proof}

\subsection{The connecting maps for \texorpdfstring{\(n\geq2\)}{n>=2}}
Identify the suspended variable \(\lambda\) with the b-covariable \(\xi\);
this is legitimate because \(xD_x\) has symbol \(\xi\) and indicial family
\(\lambda\). Then principal symbols of elements of \(B'_H\) are homogeneous
functions on \({}^bT^*_HX\setminus0=(T^*H\times\mathbb R_\xi)\setminus0\).
Write \(\zeta=(\eta,\xi)\) and \(d=n-1=\dim H\), and put
\[
 \hat\rho_H(g)=(2\pi)^{-n}\int_H\int_{\mathbb R^d}\bigl(g(y,\eta,1)-
 g(y,\eta,-1)\bigr)\,d\eta\,\nu_\partial(y)
\]
for \(g\) homogeneous of degree \(1-n\), with \(d\eta\) the density dual to
\(\nu_\partial\). The integrand is \(O(|\eta|^{-n})\), so the integral
converges.

\begin{lemma}[Residue formula]\label{lem:rhohat}
Let \(G\in B'_H\). If \(G\) has order \(\leq-n\), then
\(\overline\Tr(\partial_\lambda^pG)=0\) for \(p\geq1\). If \(G\) has order
\(1-n\) with principal symbol \(g\), then
\(\overline\Tr(\partial_\lambda G)=\hat\rho_H(g)\) and
\(\overline\Tr(\partial_\lambda^pG)=0\) for \(p\geq2\).
\end{lemma}

\begin{proof}
If the order is \(\leq-n\), then \(\partial^pG\) has order \(<-n\), so
\(\overline\Tr(\partial^pG)=\frac1{2\pi}\int\partial_\lambda\Tr(\partial^{p-1}
G)\,d\lambda=0\) by Lemma~\ref{lem:bartr}. Now let \(G\) have order
\(1-n=-d\). The case \(p\geq2\) is the previous one applied to
\(\partial G\). Changing \(G\) by a family of order \(-n\) does not change
\(\overline\Tr(\partial G)\). Since \(\overline\Tr\) is defined for
suspended families on the real line, and its definition does not use
holomorphy, we may take \(G=\operatorname{Op}(\chi g)\) for real
\(\lambda\), quantized in charts, with \(\chi\) a cut-off near \(\zeta=0\).
This family is not entire. From the
definition of \(\overline\Tr\) applied to \(a=\partial G\) (order \(-n\)) with
\(p=1\): here \(g_1(\lambda)=\Tr\,\partial G(\lambda)-\Tr\,\partial G(0)\), and the
term \(2T\Tr\,\partial G(0)\) has no constant part. Hence
\(\overline\Tr(\partial G)\) is \(\frac1{2\pi}\) times the constant term of
\(I(T)=\int_{-T}^T\Tr\,\partial_\lambda G\,d\lambda\). Since
\(\partial_\lambda G\) has order \(-n<-d\), its trace is the integral of its
symbol, \(\Tr\,\partial_\lambda G=(2\pi)^{-d}\iint\partial_\lambda(\chi g)\,
d\eta\,\nu_\partial\). This is absolutely integrable over
\(\{|\lambda|\leq T\}\times\mathbb R^d\). By Fubini,
\[
 I(T)=(2\pi)^{-d}\int_H\int_{\mathbb R^d}\bigl(g(\eta,T)-g(\eta,-T)\bigr)\,
 d\eta\,\nu_\partial
\]
for \(T\) large. Substituting \(\eta=T\eta'\) and using homogeneity of
degree \(-d\) shows that \(I(T)\) is independent of \(T\), and equal to
\((2\pi)\hat\rho_H(g)\).
\end{proof}

For a chain \(c\) of \(\Adsb\), let \(\chi(c)\) be its HKR form, computed on
principal symbols. Let \(\theta_x\) and \(\theta_{\xi x}\) denote the
coefficients of \(dx/x\) and of \(d\xi\wedge dx/x\) in the \(x\)-weight-zero
part of a form \(\theta\).

\begin{lemma}[Top-order formulas]\label{lem:top}
\begin{enumerate}
\item[(a)] \(\omega_H\) vanishes on \(1\)-chains of total order \(\leq-n\).
On \(1\)-chains of total order \(\leq1-n\),
\(\omega_H(c)=-i\,\hat\rho_H\bigl(\chi(c_{1-n})_x\bigr)\).
\item[(b)] \(\Omega_H\) vanishes on \(2\)-chains of total order
\(\leq1-n\). Assume (C-HKR). For \(2\)-chains \(c\) of total order
\(\leq2-n\) with \(b_0c_{2-n}=0\),
\(\Omega_H(c)=-\hat\rho_H\bigl(\chi(c_{2-n})_{\xi x}\bigr)\).
\end{enumerate}
\end{lemma}

\begin{proof}
(a) In \eqref{eq:omega} the term \(F_k=a_k\alpha^k(b_{-k})\) has order at most
the total order, and \(\alpha^k\) does not change principal symbols.
Lemma~\ref{lem:rhohat} gives
\(\omega_H(A\otimes B)=\sum_kik\,\hat\rho_H(\sigma(a_k)\sigma(b_{-k}))\)
in total order \(1-n\), and \(0\) below. On the other hand
\(A\,dB=\sum x^ka_k\,d(x^{-k}b_{-k})\) has weight-zero \(dx/x\)-coefficient
\(-\sum_kk\,a_kb_{-k}\).

(b) The derivation \(D\) lowers order by one, so \(\Omega_H(A,B,C)=
-\omega_H(AD(B),C)\) is computed by (a). On elementary tensors this gives
\(\Omega_H(a\otimes b\otimes c)=-\sum m\,\hat\rho_H(a_j\partial_\xi b_lc_m)\),
summed over \(j+l+m=0\). Because \(\Omega_H\) is a cocycle and vanishes in
lower order, it vanishes on \(b_0\)-boundaries of top components. By
(C-HKR), a top \(b_0\)-cycle is \(b_0\)-homologous to the
antisymmetrization of its HKR form, and on
\(a\otimes b\otimes c-a\otimes c\otimes b\), whose HKR form is
\(a\,db\wedge dc\), a direct comparison gives
\(-\hat\rho_H((a\,db\wedge dc)_{\xi x})\).
\end{proof}

\begin{proposition}[The boundary connecting maps]
\label{prop:dict}
Let \(n\geq1\), and assume (C) for \(\Adsb\), part (ii) of
Theorem~\ref{thm:full} (hence (Q1), (Q2), (S1)), and
Lemma~\ref{lem:conn} (hence (Q1)). Then both connecting maps
\(\HH_1(\Adsb)\to\HH_0(\Idb)\) and \(\HH_2(\Adsb)\to\HH_1(\Idb)\) are
surjective onto \(\C^N\). They are given on \(E^2\) by the functionals
\(\theta\mapsto-i\hat\rho_H(\theta_x)\) and
\(\theta\mapsto-\hat\rho_H(\theta_{\xi x})\).
\end{proposition}

\begin{proof}
By Proposition~\ref{prop:omega}, Lemma~\ref{lem:top}, and the isomorphism
\(\Theta\) of Proposition~\ref{prop:algconv}, the composites
\(\Tr_{\partial,H}\circ\partial_1\) and \(\varphi_H\circ\partial_2\) are the
functionals \(\theta\mapsto-i\hat\rho_H(\theta_x)\) and
\(\theta\mapsto-\hat\rho_H(\theta_{\xi x})\) on \(E^2\). We exhibit
\(\delta_b\)-closed forms on which these are nonzero. Fix a metric on
\(H\), let \(|\zeta|^2=|\eta|^2+\xi^2\), and let \(\alpha_H=\eta\cdot dy\) be
the tautological form. Put
\[
 f=1_H\,\xi|\zeta|^{-n},\qquad
 \theta_1=f\,\frac{dx}{x}+1_H|\zeta|^{-n}\alpha_H,\qquad
 \theta_2=f\,d\xi\wedge\frac{dx}{x}+1_H|\zeta|^{-n}\,d\xi\wedge\alpha_H .
\]
These have fibre weights \(1-n\) and \(2-n\) and \(x\)-weight \(0\). With
\(\{a,c\}_b=\pi_b(da,dc)\), \eqref{eq:poisson-boundary} gives
\(\{f,\log x\}_b=-\partial_\xi f\), \(\{h,y_j\}_b=-\partial_{\eta_j}h\),
\(\{f,\xi\}_b=0\) and \(\{\xi,\log x\}_b=-1\). Hence
\[
 \delta_b\theta_1=-\operatorname{div}_\zeta V,\qquad
 \delta_b\theta_2=(\operatorname{div}_\zeta V)\,d\xi,\qquad
 V=\zeta|\zeta|^{-n}.
\]
Since \(\zeta\) ranges over \(\mathbb R^n\setminus0\), the field \(V\) is
divergence free:
\(\operatorname{div}V=n|\zeta|^{-n}-n|\zeta|^{-n-2}|\zeta|^2=0\). This holds
in any coordinates, because
\[
 \sum_j\eta_j\partial_{\eta_j}|\zeta|^2+\xi\partial_\xi|\zeta|^2=2|\zeta|^2 .
\]
So \(\theta_1\) and \(\theta_2\) define
\(E^2\)-classes. Now \((\theta_1)_x=(\theta_2)_{\xi x}=f\), and
\[
 \hat\rho_{H'}(f)=\delta_{HH'}\,2(2\pi)^{-n}\int_H\int_{\mathbb R^d}
 (1+|\eta|^2)^{-n/2}\,d\eta\,\nu_\partial\neq0 .
\]
This proves surjectivity.
\end{proof}

\begin{remark}[Description of the connecting maps; sketch (S2)]
\label{rem:description}
The identification of these functionals with
\(\beta\mapsto(\int_{Z_H}\beta_\partial)_H\) and
\(\beta\mapsto(\int_{Z_H}\beta_0)_H\) is only sketched. Write
\(\varpi_H=d\eta\wedge dy\) for the symplectic form of \(T^*H\). Then
\(\pi_b^\sharp(dx/x)\) is a multiple of \(\partial_\xi\), and
\(\mu_b=d\xi\wedge\frac{dx}{x}\wedge\varpi_H^d/d!\). Hence
\(*_b(g\,dx/x)\) and \(*_b(g\,d\xi\wedge dx/x)\) are, up to signs, \(g\,
\frac{dx}{x}\wedge\varpi_H^d/d!\) and \(g\,\varpi_H^d/d!\), while the
\(dy_j\)-components are carried to forms containing \(d\xi\). The integrals
over the hyperplanes \(\xi=\pm1\) should then be compared with the
integral over \(Z_H\) by radial projection. The hyperplanes are not compact.
The two integrals separately diverge logarithmically near the equator
\(\xi=0\), and only the difference \(g(\eta,1)-g(\eta,-1)\) in
\(\hat\rho_H\) converges. A precise proof needs this cancellation and the
exact signs and constants; we have not written it out. The surjectivity in
Proposition~\ref{prop:dict} does not depend on this remark. The exact rank
\(N-1\) of \(\partial^{\mathcal A}_2\), and the explicit kernels in
Theorem~\ref{thm:full}(vii)--(viii), do.
\end{remark}

\begin{theorem}\label{thm:Ad}
Assume (C) and Theorem~\ref{thm:full}(ii). Then
\(\HH_k(\Adb)\cong\HH_k(\Adsb)\) for \(k\neq1,2\). For \(k=1,2\), the group
\(\HH_k(\Adb)\) is the kernel of the surjection of
Proposition~\ref{prop:n1} (\(n=1\)), resp.\ Proposition~\ref{prop:dict}.
Its explicit description in Theorem~\ref{thm:full}(vii) uses (S2).
\end{theorem}

\begin{proof}
Use the long exact sequence of Theorem~\ref{thm:excision} together with
Theorem~\ref{thm:Id}. Surjectivity of
\(\HH_2(\Adsb)\to\HH_1(\Idb)\) makes \(\HH_1(\Idb)\to\HH_1(\Adb)\) zero.
Surjectivity of \(\HH_1(\Adsb)\to\HH_0(\Idb)\) makes
\(\HH_0(\Idb)\to\HH_0(\Adb)\) zero. For \(k\geq3\) the groups of \(\Idb\)
vanish.
\end{proof}

\section{\texorpdfstring{The algebra \(\Ab\) and the full operator algebra}{The algebra Ab and the full operator algebra}}\label{sec:A}

\begin{lemma}\label{lem:fibre}
\(\Isb\cap\Idb=0\) in \(\Ab\), and \(\Ab\to\Adb\times_{\Adsb}\Asb\) is an
isomorphism.
\end{lemma}

\begin{proof}
\(x^\infty\Psi_b^{\Z}\cap x^{-\Z}\Psi_b^{-\infty}=x^\infty\Psi_b^{-\infty}\).
For any ring and ideals \(I,J\), the ring \(R/(I\cap J)\) is the fibre product
\(R/I\times_{R/(I+J)}R/J\).
\end{proof}

\begin{proposition}\label{prop:connA}
Let \(\partial^{\mathcal A}_k:\HH_k(\Asb)\to\HH_{k-1}(\Idb)\) be the connecting
maps of \(0\to\Idb\to\Ab\to\Asb\to0\), and let \(p:\Asb\to\Adsb\) be the
quotient map.
\begin{enumerate}
\item[(a)] \(\partial^{\mathcal A}_k=\partial^{\partial}_k\circ p_*\), where
\(\partial^\partial\) denotes the connecting maps of \(\Adb\).
\item[(b)] The composite \(\HH_1(\Idb)\to\HH_1(\Ab)\to\HH_0(\Ib)\), where
the second map is the connecting map of
\(0\to\Ib\to x^{-\Z}\Psi_b^{\Z}\to\Ab\to0\), equals
\(-\sum_H\varphi_H\).
\item[(c)] The image of \(\partial^{\mathcal A}_2\) is contained in
\(\ker\sum_H\varphi_H\), so its rank is at most \(N-1\).
\item[(d)] If \(n=1\), then \(\partial^{\mathcal A}_1\) is surjective.
\end{enumerate}
\end{proposition}

\begin{proof}
(a) and (b) express the naturality of the excision sequence for the
morphisms of extensions \((\Idb\subset\Ab)\to(\Idb\subset\Adb)\) and
\((\Ib\subset x^{-\Z}\Psi_b^{-\infty})\to(\Ib\subset x^{-\Z}\Psi_b^{\Z})\).
Part (b) then follows from Theorem~\ref{thm:Laurentres}.

(c) The composite \(\HH_2(\Asb)\to\HH_1(\Idb)\to\HH_1(\Ab)\) is zero by
exactness, and the second map followed by the connecting map of (b) is
\(-\sum\varphi_H\).

(d) Choose \(P_H\in\Psi_b^0(X)\), supported near \(H\), with full symbol
\(\chi_H(x)1_{\xi>0}\) and with no lower-order terms. Then \(x^{-1}P_Hx-P_H\)
is smoothing, because the full symbol of \(x^{-1}Px\) is
\(\sum_k\frac{(-i)^k}{k!}\partial_\xi^kp\), by Lemma~\ref{lem:conj} applied
to symbols, and \(\partial_\xi p=0\) for \(\xi\neq0\). Hence
\(x^{-1}P_H\otimes x\) is a cycle in \(\Asb\). Its image under \(p\)
differs from \(c_H\) of Proposition~\ref{prop:n1} only by elements of
\(\Idb\). Indeed, the b-covariable \(\xi\) is the symbol of \(xD_x\), whose
indicial family is \(\lambda\), so \(I_H(P_H)-\pi_+\in\cE\). Thus
\(\Tr_{\partial,H'}\partial^{\mathcal A}_1[x^{-1}P_H\otimes x]=
-\frac{i}{2\pi}\delta_{HH'}\).
\end{proof}

\begin{remark}[Several boundary components]\label{rem:MNcomponents}
Assume (C) and (S2). By Proposition~\ref{prop:dict},
Remark~\ref{rem:description}, the naturality of \(\Theta\),
and the identity part of
\(p_*\), the map \(\partial^{\mathcal A}_2\) is
\(\beta\mapsto(\int_{Z_H}\beta)_H\) on the summand \(H^{2n-2}(Y)\) of
\(\HH_2(\Asb)\), and it vanishes on the other summands. By Stokes' theorem,
\(\sum_H\int_{Z_H}\beta=\int_Yd\beta=0\), which recovers the bound of
Proposition~\ref{prop:connA}(c). Exactness of
\(H^{2n-2}(Y)\to H^{2n-2}(\partial Y)\to H^{2n-1}(Y,\partial Y)\), together
with Lefschetz duality \(H^{2n-1}(Y,\partial Y)\cong H_0(Y)\) (C3), shows
that the image is the kernel of the sum map
\(H^{2n-2}(\partial Y)\to H_0(Y)\). If \(X\) is connected and \(n\geq2\),
then \(Y\) is connected, \(H^{2n-2}(\partial Y)=\C^N\), and the rank is
\(N-1\). If \(X\) has \(c\) components, each with nonempty boundary, the rank is
\(N-c\), and \(\operatorname{coker}\partial^{\mathcal A}_2\cong\C^c\). In the cusp case
Melrose and Nistor find \(\partial[i_{\log x}\Tr_\partial]=0\)
\cite[Prop.~12]{MN}, because \(d\theta_\sigma\) extends over the whole
cosphere bundle. For each \(H\) separately, the class
\(d\theta_\sigma\cdot1_{Z_H}\) does not extend when \(N\geq2\); only the
sum over all \(H\) does. We mention this only for comparison.
\end{remark}

\begin{proof}[Proof of Theorem~\textup{\ref{thm:full}}]
(i) is Proposition~\ref{prop:I}, (ii) is Theorem~\ref{thm:Id}, and (iii) is
Theorem~\ref{thm:Laurentres}. (iv) and (vi) are
Propositions~\ref{prop:Is} and~\ref{prop:Ads}, and (v) is
Theorem~\ref{thm:main}. (vii) is Theorem~\ref{thm:Ad}, with the components
identified by Proposition~\ref{prop:dict}. For (viii), insert
\(\partial^{\mathcal A}_1=\partial^\partial_1\circ p_*\) and
\(\partial^{\mathcal A}_2\) into the long exact sequence. By
Proposition~\ref{prop:dict}, Remark~\ref{rem:description} (S2), the
naturality of the top-component map \(\Theta\) (on \(E^2\), \(p_*\) is
restriction to the boundary), and the identity part of \(p_*\),
\(\partial^{\mathcal A}_1\) is
\(\beta\mapsto(\int_{Z_H}\beta)_H\) on the summand
\(H^{2n-2}(Z)\cong\C^N\) of \(\HH_1(\Asb)\) and vanishes on the other
summands. The rank of \(\partial^{\mathcal A}_2\) is \(N-1\)
(Remark~\ref{rem:MNcomponents}). Use also \(\HH_*(\Idb)\). This gives \(\HH_0(\Ab)\cong\HH_0(\Asb)\),
\(\HH_1(\Ab)\cong\operatorname{coker}\partial^{\mathcal A}_2\oplus
\ker\partial^{\mathcal A}_1\), and \(\HH_2(\Ab)\cong\ker\partial^{\mathcal
A}_2\). (ix) follows from Corollary~\ref{cor:IF} and the long exact
sequence of \(0\to\Ib\to x^{-\Z}\Psi_b^{\Z}\to\Ab\to0\), using (i).
\end{proof}

\section{The index functional}\label{sec:index}

\begin{lemma}[Inverses of indicial families]\label{lem:autoinv}
Let \(a\in B'_H\) have order \(m\) and be elliptic. If \(a(\lambda)\) is
invertible for every \(\lambda\in\C\), then \(a^{-1}\in B'_H\). If
\(a=1+k\) with \(k\in B_H\) and \(a(\lambda)\) is invertible for every
\(\lambda\in\C\), then \(a^{-1}-1\in B_H\).
\end{lemma}

\begin{proof}
Fix a strip \(S=\{|\operatorname{Im}\lambda|\leq C\}\). The suspended
calculus (C3) gives \(q\in\Psi^{-m}_{\mathrm{sus}}\) on \(S\) with
\(aq=1-r\), where \(r\) is smoothing and decreases rapidly, in all
Sobolev norms, as \(|\operatorname{Re}\lambda|\to\infty\). The family
\(q\) need not be holomorphic. For \(|\operatorname{Re}\lambda|\geq R\)
the Neumann series gives \(a^{-1}=q(1-r)^{-1}=q+q\,r(1-r)^{-1}\). For
\(|\operatorname{Re}\lambda|\leq R\), the identity \(a^{-1}=q+a^{-1}r\)
holds with \(a^{-1}\) holomorphic, hence bounded on that compact set. So
\(a^{-1}-q\) is smoothing with the decay of \(r\), and \(a^{-1}\) is entire
because \(a\) is. Hence \(a^{-1}\in B'_H\). In the second case, \(a^{-1}\)
is uniformly bounded on each \(H^s\) over \(S\): use the Neumann series
where \(\|k\|<\tfrac12\), and compactness elsewhere. Then
\(a^{-1}-1=-k+ka^{-1}k\) is smoothing and decreases rapidly on \(S\).
\end{proof}

\begin{lemma}\label{lem:invertible}
Let \(A\in\Psi_b^m(X)\) be b-elliptic. If for every \(H\) and every
\(\lambda\in\C\) the operator \(I_H(A,\lambda)\) is invertible, then
\([A]\) is invertible in \(\Ab\).
Conversely, if \([A]\) is invertible in \(\Ab\), then \(I_H(A,\lambda)\) is
invertible for every \(\lambda\in\C\).
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:fibre} it suffices to invert \(A\) in \(\Asb\), where the
usual symbolic parametrix construction works, and in \(\Adb\). Write
\(T_H(A)=\sum_{k\geq0}x^ka_k\) with \(a_0=I_H(A)\). By \eqref{eq:skew}, the
coefficient of \(x^k\) in \(T(A)\sum x^lb_l\) is
\(\alpha^k(a_0)b_k+\sum_{j\geq1}\alpha^{k-j}(a_j)b_{k-j}\). Solve recursively,
using \(\alpha^k(a_0)^{-1}=\alpha^k(a_0^{-1})\in B'_H\)
(Lemma~\ref{lem:autoinv}). A left inverse is
constructed in the same way, and the two inverses agree. Conversely, let
the inverse be \(\sum_{l\geq-\ell}x^lb_l\). Its lowest coefficient
satisfies \(b_{-\ell}a_0=0\). Since \(a_0(\lambda)\) is invertible for large
\(|\operatorname{Re}\lambda|\), analytic continuation gives
\(b_{-\ell}=0\) if \(\ell>0\). Hence \(b_0a_0=a_0b_0=1\) on all of \(\C\).
\end{proof}

\begin{remark}[Dirac operators]
A b-Dirac operator of product type, \(c(dx/x)(x\partial_x+A_\partial)\), has
\(I(\lambda)=c(dx/x)(i\lambda+A_\partial)\), which fails to be invertible at
\(\lambda=i\mu\) for every \(\mu\in\operatorname{spec}A_\partial\). It is
therefore never invertible in \(\Ab\). Its index on \(x^\delta L^2_b\) depends
on \(\delta\), and the algebra \(\Ab\) cannot see this dependence. This is
the precise form of the ``problems with completeness'' that led
Melrose--Nistor to the cusp calculus \cite[p.~3]{MN}, where indicial
families need only be invertible on \(\mathbb R\).
\end{remark}

\begin{proposition}[Fedosov]\label{prop:fedosov}
Let \(A\in x^{-\ell}\Psi_b^m(X)\) and \(B\in x^{-\Z}\Psi_b^{\Z}(X)\) with
\(AB-1,BA-1\in\Ib\). Then
\(A:x^\delta H^s_b\to x^{\delta-\ell}H^{s-m}_b\) is Fredholm for all
\(\delta,s\), with kernel and cokernel in \(\dot C^\infty(X)\) and index
independent of \(\delta\) and \(s\). Moreover
\[
 \operatorname{Ind}(A)=\Tr(AB-BA)=-\Tr\bigl(\partial[B\otimes A]\bigr),
\]
where \(B\otimes A\) is a \(1\)-cycle of \(\Ab\).
\end{proposition}

\begin{proof}
Operators in \(\Ib\) have kernels in \(\dot C^\infty(X^2)\), so they are
trace class on every weighted Sobolev space and map into
\(\dot C^\infty(X)\). Hence \(\ker A\subset\ker BA\subset\dot C^\infty\),
and similarly for \(A^*\). The mapping properties of b-operators on
weighted b-Sobolev spaces, and the Calder\'on--Fedosov formula, are inputs
(C3). The latter gives
\(\operatorname{Ind}A=\Tr(1-BA)-\Tr(1-AB)\). Finally,
\(b(B\otimes A)=BA-AB\) lifts the boundary.
\end{proof}

\begin{theorem}[Residual perturbations of the identity]\label{thm:WH}
Let \(K\in\Psi_b^{-\infty}(X)\), and suppose that for every \(H\) the family
\(a_H(\lambda)=1+I_H(K,\lambda)\) is invertible for every \(\lambda\in\C\).
Then \(A=1+K\) is Fredholm on every
\(x^\delta H^s_b\), and
\[
 \operatorname{Ind}(A)
 =\frac{i}{2\pi}\sum_H\int_{\mathbb R}\Tr_H\bigl(a_H^{-1}\partial_\lambda a_H
 \bigr)\,d\lambda
 =-\frac12\sum_H\eta_b(a_H)
 =-\sum_H\operatorname{wind}\bigl(\det a_H\bigr),
\]
where \(\eta_b(a)=\frac1{\pi i}\int_{\mathbb R}\Tr(a^{-1}\partial_\lambda a)
\,d\lambda\), and \(\operatorname{wind}\) is the winding number of the
Fredholm determinant along \(\mathbb R\).
\end{theorem}

\begin{proof}
The symbol of \(A\) is \(1\), and \(a_H^{-1}-1\in B_H\) by
Lemma~\ref{lem:autoinv}. By Lemma~\ref{lem:invertible}, the inverse in
\(\Adb\) is \(1+\sum_{k\geq0}x^kl_k\) with \(l_0=a_H^{-1}-1\) and all
\(l_k\in B_H\). By Proposition~\ref{prop:structure}, there is
\(L\in\Psi_b^{-\infty}\) with this Taylor series, and then
\(B=1+L\) satisfies \(AB-1,BA-1\in\Ib\). By
Proposition~\ref{prop:fedosov} and Theorem~\ref{thm:defect},
\[
 \operatorname{Ind}A=\Tr([K,L])={}^b\Tr([K,L])
 =\sum_H\Tr_{\partial,H}(D(K)L).
\]
Since \(T(D(K))\) and \(T(L)\) contain only nonnegative powers of \(x\),
the constant term of their product is
\(i\partial_\lambda a_H\cdot(a_H^{-1}-1)\). The integral of
\(\Tr\,\partial_\lambda a_H\) vanishes. Moreover
\(\Tr(a^{-1}\partial a)=\partial_\lambda\log\det a\), and
\(\det a_H\to1\) at \(\pm\infty\), so the integral equals
\(2\pi i\,\operatorname{wind}\).
\end{proof}

\begin{example}[Index \(-1\)]\label{ex:WH}
Let \(e_H\) be a rank-one projection on \(H\), and put
\(a_H(\lambda)=1+(e^{2\pi i\pi_+(\lambda)}-1)e_H\), with \(\pi_+\) as in
Proposition~\ref{prop:n1}. Then \(e^{\pm2\pi i\pi_+}-1\in\cE\),
\(a_H^{-1}=1+(e^{-2\pi i\pi_+}-1)e_H\), and \(\det a_H=e^{2\pi i\pi_+}\) winds
once. Taking \(a_{H'}=1\) for \(H'\neq H\), the operator \(1+K\) has index
\(-1\). It is not a compact perturbation of the identity on \(L^2_b\),
because its normal operator is translation invariant in \(u=\log x\). Near
\(H\), in the variable \(u\), it is a Wiener--Hopf operator on a half-line
with symbol \(a_H\). The theorem is the b-calculus form of the
Gohberg--Krein index formula \cite{GohbergKrein}.
\end{example}

\begin{corollary}\label{cor:IF}
The index functional \(\Tr\circ\partial:\HH_1(\Ab)\to\C\) is nonzero.
\end{corollary}

\begin{proof}
For the operator of Example~\ref{ex:WH}, Proposition~\ref{prop:fedosov}
gives \(\Tr\partial[B\otimes A]=-\operatorname{Ind}(A)=1\).
\end{proof}

\begin{proposition}[General index formula; sketch (S3)]\label{prop:indexgeneral}
Let \(A\in\Psi_b^m(X)\) satisfy the hypotheses of
Lemma~\ref{lem:invertible}, with inverse \(B\) modulo \(\Ib\). Choose a
positive elliptic \(Q\in\Psi_b^1(X)\), dilation invariant near
\(\partial X\). Suppose that \(\Tr(x^wCQ^{-\tau})\) is meromorphic near
\((0,0)\) with at most simple poles in each variable, with \(Q^{-\tau}\) taken
from the full b-calculus. Then
\(\operatorname{Ind}A=\operatorname{FP}_{\tau=0}\operatorname{FP}_{w=0}
\Tr(x^w[A,B]Q^{-\tau})\) splits into a residue term \(\operatorname{AS}(A)\)
in the symbols of \(A\) and \(B\), and a boundary term. If \(A\) is
dilation invariant near \(\partial X\), the boundary term is
\[
 \frac{i}{2\pi}\sum_H\overline{\int}\Tr_H\bigl(I_H(A)^{-1}\partial_\lambda
 I_H(A)\bigr)\,d\lambda=-\frac12\sum_H\eta_b(I_H(A)),
\]
the regularized form of Theorem~\ref{thm:WH}.
\end{proposition}

\begin{proof}[Sketch]
Write \(\Tr([A,B]x^wQ^{-\tau})\) as the sum of the trace defects of \(x^w\)
and of \(Q^{-\tau}\), as in the proof of Theorem~\ref{thm:defect}. The
\(x^w\)-defect gives \(\widehat\Tr_\partial(D(A)B)\), regularized by
\(Q^{-\tau}\). For dilation-invariant \(A\) only the constant terms
contribute, and the computation of Theorem~\ref{thm:WH} goes through with
\(\overline{\Tr}\) in place of \(\Tr\). The \(Q^{-\tau}\)-defect is a residue
term. The analytic input, meromorphy of the double zeta function with
complex powers from the full calculus, is not proved here.
\end{proof}
